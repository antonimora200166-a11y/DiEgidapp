function instalar() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var rep = [];
  pasoSeguro_(rep, 'CONFIG', function () { asegurarConfig_(ss); });
  pasoSeguro_(rep, 'Libros mayores', function () {
    asegurarHoja_(ss, H.LOTES, COL_LOTE); asegurarHoja_(ss, H.CONS, COL_CONS);
    asegurarHoja_(ss, H.BIT, ['TIMESTAMP', 'USUARIO', 'ACCIÓN', 'DETALLE', 'MOTIVO', 'VERSIÓN']);
    asegurarHoja_(ss, H.FAC, ['INSUMO', 'UNIDAD_RECEPCION', 'FACTOR_A_BASE', 'NOTA']);
  });
  pasoSeguro_(rep, 'Semilla v1.1 (catálogo, cócteles, parámetros, taras)', function () { rep.push('   ' + sembrarV11_(ss)); });
  pasoSeguro_(rep, 'Hoja CAPTURA', function () { construirCaptura_(ss); });
  pasoSeguro_(rep, 'Instrucciones', function () { construirInstrucciones_(ss); });
  pasoSeguro_(rep, 'Disparadores', function () { instalarDisparadores_(ss); });
  pasoSeguro_(rep, 'Costear histórico', function () { rep.push('   ' + completarCostosPendientes_(true)); });
  pasoSeguro_(rep, 'Derivados', function () { recalcularDerivados_(ss); });
  pasoSeguro_(rep, 'Hoja PESAJE', function () { rep.push('   ' + construirPesaje_(ss, '') + ' medida(s) pendiente(s)'); });
  bitacora_('INSTALACIÓN', rep.join(' | '), '');
  alerta_('Instalación v' + VERSION + '\n\n' + rep.join('\n'));
}

function pasoSeguro_(lista, nombre, fn) {
  try { fn(); lista.push('✔ ' + nombre); SpreadsheetApp.flush(); }
  catch (e) { lista.push('✘ ' + nombre + ': ' + e.message); }
}

function instalarDisparadores_(ss) {
  var t = ScriptApp.getProjectTriggers();
  for (var i = 0; i < t.length; i++) {
    var f = t[i].getHandlerFunction();
    if (f === 'alEditar' || f === 'enviarAlertaDiaria') { ScriptApp.deleteTrigger(t[i]); }
  }
  ScriptApp.newTrigger('alEditar').forSpreadsheet(ss).onEdit().create();
  ScriptApp.newTrigger('enviarAlertaDiaria').timeBased().everyDays(1).atHour(22).create();
}

function alEditar(e) {
  if (!e || !e.range) { return; }
  var sh = e.range.getSheet();
  var a1 = e.range.getA1Notation();
  if (sh.getName() === H.PES) {
    if (a1 === PES.GUARDAR && e.range.getValue() === true) {
      try { aplicarPesaje_(sh); }
      catch (errP) { sh.getRange(PES.MSG).setValue('ERROR: ' + errP.message + ' — avisa a Administración.'); sh.getRange(PES.GUARDAR).setValue(false); bitacora_('ERROR', 'PESAJE: ' + errP.message, ''); }
    }
    return;
  }
  if (sh.getName() !== H.CAP) { return; }
  try {
    if (a1 === CAP.MPI) { cargarFicha_(sh); }
    else if (a1 === CAP.GUARDAR && e.range.getValue() === true) { guardarLote_(sh); }
    else if (a1 === CAP.LIMPIAR && e.range.getValue() === true) { limpiarCaptura_(sh, 'Formulario en blanco. Elige el MPI en C5.'); }
  } catch (err) {
    sh.getRange(CAP.MSG).setValue('ERROR: ' + err.message + ' — avisa a Administración.');
    bitacora_('ERROR', a1 + ': ' + err.message, '');
    if (a1 === CAP.GUARDAR) { sh.getRange(CAP.GUARDAR).setValue(false); }
  }
}

function cargarFicha_(sh) {
  var ss = sh.getParent();
  var mpi = String(sh.getRange(CAP.MPI).getValue() || '');
  var filas = [], info = 'Sin ficha: escribe los insumos a mano.';
  var cat = catalogo_(ss);
  var m = cat.porNombre[norm_(mpi)];
  if (m && m.ficha && m.ficha !== 'SIN FICHA') {
    var fd = ss.getSheetByName(H.FD).getDataRange().getValues();
    for (var i = 3; i < fd.length; i++) {
      if (String(fd[i][0]) !== m.ficha) { continue; }
      var ref = fd[i][15];
      var refTxt = (ref !== '' && ref !== null && !isNaN(Number(ref))) ? Number(ref) : (fd[i][8] !== '' ? (fd[i][8] + ' ' + fd[i][9]) : 'sin cantidad');
      filas.push([fd[i][6], refTxt]);
    }
    var fi = ss.getSheetByName(H.FI).getDataRange().getValues();
    for (i = 3; i < fi.length; i++) {
      if (String(fi[i][0]) === m.ficha) { info = 'Ficha ' + m.ficha + (fi[i][5] ? ' · neto esperado ' + fi[i][5] + ' ' + fi[i][6] : '') + ' · ' + fi[i][9]; }
    }
  }
  var bloque = sh.getRange(CAP.FILA1, 2, CAP.NFILAS, 3);
  bloque.clearContent();
  if (filas.length) {
    var v = [];
    for (i = 0; i < filas.length && i < CAP.NFILAS; i++) { v.push([filas[i][0], filas[i][1], '']); }
    sh.getRange(CAP.FILA1, 2, v.length, 3).setValues(v);
  }
  sh.getRange(CAP.FILA1, 6, CAP.NFILAS, 1).clearContent();
  sh.getRange(CAP.INFO).setValue(mpi ? info : '');
  sh.getRange(CAP.MSG).setValue(mpi ? (filas.length ? 'Ficha cargada: ' + filas.length + ' ingrediente(s). Pesa y escribe la CANTIDAD USADA de cada uno.' : info) : '');
}

function limpiarCaptura_(sh, msg) {
  sh.getRange(CAP.MPI).clearContent(); sh.getRange(CAP.INFO).clearContent();
  sh.getRange(CAP.TIEMPO).clearContent(); sh.getRange(CAP.UNID).clearContent();
  sh.getRange(CAP.FILA1, 2, CAP.NFILAS, 3).clearContent(); sh.getRange(CAP.FILA1, 6, CAP.NFILAS, 1).clearContent();
  sh.getRange(CAP.NETO).clearContent(); sh.getRange(CAP.OBS).clearContent(); sh.getRange(CAP.MOTIVO).clearContent();
  sh.getRange(CAP.FECHA).setValue(hoy_());
  sh.getRange(CAP.GUARDAR).setValue(false); sh.getRange(CAP.LIMPIAR).setValue(false);
  sh.getRange(CAP.MSG).setValue(msg || '');
}

function guardarLote_(sh) {
  var ss = sh.getParent();
  var lock = LockService.getDocumentLock();
  if (!lock.tryLock(20000)) { sh.getRange(CAP.MSG).setValue('Otro guardado está en curso. Vuelve a marcar en un minuto.'); sh.getRange(CAP.GUARDAR).setValue(false); return; }
  try {
    var f = leerCaptura_(sh);
    var cat = catalogo_(ss);
    var fatales = [], bloqueos = [], avisos = [];
    // F — fatales
    if (!esFecha_(f.fecha)) { fatales.push('F1 La fecha (C4) no es una fecha.'); }
    else if (aMedianoche_(f.fecha).getTime() > aMedianoche_(new Date()).getTime()) { fatales.push('F1 La fecha es futura.'); }
    if (!f.resp) { fatales.push('F2 Falta el responsable (F4).'); }
    var mpi = cat.porNombre[norm_(f.mpi)];
    if (!mpi || mpi.tipo !== 'MPI') { fatales.push('F3 El MPI de C5 no está en el catálogo.'); }
    var lineas = [];
    for (var i = 0; i < f.filas.length; i++) {
      var x = f.filas[i];
      if (!x.insumo && (x.cant === '' || x.cant === null)) { continue; }
      var ins = cat.porNombre[norm_(x.insumo)];
      if (!ins) { fatales.push('F5 Fila ' + x.fila + ': «' + x.insumo + '» no está en el catálogo.'); continue; }
      if (x.cant === '' || x.cant === null) { avisos.push('Fila ' + x.fila + ': ' + x.insumo + ' sin cantidad (no se registra).'); continue; }
      var q = Number(x.cant);
      if (isNaN(q) || q <= 0) { fatales.push('F6 Fila ' + x.fila + ': cantidad no válida.'); continue; }
      if (q > Number(cfg_('MAX_CANTIDAD_LINEA'))) { bloqueos.push('B1 Fila ' + x.fila + ': ' + q + ' ' + ins.ub + ' de ' + ins.nombre + ' — ¿sobra un dígito?'); }
      lineas.push({ ins: ins, cant: q, unidad: ins.ub, nombre: x.insumo, nota: x.nota });
    }
    if (!lineas.length) { fatales.push('F4 No hay ningún ingrediente con cantidad.'); }
    var neto = Number(f.neto);
    if (!f.neto || isNaN(neto) || neto <= 0) { fatales.push('F7 Falta el PESO NETO FINAL (C35).'); }
    var entrada = 0;
    for (i = 0; i < lineas.length; i++) { entrada += lineas[i].cant; }
    if (neto > 0 && entrada > 0 && neto > entrada * 1.10) { bloqueos.push('B2 El neto (' + neto + ' g) supera la entrada (' + Math.round(entrada) + ') en más de 10 %.'); }
    if (fatales.length) { return rechazar_(sh, 'NO SE GUARDÓ — corrige:\n' + lista8_(fatales), fatales); }
    var motivo = String(f.motivo || '');
    if (bloqueos.length && motivo.length < Number(cfg_('MIN_MOTIVO_OVERRIDE'))) {
      return rechazar_(sh, 'BLOQUEADO — escribe el motivo en F38 (mínimo ' + cfg_('MIN_MOTIVO_OVERRIDE') + ' caracteres) y vuelve a marcar GUARDAR:\n' + lista8_(bloqueos), bloqueos);
    }
    // costeo
    var ctx = contextoCosto_(ss);
    var idLote = siguienteIdLote_(ss, f.fecha);
    var lote = { id: idLote, fecha: aMedianoche_(f.fecha), mpi: mpi, neto: neto, entrada: entrada };
    var res = costearLineas_(ctx, lote, lineas);
    var rendFicha = rendimientoFicha_(ss, mpi.ficha);
    var rend = entrada > 0 ? neto / entrada : '';
    var desvio = (rendFicha && rend !== '') ? (rend - rendFicha) / rendFicha : '';
    var hall = [];
    if (desvio !== '' && Math.abs(desvio) * 100 > Number(cfg_('DESVIO_REND_AVISO_PCT'))) { hall.push('REND_FUERA'); avisos.push('Rendimiento ' + pct_(rend) + ' vs ficha ' + pct_(rendFicha) + ' (' + pct_(desvio) + ').'); }
    if (res.sinPrecio) { hall.push('COSTO_PARCIAL'); avisos.push(res.sinPrecio + ' insumo(s) sin precio: costo parcial.'); }
    if (bloqueos.length) { hall.push('OVERRIDE'); }
    var costoG = (!res.sinPrecio && neto > 0) ? res.total / neto : '';
    var usuario = usuario_();
    var ahora = new Date();
    var filaLote = [idLote, lote.fecha, '', mpi.cod, mpi.nombre, f.mpi, f.resp, '', '', f.tiempo === '' ? '' : Number(f.tiempo),
      entrada, neto, f.unid === '' ? '' : Number(f.unid), '', '', '', rend, rendFicha || '', desvio, res.total, costoG,
      f.obs, usuario, ahora, 'ACTIVA', '', 'CAPTURA', hall.join(';'), res.sinPrecio,
      res.sinPrecio ? (res.total > 0 ? 'PARCIAL' : 'SIN COSTO') : 'COMPLETO', bloqueos.length ? motivo : ''];
    var filasCons = [];
    for (i = 0; i < res.lineas.length; i++) {
      var L = res.lineas[i];
      filasCons.push([idLote + '-' + pad3_(i + 1), idLote, lote.fecha, L.ins.cod, L.ins.nombre, L.ins.tipo, L.nombre, L.cant, L.unidad, '', L.loteOrigen || '',
        L.cu === null ? '' : L.cu, L.costo === null ? '' : L.costo, L.fuente, 'ACTIVA', L.hall || '']);
    }
    agregarFilas_(ss.getSheetByName(H.LOTES), [filaLote]);
    agregarFilas_(ss.getSheetByName(H.CONS), filasCons);
    bitacora_('LOTE GUARDADO', idLote + ' · ' + mpi.nombre + ' · ' + neto + ' g · ' + (costoG !== '' ? '$' + fmt_(costoG, 2) + '/g' : 'costo ' + filaLote[29]) + (bloqueos.length ? ' · FORZADO' : ''), bloqueos.length ? motivo : '');
    var msg = '✔ GUARDADO ' + idLote + ' · ' + mpi.nombre + ' · ' + fmt_(neto, 0) + ' g · costo $' + fmt_(res.total, 0) +
      (costoG !== '' ? ' = $' + fmt_(costoG, 2) + ' por gramo' : ' (parcial)') + (rend !== '' ? ' · rendimiento ' + pct_(rend) : '');
    if (avisos.length) { msg += '\nAvisos: ' + lista8_(avisos).replace(/\n/g, ' · '); }
    limpiarCaptura_(sh, msg);
    try { recalcularDerivados_(ss); } catch (e2) { bitacora_('AVISO', 'Derivados no recalculados: ' + e2.message, ''); }
  } finally {
    try { lock.releaseLock(); } catch (e3) {}
  }
}

function siguienteIdLote_(ss, fecha) {

}

/** Lee catálogo, parámetros, factores, compras de recepción y lotes costeados. */
function contextoCosto_(ss) {
  var cat = catalogo_(ss);
  var par = {};
  var p = ss.getSheetByName(H.PAR).getDataRange().getValues();
  for (var i = 3; i < p.length; i++) {
    if (p[i][0] !== '' && p[i][3] !== '' && p[i][3] !== null && !isNaN(Number(p[i][3]))) { par[String(p[i][0])] = Number(p[i][3]); }
  }
  var fac = {};
  var fs = ss.getSheetByName(H.FAC);
  if (fs) {
    var f = fs.getDataRange().getValues();
    for (i = 1; i < f.length; i++) {
      if (f[i][0] && f[i][1] && f[i][2] !== '' && !isNaN(Number(f[i][2]))) { fac[norm_(f[i][0]) + '|' + String(f[i][1]).toUpperCase()] = Number(f[i][2]); }
    }
  }
  var ctx = { cat: cat, par: par, fac: fac, compras: {}, sinFactor: {}, lotes: {}, porMpi: {} };
  leerCompras_(ctx);
  var lv = ss.getSheetByName(H.LOTES).getDataRange().getValues();
  var c = idx_(lv[0]);
  for (i = 1; i < lv.length; i++) {
    var id = String(lv[i][c.ID_LOTE] || ''); if (!id) { continue; }
    var o = { id: id, fecha: lv[i][c.FECHA], mpi: String(lv[i][c.MPI_COD]), costoG: lv[i][c.COSTO_G], estado: String(lv[i][c.ESTADO_LOTE]), estadoCosto: String(lv[i][c.ESTADO_COSTO]) };
    ctx.lotes[id] = o;
    if (o.estado !== 'REVERSADA' && o.costoG !== '' && o.costoG !== null) { (ctx.porMpi[o.mpi] = ctx.porMpi[o.mpi] || []).push(o); }
  }
  return ctx;
}

function leerCompras_(ctx) {
  ctx.fuentes = [];
  leerFuente_(ctx, 'BAR', String(cfg_('ID_RECEPCION_BAR')), String(cfg_('HOJA_RECEPCION')), true);
  var idc = String(cfg_('ID_RECEPCION_COCINA') || '');
  if (idc) { leerFuente_(ctx, 'COCINA', idc, String(cfg_('HOJA_RECEPCION_COCINA') || 'BD_RECEPCIONES'), false); }
  ctx.descartadas = {};
  for (var k in ctx.compras) {
    if (!Object.prototype.hasOwnProperty.call(ctx.compras, k)) { continue; }
    var cs = ctx.compras[k];
    // precios contradictorios: un error de unidad no puede entrar al promedio
    var pus = [], i;
    for (i = 0; i < cs.length; i++) { pus.push(cs[i].valor / cs[i].base); }
    var orden = pus.slice(0).sort(function (a, b) { return a - b; });
    var med = orden.length % 2 ? orden[(orden.length - 1) / 2] : (orden[orden.length / 2 - 1] + orden[orden.length / 2]) / 2;
    var quedan = [];
    for (i = 0; i < cs.length; i++) {
      var malo = (cs.length >= 3 && (pus[i] > med * 5 || pus[i] < med / 5)) || (cs.length === 2 && (orden[1] / orden[0]) > 5);
      if (malo) { ctx.descartadas[k] = (ctx.descartadas[k] || 0) + 1; } else { quedan.push(cs[i]); }
    }
    quedan.sort(function (a, b) { return a.fecha - b.fecha; });
    if (quedan.length) { ctx.compras[k] = quedan; } else { delete ctx.compras[k]; }
  }
}

/** Lee un libro mayor de recepción (bar o cocina) ubicando las columnas por su encabezado. */
function leerFuente_(ctx, etiqueta, id, hoja, obligatoria) {
  var ext, hs;
  try { ext = SpreadsheetApp.openById(id); hs = ext.getSheetByName(hoja); if (!hs) { throw new Error('no existe la hoja ' + hoja); } }
  catch (e) {
    if (obligatoria) { throw new Error('No puedo leer la recepción de ' + etiqueta + ' (' + e.message + '). La cuenta que corre el motor necesita acceso a ese archivo.'); }
    ctx.fuentes.push(etiqueta + ': NO LEÍDA (' + e.message + ')'); return;
  }
  var v = hs.getDataRange().getValues(), fe = -1;
  for (var r = 0; r < Math.min(12, v.length); r++) { if (norm_(v[r][0]).indexOf('id_linea') === 0) { fe = r; break; } }
  if (fe < 0) { throw new Error('Recepción de ' + etiqueta + ': no encuentro la fila de encabezados (ID_LINEA).'); }
  var enc = [];
  for (var j = 0; j < v[fe].length; j++) { enc.push(norm_(String(v[fe][j]).split('\n')[0])); }
  var col = function (cands) { for (var a = 0; a < cands.length; a++) { var p = enc.indexOf(norm_(cands[a])); if (p >= 0) { return p; } } return -1; };
  var c = { fecha: col(['FECHA_RECEPCION', 'Fecha']), prod: col(['PRODUCTO', 'Producto']), un: col(['UNIDAD', 'Un.']),
    cant: col(['CANT_RECIBIDA', 'Cant. real']), valor: col(['VALOR_TOTAL', 'Valor línea']), est: col(['ESTADO_LINEA', 'Estado de']) };
  for (var x in c) { if (Object.prototype.hasOwnProperty.call(c, x) && c[x] < 0) { throw new Error('Recepción de ' + etiqueta + ': falta la columna ' + x); } }
  var alias = ctx.cat.porRecepcion, n = 0;
  for (var i = fe + 1; i < v.length; i++) {
    var row = v[i];
    if (!row[0] || String(row[c.est]) === 'REVERSADA') { continue; }
    var ins = alias[norm_(row[c.prod])];
    if (!ins) { continue; }
    var q = Number(row[c.cant]), val = Number(row[c.valor]);
    if (!(q > 0) || !(val > 0) || !esFecha_(row[c.fecha])) { continue; }
    var u = String(row[c.un] || '').toUpperCase();
    if (u === 'G') { u = 'GR'; } if (u === 'UN') { u = 'UND'; }
    var factor = factorCompra_(ctx, ins, u);
    if (factor === null) { if (String(ctx.sinFactor[ins.cod] || '').indexOf(u) < 0) { ctx.sinFactor[ins.cod] = (ctx.sinFactor[ins.cod] ? ctx.sinFactor[ins.cod] + ', ' : '') + u; } continue; }
    (ctx.compras[ins.cod] = ctx.compras[ins.cod] || []).push({ fecha: aMedianoche_(row[c.fecha]), base: q * factor, valor: val, id: etiqueta + ':' + String(row[0]) });
    n++;
  }
  ctx.fuentes.push(etiqueta + ': ' + n + ' compra(s) utilizables');
}

function factorCompra_(ctx, ins, u) {
  var ub = ins.ub;
  if (ub === 'g' && (u === 'GR' || u === 'G')) { return 1; }
  if (ub === 'g' && u === 'KG') { return 1000; }
  if (ub === 'ml' && u === 'ML') { return 1; }
  if (ub === 'ml' && (u === 'L' || u === 'LT')) { return 1000; }
  if (ub === 'und' && (u === 'UND' || u === 'UN')) { return 1; }
  var k = norm_(ins.nombre) + '|' + u;
  if (Object.prototype.hasOwnProperty.call(ctx.fac, k)) { return ctx.fac[k]; }
  if (ub === 'g' && u === 'UND') {
    var p = ins.nombre + '|und→g';
    if (Object.prototype.hasOwnProperty.call(ctx.par, p)) { return ctx.par[p]; }
  }
  return null;
}

/** Precio por unidad base de un insumo comprado, vigente a una fecha. */
function precioInsumo_(ctx, cod, fecha) {
  var cs = ctx.compras[cod];
  if (!cs || !cs.length) { return null; }
  var t = aMedianoche_(fecha).getTime();
  var ventana = Number(cfg_('VENTANA_COSTO_DIAS')) * 86400000;
  var v = 0, q = 0, n = 0, previas = 0, i;
  for (i = 0; i < cs.length; i++) {
    var ct = cs[i].fecha.getTime();
    if (ct <= t) { previas++; if (ct >= t - ventana) { v += cs[i].valor; q += cs[i].base; n++; } }
  }
  if (n) { return { pu: v / q, fuente: 'PROMEDIO ' + n + ' COMPRA(S) ≤' + cfg_('VENTANA_COSTO_DIAS') + 'd' }; }
  if (previas) {
    for (i = cs.length - 1; i >= 0; i--) { if (cs[i].fecha.getTime() <= t) { return { pu: cs[i].valor / cs[i].base, fuente: 'ÚLTIMA COMPRA (' + fmtFecha_(cs[i].fecha) + ', fuera de ventana)' }; } }
  }
  return { pu: cs[0].valor / cs[0].base, fuente: 'PRIMERA COMPRA POSTERIOR (' + fmtFecha_(cs[0].fecha) + ')' };
}

/** Costo por gramo de un MPI para un lote: el lote de origen indicado o el último costeado ≤ fecha. */
function precioMpi_(ctx, cod, fecha, loteOrigen, idPropio) {
  if (loteOrigen && ctx.lotes[loteOrigen] && ctx.lotes[loteOrigen].costoG !== '' && ctx.lotes[loteOrigen].costoG !== null && ctx.lotes[loteOrigen].estado !== 'REVERSADA') {
    return { pu: Number(ctx.lotes[loteOrigen].costoG), fuente: 'LOTE ' + loteOrigen, lote: loteOrigen };
  }
  var ls = ctx.porMpi[cod] || [];
  var t = aMedianoche_(fecha).getTime(), mejor = null;
  for (var i = 0; i < ls.length; i++) {
    if (ls[i].id === idPropio || !esFecha_(ls[i].fecha)) { continue; }
    var lt = aMedianoche_(ls[i].fecha).getTime();
    if (lt <= t && (!mejor || lt >= aMedianoche_(mejor.fecha).getTime())) { mejor = ls[i]; }
  }
  if (mejor) { return { pu: Number(mejor.costoG), fuente: 'ÚLTIMO LOTE ' + mejor.id, lote: mejor.id }; }
  return null;
}

function aBase_(ctx, ins, cant, unidad) {
  var u = String(unidad || ins.ub).toLowerCase();
  if (u === ins.ub) { return cant; }
  if ((u === 'g' && ins.ub === 'ml') || (u === 'ml' && ins.ub === 'g')) {
    if (ins.tipo === 'SIN_COSTO') { return cant; }
    var k = ins.nombre + '|' + u + '→' + ins.ub;
    if (Object.prototype.hasOwnProperty.call(ctx.par, k)) { return cant * ctx.par[k]; }
    return null;
  }
  var k2 = ins.nombre + '|' + u + '→' + ins.ub;
  if (Object.prototype.hasOwnProperty.call(ctx.par, k2)) { return cant * ctx.par[k2]; }
  return null;
}

/** lineas: [{ins, cant, unidad, nombre, loteOrigen}] → costo por línea y total. */
function costearLineas_(ctx, lote, lineas) {
  var out = [], total = 0, sinPrecio = 0;
  for (var i = 0; i < lineas.length; i++) {
    var L = lineas[i], ins = L.ins, r = { ins: ins, cant: L.cant, unidad: L.unidad || ins.ub, nombre: L.nombre, loteOrigen: L.loteOrigen || '', cu: null, costo: null, fuente: '', hall: L.hall || '' };
    if (ins.tipo === 'SIN_COSTO') { r.cu = 0; r.costo = 0; r.fuente = 'SIN COSTO'; out.push(r); continue; }
    var base = aBase_(ctx, ins, L.cant, r.unidad);
    if (base === null) { r.fuente = 'FALTA PARÁMETRO ' + ins.nombre + '|' + String(r.unidad).toLowerCase() + '→' + ins.ub; sinPrecio++; out.push(r); continue; }
    var p = ins.tipo === 'MPI' ? precioMpi_(ctx, ins.cod, lote.fecha, L.loteOrigen, lote.id) : precioInsumo_(ctx, ins.cod, lote.fecha);
    if (!p) {
      r.fuente = ins.tipo === 'MPI' ? 'MPI SIN LOTE COSTEADO' : (ctx.sinFactor[ins.cod] ? 'COMPRA EN ' + ctx.sinFactor[ins.cod] + ' SIN FACTOR' : 'SIN COMPRAS REGISTRADAS');
      sinPrecio++; out.push(r); continue;
    }
    r.cu = p.pu; r.costo = base * p.pu; r.fuente = p.fuente; if (p.lote && !r.loteOrigen) { r.loteOrigen = p.lote; }
    total += r.costo; out.push(r);
  }
  return { lineas: out, total: total, sinPrecio: sinPrecio };
}

/** Rendimiento esperado de una ficha = neto / suma de entradas en base (solo si está completa). */
function rendimientoFicha_(ss, ficha) {
  if (!ficha || ficha === 'SIN FICHA') { return null; }
  var fi = ss.getSheetByName(H.FI).getDataRange().getValues(), neto = null;
  for (var i = 3; i < fi.length; i++) { if (String(fi[i][0]) === ficha) { neto = Number(fi[i][5]) || null; } }
  if (!neto) { return null; }
  var fd = ss.getSheetByName(H.FD).getDataRange().getValues(), suma = 0;
  for (i = 3; i < fd.length; i++) {
    if (String(fd[i][0]) !== ficha) { continue; }
    var b = fd[i][15];
    if (b === '' || b === null || isNaN(Number(b))) { return null; }
    suma += Number(b);
  }
  return suma > 0 ? neto / suma : null;
}

/**
 * Costea lotes con ESTADO_COSTO distinto de COMPLETO (histórico migrado o lotes con faltantes).
 * Nunca toca un lote COMPLETO: su costo quedó fijado al guardarse.
 */
function completarCostosPendientes() { alerta_(completarCostosPendientes_(false)); }
function completarCostosPendientes_(silencioso) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var lock = LockService.getDocumentLock();
  if (!lock.tryLock(30000)) { throw new Error('Otro proceso está escribiendo.'); }
  try {
    var ctx = contextoCosto_(ss);
    var hl = ss.getSheetByName(H.LOTES), hc = ss.getSheetByName(H.CONS);
    var lv = hl.getDataRange().getValues(), cv = hc.getDataRange().getValues();
    var cl = idx_(lv[0]), cc = idx_(cv[0]);
    var porLote = {};
    for (var i = 1; i < cv.length; i++) { var id = String(cv[i][cc.ID_LOTE] || ''); if (id) { (porLote[id] = porLote[id] || []).push(i); } }
    var orden = [];
    for (i = 1; i < lv.length; i++) {
      if (!lv[i][cl.ID_LOTE] || String(lv[i][cl.ESTADO_LOTE]) === 'REVERSADA' || String(lv[i][cl.ESTADO_COSTO]) === 'COMPLETO') { continue; }
      orden.push(i);
    }
    orden.sort(function (a, b) {
      var fa = esFecha_(lv[a][cl.FECHA]) ? lv[a][cl.FECHA].getTime() : 0, fb = esFecha_(lv[b][cl.FECHA]) ? lv[b][cl.FECHA].getTime() : 0;
      return fa !== fb ? fa - fb : (String(lv[a][cl.ID_LOTE]) < String(lv[b][cl.ID_LOTE]) ? -1 : 1);
    });
    var hechos = 0, completos = 0, revisados = {};
    for (var pasada = 0; pasada < 4; pasada++) {
    var mejoro = false;
    for (var k = 0; k < orden.length; k++) {
      var r = orden[k], idl = String(lv[r][cl.ID_LOTE]);
      if (String(lv[r][cl.ESTADO_COSTO]) === 'COMPLETO' && revisados[idl]) { continue; }
      var filasC = porLote[idl] || [];
      var lineas = [], idxs = [];
      for (var j = 0; j < filasC.length; j++) {
        var rc = cv[filasC[j]];
        if (String(rc[cc.ESTADO]) === 'REVERSADA') { continue; }
        var ins = ctx.cat.porCodigo[String(rc[cc.INSUMO_COD])];
        var q = Number(rc[cc.CANTIDAD]);
        if (!ins || !(q > 0)) { continue; }
        lineas.push({ ins: ins, cant: q, unidad: String(rc[cc.UNIDAD] || ins.ub), nombre: rc[cc.NOMBRE_ORIGINAL], loteOrigen: String(rc[cc.LOTE_ORIGEN] || ''), hall: rc[cc.HALLAZGOS] });
        idxs.push(filasC[j]);
      }
      var fecha = esFecha_(lv[r][cl.FECHA]) ? lv[r][cl.FECHA] : new Date();
      var res = costearLineas_(ctx, { id: idl, fecha: fecha }, lineas);
      var entrada = 0;
      for (j = 0; j < lineas.length; j++) { entrada += aBase_(ctx, lineas[j].ins, lineas[j].cant, lineas[j].unidad) || lineas[j].cant; }
      for (j = 0; j < res.lineas.length; j++) {
        var L = res.lineas[j], row = cv[idxs[j]];
        row[cc.COSTO_UNIT] = L.cu === null ? '' : L.cu; row[cc.COSTO_LINEA] = L.costo === null ? '' : L.costo;
        row[cc.FUENTE_COSTO] = L.fuente; row[cc.LOTE_ORIGEN] = L.loteOrigen || row[cc.LOTE_ORIGEN];
      }
      var neto = Number(lv[r][cl.NETO_G]);
      var costoG = (!res.sinPrecio && neto > 0 && lineas.length) ? res.total / neto : '';
      var rendF = rendimientoFicha_(ss, ctx.cat.porCodigo[String(lv[r][cl.MPI_COD])] ? ctx.cat.porCodigo[String(lv[r][cl.MPI_COD])].ficha : '');
      var rend = (neto > 0 && entrada > 0) ? neto / entrada : '';
      lv[r][cl.ENTRADA_TOTAL] = entrada || ''; lv[r][cl.RENDIMIENTO] = rend; lv[r][cl.REND_FICHA] = rendF || '';
      lv[r][cl.DESVIO_REND_PCT] = (rendF && rend !== '') ? (rend - rendF) / rendF : '';
      lv[r][cl.COSTO_TOTAL] = res.total; lv[r][cl.COSTO_G] = costoG; lv[r][cl.LINEAS_SIN_PRECIO] = res.sinPrecio;
      lv[r][cl.ESTADO_COSTO] = !lineas.length ? 'SIN INGREDIENTES' : (res.sinPrecio ? (res.total > 0 ? 'PARCIAL' : 'SIN COSTO') : (neto > 0 ? 'COMPLETO' : 'SIN NETO'));
      // el lote recién costeado ya sirve de origen a los siguientes (cascada)
      ctx.lotes[idl] = { id: idl, fecha: fecha, mpi: String(lv[r][cl.MPI_COD]), costoG: costoG, estado: String(lv[r][cl.ESTADO_LOTE]) };
      if (costoG !== '') { (ctx.porMpi[String(lv[r][cl.MPI_COD])] = ctx.porMpi[String(lv[r][cl.MPI_COD])] || []).push(ctx.lotes[idl]); if (pasada > 0) { mejoro = true; } }
      revisados[idl] = 1;
    }
    if (pasada > 0 && !mejoro) { break; }
    }
    for (k = 0; k < orden.length; k++) { hechos++; if (String(lv[orden[k]][cl.ESTADO_COSTO]) === 'COMPLETO') { completos++; } }
    if (lv.length > 1) { hl.getRange(1, 1, lv.length, lv[0].length).setValues(lv); }
    if (cv.length > 1) { hc.getRange(1, 1, cv.length, cv[0].length).setValues(cv); }
    var msg = 'Costeo de pendientes: ' + hechos + ' lote(s) revisado(s), ' + completos + ' con costo completo.';
    bitacora_('COSTEO PENDIENTES', msg, '');
    if (!silencioso) { try { recalcularDerivados_(ss); } catch (e) {} }
    return msg;
  } finally { try { lock.releaseLock(); } catch (e2) {} }
}

function recalcularTodo() { var ss = SpreadsheetApp.getActiveSpreadsheet(); recalcularDerivados_(ss); alerta_('Costos y tablero recalculados.'); }

function recalcularDerivados_(ss) {
  var ctx = contextoCosto_(ss);
  var hoyD = hoy_();
  var aviso = Number(cfg_('DESVIO_PRECIO_AVISO_PCT')) / 100;
  // --- COSTO_INSUMOS
  var filas = [], alertasPrecio = 0, conPrecio = 0, mpe = 0;
  var cods = ctx.cat.lista;

    fm.push([m.cod, m.nombre, m.ficha, g ? g.n : 0, g && g.ult ? g.ult.id : '', g && g.ult ? g.ult.fecha : '',

function costoEstandar_(ss, ctx, ficha, fecha) {
  if (!ficha || ficha === 'SIN FICHA') { return { total: '', porUnidad: '', nota: 'Sin ficha' }; }
  var fd = ss.getSheetByName(H.FD).getDataRange().getValues(), lineas = [], faltan = 0;
  for (var i = 3; i < fd.length; i++) {
    if (String(fd[i][0]) !== ficha) { continue; }
    var ins = ctx.cat.porCodigo[String(fd[i][5])], b = fd[i][15];
    if (!ins || b === '' || b === null || isNaN(Number(b))) { faltan++; continue; }
    lineas.push({ ins: ins, cant: Number(b), unidad: ins.ub, nombre: ins.nombre });
  }
  if (!lineas.length) { return { total: '', porUnidad: '', nota: 'Ficha sin cantidades' }; }
  var res = costearLineas_(ctx, { id: '', fecha: fecha }, lineas);
  var neto = null, fi = ss.getSheetByName(H.FI).getDataRange().getValues(), um = '';
  for (i = 3; i < fi.length; i++) { if (String(fi[i][0]) === ficha) { neto = Number(fi[i][5]) || null; um = fi[i][6]; } }
  var nota = faltan ? faltan + ' línea(s) de la ficha sin cantidad en base (ver PARAMETROS)' : (res.sinPrecio ? res.sinPrecio + ' insumo(s) sin precio' : 'Completo');
  var completo = !faltan && !res.sinPrecio;
  return { total: completo ? res.total : '', porUnidad: (completo && neto) ? res.total / neto : '', nota: completo && neto ? 'Completo · por ' + um : nota };
}

function diagnostico_(ss, ctx) {
  var lv = ss.getSheetByName(H.LOTES).getDataRange().getValues(), cl = idx_(lv[0]);
  var cv = ss.getSheetByName(H.CONS).getDataRange().getValues(), cc = idx_(cv[0]);
  var ids = {}, dup = 0, revisar = 0, sinCosto = 0, sinNeto = 0, forz = 0, rev = 0, lotesIds = {};
  for (var i = 1; i < lv.length; i++) {
    var id = String(lv[i][cl.ID_LOTE] || ''); if (!id) { continue; }
    if (ids[id]) { dup++; } ids[id] = 1; lotesIds[id] = 1;
    var est = String(lv[i][cl.ESTADO_LOTE]);
    if (est === 'REVERSADA') { rev++; continue; }
    if (est === 'REVISAR') { revisar++; }
    if (String(lv[i][cl.ESTADO_COSTO]) !== 'COMPLETO') { sinCosto++; }
    if (!(Number(lv[i][cl.NETO_G]) > 0)) { sinNeto++; }
    if (String(lv[i][cl.HALLAZGOS]).indexOf('OVERRIDE') >= 0) { forz++; }
  }
  var huerf = 0, idsL = {}, dupL = 0, faltaPar = {}, sinCompra = {};
  for (i = 1; i < cv.length; i++) {
    var il = String(cv[i][cc.ID_LINEA] || ''); if (!il) { continue; }
    if (idsL[il]) { dupL++; } idsL[il] = 1;
    if (!lotesIds[String(cv[i][cc.ID_LOTE])]) { huerf++; }
    var fu = String(cv[i][cc.FUENTE_COSTO]);
    if (fu.indexOf('FALTA PARÁMETRO') === 0) { faltaPar[fu.replace('FALTA PARÁMETRO ', '')] = 1; }
    if (fu === 'SIN COMPRAS REGISTRADAS' || fu.indexOf('SIN FACTOR') > 0) { sinCompra[String(cv[i][cc.INSUMO])] = 1; }
  }
  var nPar = 0, nSC = 0, k;
  for (k in faltaPar) { if (Object.prototype.hasOwnProperty.call(faltaPar, k)) { nPar++; } }
  for (k in sinCompra) { if (Object.prototype.hasOwnProperty.call(sinCompra, k)) { nSC++; } }
  var d = [
    ['INT-01', 'IDs de lote duplicados', dup, 0], ['INT-02', 'IDs de línea duplicados', dupL, 0],
    ['INT-03', 'Líneas de consumo sin lote', huerf, 0], ['INT-04', 'Lotes sin costo completo', sinCosto, 0],
    ['INT-05', 'Parámetros físicos que faltan (en lotes)', nPar, 0], ['INT-06', 'Insumos usados sin compras con precio', nSC, 0],
    ['INT-07', 'Lotes sin peso neto', sinNeto, 0], ['INT-08', 'Lotes del histórico en REVISAR', revisar, 0],
    ['INT-09', 'Lotes guardados con motivo (forzados)', forz, '—'], ['INT-10', 'Lotes reversados', rev, '—']
  ];
  var filas = [];
  for (i = 0; i < d.length; i++) {
    var estado = d[i][3] === '—' ? 'RECONOCIDO' : (Number(d[i][2]) > 0 ? 'ABIERTO' : 'OK');
    filas.push([d[i][0], d[i][1], d[i][2], d[i][3], estado]);
  }
  var det = [];
  for (k in faltaPar) { if (Object.prototype.hasOwnProperty.call(faltaPar, k)) { det.push(['INT-05', k, '', '', 'Llenar en PARAMETROS']); } }
  for (k in sinCompra) { if (Object.prototype.hasOwnProperty.call(sinCompra, k)) { det.push(['INT-06', k, '', '', 'Registrar su compra en la recepción o un factor en FACTORES_COMPRA']); } }
  escribirDerivado_(ss, H.DIA, 'DIAGNÓSTICO DE INTEGRIDAD', 'Un control ABIERTO es trabajo pendiente. Debajo, el detalle de qué falta.',
    ['CÓDIGO', 'CONTROL', 'VALOR', 'META', 'ESTADO'], filas.concat([['', '', '', '', '']]).concat(det), {}, [9, 50, 10, 8, 50]);
  return filas;
}

function reversarLoteMenu() {
  var ui = SpreadsheetApp.getUi();
  var r1 = ui.prompt('Reversar lote', 'ID del lote (ej. PRO-2026-066):', ui.ButtonSet.OK_CANCEL);
  if (r1.getSelectedButton() !== ui.Button.OK) { return; }
  var r2 = ui.prompt('Reversar ' + r1.getResponseText(), 'Motivo (mínimo ' + cfg_('MIN_MOTIVO_OVERRIDE') + ' caracteres):', ui.ButtonSet.OK_CANCEL);
  if (r2.getSelectedButton() !== ui.Button.OK) { return; }
  ui.alert(reversarLote_(String(r1.getResponseText()).replace(/^\s+|\s+$/g, ''), r2.getResponseText()));
}
function reversarLote_(id, motivo) {
  if (String(motivo || '').length < Number(cfg_('MIN_MOTIVO_OVERRIDE'))) { return 'Motivo demasiado corto. No se reversó nada.'; }
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var hl = ss.getSheetByName(H.LOTES), hc = ss.getSheetByName(H.CONS);
  var lv = hl.getDataRange().getValues(), cl = idx_(lv[0]), hecho = false;
  for (var i = 1; i < lv.length; i++) {
    if (String(lv[i][cl.ID_LOTE]) === id && String(lv[i][cl.ESTADO_LOTE]) !== 'REVERSADA') {
      hl.getRange(i + 1, cl.ESTADO_LOTE + 1).setValue('REVERSADA');
      hl.getRange(i + 1, cl.REF_REVERSA + 1).setValue(fmtFecha_(new Date()) + ' · ' + usuario_() + ' · ' + motivo);
      hecho = true;
    }
  }
  if (!hecho) { return 'No encontré un lote vigente con ID ' + id + '.'; }
  var cv = hc.getDataRange().getValues(), cc = idx_(cv[0]), n = 0;
  for (i = 1; i < cv.length; i++) { if (String(cv[i][cc.ID_LOTE]) === id) { hc.getRange(i + 1, cc.ESTADO + 1).setValue('REVERSADA'); n++; } }
  bitacora_('LOTE REVERSADO', id + ' · ' + n + ' línea(s)', motivo);
  try { recalcularDerivados_(ss); } catch (e) {}
  return 'Lote ' + id + ' reversado (' + n + ' línea(s)). El registro original queda en el libro mayor.';
}

function costoCocteles_(ss, ctx) {
  var hc = ss.getSheetByName(H.COC), hr = ss.getSheetByName(H.REC);
  if (!hc || !hr) { return null; }
  var cv = hc.getDataRange().getValues(), rv = hr.getDataRange().getValues(), i;
  var lista = [], porId = {};
  for (i = 3; i < cv.length; i++) {
    var id = String(cv[i][0] || '');
    if (!id || String(cv[i][4]).toUpperCase() === 'NO') { continue; }
    var o = { id: id, nombre: String(cv[i][1]), cat: String(cv[i][2]), precio: Number(cv[i][3]) || 0, n: 0, nCost: 0, costo: 0, falta: [], sup: 0 };
    lista.push(o); porId[id] = o;
  }
  var fecha = hoy_(), dSup = Number(cfg_('DENSIDAD_SUPUESTA_ML_G')) || 0, det = [], bloq = {}, cache = {};
  for (i = 3; i < rv.length; i++) {
    var oc = porId[String(rv[i][0] || '')];
    if (!oc) { continue; }
    var cod = String(rv[i][4] || '').replace(/^\s+|\s+$/g, '');
    var ins = ctx.cat.porCodigo[cod] || ctx.cat.porNombre[norm_(rv[i][5])] || null;
    var q = rv[i][6], u = String(rv[i][7] || '').toLowerCase().replace(/^\s+|\s+$/g, '');
    var res;
    oc.n++;
    if (!ins) { res = { base: '', pu: '', costo: '', fuente: 'CÓDIGO FUERA DEL CATÁLOGO', falta: 'Código ' + (cod || '(vacío)') + ' en RECETAS_COCTEL' }; }
    else if (UNID_TRAZA_[u]) { res = { base: '', pu: '', costo: 0, fuente: 'TRAZA (no se costea)' }; }
    else if (q === '' || q === null || isNaN(Number(q)) || Number(q) <= 0) { res = { base: '', pu: '', costo: '', fuente: 'SIN CANTIDAD EN LA FICHA', falta: 'Cantidad de ' + ins.nombre + ' en la receta' }; }
    else { res = costearLineaCoctel_(ss, ctx, ins, Number(q), u, fecha, dSup, cache); }
    if (res.costo === '') {
      if (oc.falta.indexOf(res.falta) < 0) { oc.falta.push(res.falta); }
      var b = bloq[res.falta] = bloq[res.falta] || { n: 0, ids: {} };
      if (!b.ids[oc.id]) { b.ids[oc.id] = 1; b.n++; }
    } else { oc.costo += res.costo; oc.nCost++; if (res.sup) { oc.sup++; } }
    det.push([oc.id, oc.nombre, rv[i][3], cod, ins ? ins.nombre : '', q, u, res.base, res.pu, res.costo, res.fuente]);
  }
  var imp = Number(cfg_('IMPOCONSUMO_PCT')) || 0, meta = Number(cfg_('META_COSTO_BEBIDA_PCT')) || 0;
  var filas = [], completos = 0, conPrecio = 0, sumPct = 0, sobreMeta = 0;
  for (i = 0; i < lista.length; i++) {
    var x = lista[i];
    var completo = x.n > 0 && x.nCost === x.n;
    var estado = !x.n ? 'SIN RECETA' : (completo ? (x.sup ? 'COMPLETO (densidad supuesta)' : 'COMPLETO') : (x.costo > 0 ? 'PARCIAL' : 'SIN COSTO'));
    var neto = x.precio > 0 ? x.precio / (1 + imp / 100) : '';
    var pct = (completo && neto) ? x.costo / neto : '';
    var vsMeta = '';
    if (pct !== '' && meta > 0) { vsMeta = pct * 100 > meta ? '⚠ ' + pct_(pct - meta / 100) + ' sobre la meta' : '✔ dentro de la meta'; if (pct * 100 > meta) { sobreMeta++; } }
    if (completo) { completos++; }
    if (pct !== '') { conPrecio++; sumPct += pct; }
    filas.push([x.id, x.nombre, x.cat, estado, x.n ? x.costo : '', x.nCost + ' de ' + x.n, x.precio || '', neto, pct, neto !== '' && completo ? neto - x.costo : '', vsMeta,
      x.falta.join(' · ') || (completo ? '' : '—')]);
  }
  escribirDerivado_(ss, H.CC, 'COSTO REAL DE CADA CÓCTEL', 'Costo de la receta con el precio vigente de hoy (insumos) y el último lote costeado (MPI). Sin decoración. % de costo sobre el precio sin impoconsumo (' + imp + ' %). Lo escribe el motor.',
    ['ID', 'CÓCTEL', 'CATEGORÍA', 'ESTADO', 'COSTO RECETA $', 'LÍNEAS CON COSTO', 'PRECIO CARTA $', 'PRECIO SIN IMPOCONSUMO $', '% COSTO', 'MARGEN BRUTO $', 'VS META', 'QUÉ FALTA PARA COMPLETARLO'],
    filas, { 5: '$#,##0', 7: '$#,##0', 8: '$#,##0', 9: '0.0%', 10: '$#,##0' }, [6, 28, 16, 16, 13, 11, 13, 14, 9, 13, 18, 80]);
  escribirDerivado_(ss, H.CCD, 'DETALLE DEL COSTO POR INGREDIENTE', 'Una fila por línea de receta: cantidad, conversión a la unidad base, precio por unidad base y de dónde sale.',
    ['ID', 'CÓCTEL', 'INGREDIENTE (ficha)', 'CÓDIGO', 'INSUMO', 'CANTIDAD', 'UNIDAD', 'CANT. EN UNIDAD BASE', '$ / UNIDAD BASE', 'COSTO $', 'FUENTE DEL COSTO'],
    det, { 8: '#,##0.##', 9: '#,##0.00', 10: '$#,##0' }, [6, 26, 30, 14, 28, 9, 7, 12, 12, 11, 60]);
  var top = [];
  for (var kk in bloq) { if (Object.prototype.hasOwnProperty.call(bloq, kk)) { top.push([kk, bloq[kk].n]); } }
  top.sort(function (a, c2) { return c2[1] - a[1]; });
  return { total: lista.length, completos: completos, conPrecio: conPrecio, pctProm: conPrecio ? sumPct / conPrecio : '', sobreMeta: sobreMeta, meta: meta, top: top };
}

/** Costea una línea de receta. Devuelve {base, pu, costo, fuente, falta?, sup?}; costo '' = no se pudo. */
function costearLineaCoctel_(ss, ctx, ins, q, u, fecha, dSup, cache) {
  if (ins.tipo === 'SIN_COSTO') { return { base: q, pu: 0, costo: 0, fuente: 'SIN COSTO (subproducto ya pagado)' }; }
  var base, sup = false;
  if (ins.ub === 'und' && (u === 'bot' || u === 'lata' || u === 'und')) { base = q; }
  else { base = aBase_(ctx, ins, q, u || ins.ub); }
  if (base === null && dSup > 0 && ((u === 'ml' && ins.ub === 'g') || (u === 'g' && ins.ub === 'ml'))) { base = u === 'ml' ? q * dSup : q / dSup; sup = true; }
  if (base === null) {
    return { base: '', pu: '', costo: '', fuente: 'FALTA PARÁMETRO ' + ins.nombre + '|' + u + '→' + ins.ub, falta: 'Medir ' + ins.nombre + ' (' + u + '→' + ins.ub + ') en PESAJE' };
  }
  var p;
  if (ins.tipo === 'MPI') {
    if (!Object.prototype.hasOwnProperty.call(cache, ins.cod)) { cache[ins.cod] = precioMpiCoctel_(ss, ctx, ins, fecha); }
    p = cache[ins.cod];
  } else { p = precioInsumo_(ctx, ins.cod, fecha); }
  if (!p) {
    if (ins.tipo === 'MPI') { return { base: base, pu: '', costo: '', fuente: 'MPI SIN COSTO (sin lote costeado ni ficha completa)', falta: 'Costo del MPI ' + ins.nombre + ' (registrar un lote o completar su ficha)' }; }
    if (ctx.sinFactor[ins.cod]) { return { base: base, pu: '', costo: '', fuente: 'COMPRA EN ' + ctx.sinFactor[ins.cod] + ' SIN FACTOR', falta: 'Cuánto trae 1 ' + ctx.sinFactor[ins.cod] + ' de ' + ins.nombre + ' (PESAJE)' }; }
    return { base: base, pu: '', costo: '', fuente: 'SIN COMPRAS REGISTRADAS', falta: 'Registrar en la recepción una compra de ' + ins.nombre };
  }
  return { base: base, pu: p.pu, costo: base * p.pu, fuente: p.fuente + (sup ? ' · DENSIDAD SUPUESTA' : ''), sup: sup };
}

/** Costo por unidad base de un MPI para cócteles: último lote costeado; si no hay, la ficha estándar (si su unidad coincide). */
function precioMpiCoctel_(ss, ctx, ins, fecha) {
  var p = precioMpi_(ctx, ins.cod, fecha, '', '');
  if (p) { return p; }
  if (!ins.ficha || ins.ficha === 'SIN FICHA') { return null; }
  var fi = ss.getSheetByName(H.FI).getDataRange().getValues(), um = '';
  for (var i = 3; i < fi.length; i++) { if (String(fi[i][0]) === ins.ficha) { um = String(fi[i][6] || '').toLowerCase(); } }
  if (um !== ins.ub) { return null; }
  var est = costoEstandar_(ss, ctx, ins.ficha, fecha);
  if (est.porUnidad === '' || est.porUnidad === null) { return null; }
  return { pu: Number(est.porUnidad), fuente: 'ESTÁNDAR FICHA ' + ins.ficha + ' (sin lote costeado)' };
}

function construirPesaje_(ss, msg) {
  var sh = ss.getSheetByName(H.PES);
  if (!sh) { sh = ss.insertSheet(H.PES, 1); }
  var it = [], i;
  // 1) factores de compra (cada uno destraba todas las compras de ese insumo)
  var cat = catalogo_(ss);
  var fv = ss.getSheetByName(H.FAC).getDataRange().getValues();
  for (i = 1; i < fv.length; i++) {
    if (!fv[i][0] || !fv[i][1] || (fv[i][2] !== '' && fv[i][2] !== null)) { continue; }
    var ins = cat.porNombre[norm_(fv[i][0])];
    it.push({ g: 1, peso: 100, que: '¿Cuánto trae 1 ' + fv[i][1] + ' de ' + fv[i][0] + '?', como: String(fv[i][3] || 'Leer la etiqueta o contar'),
      unidad: ins ? ins.ub + ' por ' + fv[i][1] : 'unidad base', dest: H.FAC, clave: fv[i][0] + '|' + fv[i][1] });
  }
  // 2) parámetros físicos (peso = líneas de ficha de producción + líneas de cóctel que destraba)
  var enCoctel = clavesCoctel_(ss, cat);
  var pv = ss.getSheetByName(H.PAR).getDataRange().getValues();
  for (i = 3; i < pv.length; i++) {
    if (!pv[i][0] || (pv[i][3] !== '' && pv[i][3] !== null)) { continue; }
    var conv = String(pv[i][2]), de = conv.split('→')[0], a = conv.split('→')[1] || '';
    var que = conv === 'ml→g' ? '¿Cuántos gramos pesa 1 ml de ' + pv[i][1] + '?' : (conv === 'bot→ml' ? '¿Cuántos ml trae la botella de ' + pv[i][1] + '?' : '¿Cuánto pesa 1 ' + de + ' de ' + pv[i][1] + '?');

    it.push({ g: 2, peso: nf + nc, txt: (nf ? nf + ' de producción' : '') + (nf && nc ? ' · ' : '') + (nc ? nc + ' de cóctel' : '') || '—',
      que: que, como: String(pv[i][5] || ''), unidad: a + ' por ' + de, dest: H.PAR, clave: String(pv[i][0]) });
  }
  // 3) taras de botellas
  var ht = ss.getSheetByName(H.TAR);
  if (ht) {
    var tv = ht.getDataRange().getValues();
    for (i = 3; i < tv.length; i++) {
      if (!tv[i][0] || (tv[i][3] !== '' && tv[i][3] !== null)) { continue; }
      it.push({ g: 3, peso: 0, que: 'Peso de la botella VACÍA de ' + tv[i][0] + ' (' + tv[i][2] + ' ml)', como: 'Báscula en g. Botella vacía, limpia y seca, con tapa y pico, como se pesa en el conteo.',
        unidad: 'g', dest: H.TAR, clave: tv[i][0] + '|' + tv[i][2] });
    }
  }
  it.sort(function (x, y) { return x.g !== y.g ? x.g - y.g : y.peso - x.peso; });
  // hoja
  sh.clear();
  try { sh.getRange(1, 1, Math.max(sh.getMaxRows(), 10), 8).clearDataValidations(); } catch (e0) {}
  sh.getRange('A1:H1').setBackground('#0D3B33');
  sh.getRange('A1').setValue('SGI-COSTOS · PESAJE — lo que falta medir').setFontColor('#FFFFFF').setFontWeight('bold').setFontSize(13);
  sh.getRange('A2').setValue('Mide, escribe SOLO la columna VALOR (amarilla) y marca GUARDAR MEDIDAS. Puedes guardar de a pocas: lo guardado sale de la lista.')
    .setFontStyle('italic').setFontColor('#0D3B33');
  sh.getRange('B4').setValue('GUARDAR MEDIDAS ▶').setFontWeight('bold').setFontColor('#0D3B33');
  sh.getRange(PES.GUARDAR).insertCheckboxes();
  sh.getRange(PES.MSG + ':H5').merge().setWrap(true).setFontWeight('bold');
  sh.getRange(PES.MSG).setValue(msg || '');
  var enc = ['#', 'QUÉ MEDIR', 'CÓMO', 'VALOR', 'UNIDAD DEL VALOR', 'DESTRABA', 'VA A', 'CLAVE'];
  sh.getRange(PES.FILA_ENC, 1, 1, enc.length).setValues([enc]).setBackground('#B08D57').setFontColor('#FFFFFF').setFontWeight('bold');
  if (it.length) {
    var v = [];
    for (i = 0; i < it.length; i++) {
      v.push([i + 1, it[i].que, it[i].como, '', it[i].unidad, it[i].g === 1 ? 'todas las compras' : (it[i].g === 2 ? it[i].txt : 'inventario'), it[i].dest, it[i].clave]);
    }
    sh.getRange(PES.FILA1, 1, v.length, enc.length).setValues(v);
    sh.getRange(PES.FILA1, 4, v.length, 1).setBackground('#FFF2CC').setNumberFormat('#,##0.###')
      .setDataValidation(SpreadsheetApp.newDataValidation().requireNumberGreaterThan(0).setAllowInvalid(false).build());
    sh.getRange(PES.FILA1, 7, v.length, 2).setFontColor('#888888');
    sh.getRange(PES.FILA1, 2, v.length, 2).setWrap(true);
  }
  var an = [30, 330, 330, 80, 110, 100, 120, 200];
  for (i = 0; i < an.length; i++) { sh.setColumnWidth(i + 1, an[i]); }
  sh.setFrozenRows(PES.FILA_ENC);
  try { sh.setHiddenGridlines(true); } catch (e1) {}
  return it.length;
}

/** Cuántas líneas de receta de cóctel necesitan cada clave de PARAMETROS (conversión a la unidad base). */
function clavesCoctel_(ss, cat) {
  var out = {}, hr = ss.getSheetByName(H.REC);
  if (!hr) { return out; }
  var rv = hr.getDataRange().getValues();
  for (var i = 3; i < rv.length; i++) {
    var ins = cat.porCodigo[String(rv[i][4] || '')];
    var u = String(rv[i][7] || '').toLowerCase();
    if (!ins || !u || u === ins.ub || UNID_TRAZA_[u] || ins.tipo === 'SIN_COSTO') { continue; }
    if (ins.ub === 'und' && (u === 'bot' || u === 'lata')) { continue; }
    var k = ins.nombre + '|' + u + '→' + ins.ub;
    out[k] = (out[k] || 0) + 1;
  }
  return out;
}

/** Rango creíble para cada tipo de medida: un dígito de más no entra. */
function rangoPesaje_(dest, clave) {
  if (dest === H.TAR) { return [50, 3000]; }
  if (dest === H.FAC) { return [0.01, 100000]; }
  var conv = String(clave).split('|')[1] || '';
  if (conv === 'ml→g') { return [0.5, 2]; }
  if (conv === 'bot→ml') { return [100, 3000]; }
  if (conv === 'gota→ml') { return [0.01, 0.2]; }
  return [0.05, 5000];
}

function aplicarPesaje_(sh) {
