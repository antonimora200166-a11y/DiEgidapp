function ss_(){ return SpreadsheetApp.getActive(); }
function hoja_(n){
  var s = ss_().getSheetByName(n);
  if(!s) throw new Error('No existe la hoja "'+n+'". Revise que el archivo sea el FMT-REC-COC-003 y que se haya ejecutado aplicarParche34.');

}
function num_(v){
  if(v===''||v===null||v===undefined) return null;
  if(typeof v==='number') return isNaN(v)?null:v;
  var s=String(v).replace(/\s|\$/g,'').replace(/\.(?=\d{3}\b)/g,'').replace(',','.');
  var f=parseFloat(s);
  return isNaN(f)?null:f;
}
function txt_(v){ return (v===null||v===undefined)?'':String(v).trim(); }
function esFecha_(v){ return !!(v && typeof v.getTime==='function' && !isNaN(v.getTime())); }
function hoy_(){ var d=new Date(); return new Date(d.getFullYear(),d.getMonth(),d.getDate()); }
function dias_(a,b){ return Math.round((b-a)/86400000); }
function fmt_(d, patron){ return Utilities.formatDate(d, Session.getScriptTimeZone(), patron||'dd/MM/yyyy HH:mm'); }
function mediana_(a){
  if(!a.length) return null;
  var v=a.slice().sort(function(x,y){return x-y;}), m=Math.floor(v.length/2);
  return v.length%2 ? v[m] : (v[m-1]+v[m])/2;
}
function usuario_(){
  var e='';
  try{ e = Session.getActiveUser().getEmail()||''; }catch(x){}
  if(!e){ try{ e = Session.getEffectiveUser().getEmail()||''; }catch(x){} }
  return e || 'no-identificado';
}
function enLista_(v, lista){
  for(var i=0;i<lista.length;i++) if(lista[i]===v) return true;
  return false;
}

/** Lee la hoja CONFIG como objeto {PARAMETRO: valor}. */
function config_(){
  var v = hoja_(H.CFG).getDataRange().getValues(), c={};
  for(var i=0;i<v.length;i++){
    var k=txt_(v[i][0]);
    if(k && k.indexOf(' ')<0 && k===k.toUpperCase()) c[k]=v[i][1];
  }
  var resp=[], crudo=txt_(c.RESPONSABLES_CATALOGO).split(',');
  for(var j=0;j<crudo.length;j++){ var r=txt_(crudo[j]); if(r) resp.push(r); }
  if(!resp.length) resp=['Chef','Sous chef','Administración'];
  return {
    correoAdmin:      txt_(c.CORREO_ADMIN),
    correoChef:       txt_(c.CORREO_CHEF),
    umbralAlerta:     num_(c.UMBRAL_VARIACION_ALERTA) || 0.15,
    umbralBloqueo:    num_(c.UMBRAL_VARIACION_BLOQUEO) || 0.60,
    factorBanda:      num_(c.FACTOR_BANDA_PRECIO) || 20,
    factorProducto:   num_(c.FACTOR_BANDA_PRODUCTO) || 3,
    diasLegalizacion: num_(c.DIAS_LEGALIZACION) || 2,
    diasEscalamiento: num_(c.DIAS_ESCALAMIENTO) || 5,
    minMotivo:        num_(c.MIN_CARACTERES_MOTIVO) || 15,
    exigirVenc:       txt_(c.EXIGIR_VENCIMIENTO_REFRIG).toUpperCase()!=='NO',
    exigirTemp:       txt_(c.EXIGIR_TEMP_REFRIG).toUpperCase()!=='NO',
    responsables:     resp
  };
}

/** Lee el bloque de datos del CATALOGO con las columnas que existan (antes del parche son 8). */
function leerCatalogoCrudo_(){
  var s=hoja_(H.CAT), n=s.getLastRow()-CAT_F_INI+1;
  if(n<1) return [];
  var nc=Math.min(C.N, s.getMaxColumns());
  var v=s.getRange(CAT_F_INI,1,n,nc).getValues();
  for(var i=0;i<v.length;i++) while(v[i].length<C.N) v[i].push('');
  return v;
}

/** Catálogo indexado por nombre de producto (activos e inactivos). */
function catalogo_(){
  var v=leerCatalogoCrudo_(), m={};
  for(var i=0;i<v.length;i++){
    var nom=txt_(v[i][C.NOMBRE-1]);
    if(!nom) continue;
    var vig=txt_(v[i][C.VIG-1]).toUpperCase()===INACTIVO ? INACTIVO : ACTIVO;
    m[nom]={nombre:nom, base:txt_(v[i][C.BASE-1]), cat:txt_(v[i][C.CAT-1]),
            cons:txt_(v[i][C.CONS-1]), tmax:num_(v[i][C.TMAX-1]),
            vig:vig, mov:txt_(v[i][C.MOV-1]), fila:CAT_F_INI+i};
  }
  return m;
}
function activos_(cat){
  var out={}, k=Object.keys(cat);
  for(var i=0;i<k.length;i++) if(cat[k[i]].vig===ACTIVO) out[k[i]]=cat[k[i]];
  return out;
}

/** Libro mayor completo como matriz. */
function leerBD_(){

  if(n<1) return [];
  return s.getRange(BD_F_INI,1,n,B.N).getValues().filter(function(f){ return txt_(f[B.ID_LINEA-1])!==''; });
}

/** Bandas de plausibilidad: {cat|base: {inf,sup}} con respaldo (GLOBAL). */
function bandas_(){
  var s=ss_().getSheetByName(H.BAN); if(!s) return {};
  var n=s.getLastRow()-5; if(n<1) return {};
  var v=s.getRange(6,1,n,7).getValues(), m={};
  for(var i=0;i<v.length;i++){
    var c=txt_(v[i][0]), b=txt_(v[i][1]);
    if(!c||!b) continue;
    m[c+'|'+b]={inf:num_(v[i][4]), sup:num_(v[i][5])};
  }
  return m;
}
function bandaDe_(bn, cat, base){
  return bn[cat+'|'+base] || bn['(GLOBAL)|'+base] || null;
}

/** Un N° de documento tiene que identificar algo: ni relleno ni cadena vacía disfrazada. */
function documentoValido_(doc){
  var d=txt_(doc).replace(/\s/g,'');
  if(d.length<4) return false;
  if(/^[0\-\.\_\/]+$/.test(d)) return false;
  if(['na','n/a','no','sin','ninguno','nose','xxxx','1234','0000'].indexOf(d.toLowerCase())>=0) return false;
  if(!/[0-9]/.test(d)) return false;
  return true;
}

/** Escribe una línea en BITACORA (append-only). */
function bitacora_(accion, resultado, idRec, nLineas, motivo, detalle){
  try{
    hoja_(H.BIT).appendRow([new Date(), usuario_(), accion, resultado,
      idRec||'', nLineas||0, motivo||'', String(detalle||'').substring(0,1000)]);
  }catch(x){}
}

/** Mensaje de sistema en la hoja RECEPCION. */
function mensaje_(texto, tipo){
  var s=hoja_(H.REC), c=s.getRange(REC.MENSAJE);
  c.setValue(texto);
  var col = tipo==='ok' ? '#1A6B5C' : (tipo==='warn' ? '#9C6500' : '#CC0000');
  var fon = tipo==='ok' ? '#E2EFDA' : (tipo==='warn' ? '#FFF2CC' : '#F4CCCC');
  s.getRange(REC.MENSAJE+':V62').setBackground(fon).setFontColor(col).setFontWeight('bold');
  try{ ss_().toast(String(texto).substring(0,240), tipo==='ok'?'Guardado':'Revisar', tipo==='ok'?6:12); }catch(x){}
}

function alEditar(e){
  var nombreHoja='', a1='';
  try{
    if(!e || !e.range) return;
    var h = e.range.getSheet();
    nombreHoja = h.getName();
    a1 = e.range.getA1Notation();
    if(nombreHoja===H.REC){
      if(a1===REC.GUARDAR && e.range.getValue()===true) guardarRecepcion();
      return;
    }
    if(nombreHoja===H.PRD){
      if(a1===P.A_CHECK && e.range.getValue()===true) crearProductoHoja();
      else if(a1===P.B_CHECK && e.range.getValue()===true) bajaProductoHoja();
      return;
    }
    if(nombreHoja===H.CFG){
      // al cambiar la lista de responsables, las listas desplegables se actualizan solas
      var fila=e.range.getRow(), k=txt_(h.getRange(fila,1).getValue());
      if(k==='RESPONSABLES_CATALOGO') actualizarResponsables_();
      return;
    }
    if(nombreHoja===H.DUP){
      if(a1===DUPC.CHECK && e.range.getValue()===true) aplicarDecisionesDuplicados();
      return;
    }
  }catch(err){
    bitacora_(nombreHoja===H.REC?'GUARDAR':'CATÁLOGO','ERROR TÉCNICO','',0,'',err.message);
    try{
      if(nombreHoja===H.REC){
        mensaje_('⛔ Error inesperado: '+err.message, 'err');
        hoja_(H.REC).getRange(REC.GUARDAR).setValue(false);
      } else if(nombreHoja===H.PRD){
        var msg = (a1===P.B_CHECK) ? P.B_MSG : P.A_MSG;
        msgProductos_(msg, '⛔ Error inesperado: '+err.message, 'err');
        hoja_(H.PRD).getRange(a1).setValue(false);
      } else if(nombreHoja===H.DUP){
        hoja_(H.DUP).getRange(DUPC.AVISO).setValue('⛔ Error inesperado: '+err.message);
        hoja_(H.DUP).getRange(DUPC.CHECK).setValue(false);
      }
    }catch(x){}
  }
}

// 03 · NÚCLEO — guardado atómico
// ===========================================================================
function guardarRecepcion(){
  var lock = LockService.getDocumentLock();
  if(!lock.tryLock(25000)){
    mensaje_('⏳ Otro guardado está en curso. Espere unos segundos y vuelva a marcar la casilla.','warn');
    hoja_(H.REC).getRange(REC.GUARDAR).setValue(false);
    return;
  }
  try{
    var s   = hoja_(H.REC);

    // --- lectura del encabezado y de las líneas ---
    var enc = {
      fecha:     s.getRange(REC.FECHA).getValue(),
      soporte:   txt_(s.getRange(REC.SOPORTE).getValue()),
      proveedor: txt_(s.getRange(REC.PROVEEDOR).getValue()),
      documento: txt_(s.getRange(REC.DOCUMENTO).getValue()),
      recibe:    txt_(s.getRange(REC.RECIBE).getValue()),
      transporte:txt_(s.getRange(REC.TRANSPORTE).getValue()),
      precinto:  txt_(s.getRange(REC.PRECINTO).getValue())
    };
    var bruto = s.getRange(REC.F_INI,1,REC.F_FIN-REC.F_INI+1,REC.N_COL).getValues();
    var lineas = [];
    for(var i=0;i<bruto.length;i++){
      if(txt_(bruto[i][R.PRODUCTO-1])!=='') lineas.push({fila:REC.F_INI+i, v:bruto[i]});
    }

    // --- formulario vacío: casi siempre es la casilla marcada otra vez justo
    //     después de un guardado exitoso (el iPad tarda en mostrar que se limpió).
    //     No es un rechazo: no se registra y no se tapa el mensaje de éxito.
    if(!txt_(enc.fecha) && !enc.proveedor && !enc.soporte && !enc.recibe && !lineas.length){
      var ult=ultimoGuardado_();
      if(ult && (new Date().getTime()-ult.t) < 5*60000){
        try{ ss_().toast('La recepción '+ult.id+' ya quedó guardada. No hace falta volver a marcar.','Ya está guardada',8); }catch(x){}
      } else {
        mensaje_('⬜ El formulario está vacío: no hay nada que guardar.','warn');
      }
      return;
    }

    var cfg = config_();
    var cat = catalogo_();
    var bn  = bandas_();
    var bd  = leerBD_();

    var forzar = s.getRange(REC.FORZAR).getValue()===true;
    var motivo = txt_(s.getRange(REC.MOTIVO).getValue());

    // --- compuertas de validación ---
    var val = validar_(enc, lineas, cat, bn, bd, cfg);

    // Errores que NO admiten forzado nunca: se informan primero
    if(val.fatales.length){
      mensaje_('⛔ NO SE GUARDÓ y no admite forzado:\n• '+val.fatales.join('\n• '),'err');
      bitacora_('GUARDAR','RECHAZADO: ERROR FATAL','',lineas.length,motivo,val.fatales.join(' | '));
      return;
    }
    if(val.bloqueos.length && !forzar){
      var m = '⛔ NO SE GUARDÓ. '+val.bloqueos.length+' condición(es) lo impiden:\n• '+
              val.bloqueos.join('\n• ')+
              '\n\nCorrija y vuelva a marcar la casilla. Si el pedido debe quedar registrado igual, '+
              'escriba el motivo en la celda '+REC.MOTIVO+', marque «Forzar guardado» y guarde de nuevo.';
      mensaje_(m,'err');
      bitacora_('GUARDAR','RECHAZADO','',lineas.length,'',val.bloqueos.join(' | '));
      return;
    }
    if(val.bloqueos.length && forzar && motivo.length < cfg.minMotivo){
      mensaje_('⛔ El forzado exige un motivo escrito de al menos '+cfg.minMotivo+
               ' caracteres en la celda '+REC.MOTIVO+'. Quedará firmado con su nombre en la BITÁCORA.','err');
      bitacora_('GUARDAR','RECHAZADO: MOTIVO INSUFICIENTE','',lineas.length,motivo,val.bloqueos.join(' | '));
      return;
    }

    // --- asignación de identidad (consecutivo real, a prueba de colisiones) ---
    var f = new Date(enc.fecha);
    var yyyymmdd = fmt_(f, 'yyyyMMdd');
    var usados = {};
    for(var k=0;k<bd.length;k++) usados[txt_(bd[k][B.ID_REC-1])]=true;
    var nn=1, idRec;
    do { idRec = 'REC-COC-'+yyyymmdd+'-'+('0'+nn).slice(-2); nn++; } while(usados[idRec] && nn<100);

    // --- construcción de las filas del libro mayor ---
    var ts = new Date(), usr = usuario_(), ovr = (val.bloqueos.length && forzar) ? 'SÍ' : 'NO';
    var filas = [];
    for(var j=0;j<lineas.length;j++){
      var v = lineas[j].v, c = cat[txt_(v[R.PRODUCTO-1])] || {};
      var fila = new Array(B.N);
      fila[B.ID_LINEA-1]  = idRec+'-'+('00'+(j+1)).slice(-3);
      fila[B.ID_REC-1]    = idRec;
      fila[B.FECHA-1]     = f;
      fila[B.PROV-1]      = enc.proveedor;
      fila[B.TIPO-1]      = enc.soporte;
      fila[B.DOC-1]       = enc.documento;
      fila[B.RECIBE-1]    = enc.recibe;
      fila[B.PRODUCTO-1]  = txt_(v[R.PRODUCTO-1]);
      fila[B.BASE-1]      = txt_(v[R.BASE-1]) || c.base || '';
      fila[B.PRESENT-1]   = txt_(v[R.PRESENT-1]);
      fila[B.CANT_P-1]    = num_(v[R.CANT_P-1]);
      fila[B.CONT_P-1]    = num_(v[R.CONT_P-1]);
      fila[B.CANT_FACT-1] = num_(v[R.CANT_FACT-1]);
      fila[B.CANT_PED-1]  = num_(v[R.CANT_PED-1]);
      fila[B.CANT_REAL-1] = num_(v[R.CANT_REAL-1]);
      fila[B.DIF-1]       = num_(v[R.DIF-1]);
      fila[B.VALOR-1]     = num_(v[R.VALOR-1]);
      fila[B.PU-1]        = num_(v[R.PU-1]);
      fila[B.MERMA-1]     = num_(v[R.MERMA-1]);
      fila[B.COND-1]      = txt_(v[R.COND-1]);
      fila[B.CONS-1]      = txt_(v[R.CONS-1]) || c.cons || '';
      fila[B.TEMP-1]      = v[R.TEMP-1];
      fila[B.TEMP_OK-1]   = txt_(v[R.TEMP_OK-1]);
      fila[B.VENC-1]      = v[R.VENC-1];
      fila[B.LOTE-1]      = txt_(v[R.LOTE-1]);
      fila[B.OBS-1]       = txt_(v[R.OBS-1]);
      fila[B.ORIGEN-1]    = VERSION;
      fila[B.REVISION-1]  = val.revision[lineas[j].fila] || '';
      fila[B.USUARIO-1]   = usr;
      fila[B.TS-1]        = ts;
      fila[B.OVERRIDE-1]  = ovr;
      fila[B.MOTIVO-1]    = ovr==='SÍ' ? motivo : '';
      fila[B.ESTADO-1]    = 'ACTIVA';
      fila[B.REF_REV-1]   = '';
      filas.push(fila);
    }

    // --- escritura en bloque, siempre al final (append-only) ---
    var hb = hoja_(H.BD);
    var destino = Math.max(hb.getLastRow()+1, BD_F_INI);
    if(destino + filas.length > hb.getMaxRows()) hb.insertRowsAfter(hb.getMaxRows(), filas.length+200);
    hb.getRange(destino,1,filas.length,B.N).setValues(filas);
    hb.getRange(destino,B.FECHA,filas.length,1).setNumberFormat('dd/mm/yyyy');
    hb.getRange(destino,B.TS,filas.length,1).setNumberFormat('dd/mm/yyyy hh:mm');
    SpreadsheetApp.flush();
    marcarUltimoGuardado_(idRec);

    // --- limpieza del formulario ANTES del recálculo: si el recálculo tarda,
    //     el formulario ya se ve limpio y nadie vuelve a marcar la casilla.
    limpiarFormulario_();
    var aviso = val.avisos.length ? '\n⚠ Quedó registrado con '+val.avisos.length+' aviso(s): '+val.avisos.join(' | ') : '';
    var txtOvr = ovr==='SÍ' ? '\n🔴 GUARDADO FORZADO. Motivo firmado en BITÁCORA.' : '';
    mensaje_('✅ Recepción guardada como '+idRec+' — '+filas.length+' línea(s). No hace falta volver a marcar la casilla.'+
             txtOvr+aviso, ovr==='SÍ'?'warn':'ok');
    bitacora_('GUARDAR', ovr==='SÍ'?'GUARDADO CON OVERRIDE':'GUARDADO', idRec, filas.length, ovr==='SÍ'?motivo:'',
              val.avisos.join(' | '));

    // --- derivados: se recalculan en el único momento en que pueden cambiar ---
    try{ recalcularTodo(); }
    catch(eR){ bitacora_('RECÁLCULO','ERROR',idRec,0,'',eR.message+' — la recepción sí quedó guardada.'); }

  } finally {
    try{ hoja_(H.REC).getRange(REC.GUARDAR).setValue(false); }catch(x){}
    lock.releaseLock();
  }
}

function marcarUltimoGuardado_(id){
  try{ PropertiesService.getDocumentProperties().setProperty('ultimo_guardado',
       JSON.stringify({id:id, t:new Date().getTime()})); }catch(x){}
}
function ultimoGuardado_(){
  try{
    var s=PropertiesService.getDocumentProperties().getProperty('ultimo_guardado');
    return s ? JSON.parse(s) : null;
  }catch(x){ return null; }
}

/** Borra solo las celdas de captura. Nunca toca fórmulas. */
function limpiarFormulario_(){
  var s=hoja_(H.REC);
  var enc=[REC.FECHA,REC.SOPORTE,REC.PROVEEDOR,REC.DOCUMENTO,REC.RECIBE,REC.TRANSPORTE,REC.PRECINTO,REC.HORA];
  for(var i=0;i<enc.length;i++) s.getRange(enc[i]).clearContent();
  var manual=[R.PRODUCTO,R.PRESENT,R.CANT_P,R.CONT_P,R.CANT_PED,R.CANT_REAL,R.VALOR,
              R.COND,R.TEMP,R.VENC,R.LOTE,R.OBS];
  var n=REC.F_FIN-REC.F_INI+1;
  for(var j=0;j<manual.length;j++) s.getRange(REC.F_INI,manual[j],n,1).clearContent();
  s.getRange(REC.FORZAR).setValue(false);
  s.getRange(REC.MOTIVO).clearContent();
}

function validar_(enc, lineas, cat, bn, bd, cfg){
  var bloqueos=[], avisos=[], fatales=[], revision={};

  // G1 — encabezado completo
  if(!esFecha_(enc.fecha)) fatales.push('G1 · La fecha de recepción está vacía o no es una fecha válida.');
  if(!enc.proveedor) fatales.push('G1 · Falta el proveedor.');
  if(!enc.soporte)   fatales.push('G1 · Falta el tipo de soporte.');
  if(!enc.recibe)    fatales.push('G1 · Falta el nombre de quien recibe.');

  // G2 — fecha no futura
  if(esFecha_(enc.fecha)){
    var d=new Date(enc.fecha.getFullYear(),enc.fecha.getMonth(),enc.fecha.getDate());
    if(dias_(hoy_(),d) > 0) fatales.push('G2 · La fecha de recepción es futura ('+fmt_(d,'dd/MM/yyyy')+').');
    if(dias_(d,hoy_()) > 60) avisos.push('G2 · La recepción tiene más de 60 días de antigüedad.');
  }

  // G3 — al menos una línea
  if(!lineas.length) fatales.push('G3 · No hay ningún producto registrado.');
  if(fatales.length) return {bloqueos:bloqueos, avisos:avisos, fatales:fatales, revision:revision};

  // G4 — soporte documental
  if(enc.soporte==='Sin soporte'){
    bloqueos.push('G4 · La recepción no tiene factura ni remisión.');
  } else if(!enc.documento){
    bloqueos.push('G4 · Falta el N° de documento del soporte.');
  } else if(!documentoValido_(enc.documento)){
    bloqueos.push('G4 · «'+enc.documento+'» no es un número de documento válido. '+
                  'Debe ser el identificador real de la factura o remisión, con al menos 4 caracteres.');
  }

  // G10 — idempotencia
  if(enc.documento){
    for(var k=0;k<bd.length;k++){
      if(txt_(bd[k][B.DOC-1])===enc.documento && txt_(bd[k][B.PROV-1])===enc.proveedor
         && txt_(bd[k][B.ESTADO-1])==='ACTIVA'){
        fatales.push('G10 · El documento '+enc.documento+' de '+enc.proveedor+
                     ' ya está registrado en el libro mayor ('+txt_(bd[k][B.ID_REC-1])+'). '+
                     'Si es una corrección, use la función reversarRecepcion en lugar de volver a guardar.');
        break;
      }
    }
  }

  // Histórico por producto, LIMPIO (sin precios fuera de banda)
  var hist={};
  for(var i=0;i<bd.length;i++){
    if(txt_(bd[i][B.ESTADO-1])!=='ACTIVA') continue;
    var p=txt_(bd[i][B.PRODUCTO-1]), pu=num_(bd[i][B.PU-1]);
    if(!p || !pu || pu<=0) continue;
    var cc=cat[p]; if(!cc) continue;
    var bb=bandaDe_(bn, cc.cat, cc.base);
    if(bb && bb.inf && bb.sup && (pu<bb.inf || pu>bb.sup)) continue;
    (hist[p]=hist[p]||[]).push(pu);
  }

  var vistos={};
  for(var j=0;j<lineas.length;j++){
    var L=lineas[j], v=L.v, f=L.fila, pre='Fila '+f+' ';
    var prod=txt_(v[R.PRODUCTO-1]);
    var c=cat[prod];

    // G5 — integridad referencial (se revalida aquí porque un pegado puede saltarse la lista)
    if(!c){ fatales.push('G5 · '+pre+'— «'+prod+'» no existe en el CATÁLOGO. Créelo en la hoja PRODUCTOS (bloque 1) y vuelva a marcar.'); continue; }
    if(c.vig===INACTIVO){
      fatales.push('G5 · '+pre+'— «'+prod+'» está dado de baja'+(c.mov?(' ('+c.mov+')'):'')+
                   '. Si se volvió a comprar, reactívelo en la hoja PRODUCTOS (bloque 2) y vuelva a marcar.');
      continue;
    }

    // G6 — producto repetido en la misma recepción
    if(vistos[prod]) avisos.push('G6 · '+prod+' aparece más de una vez en esta recepción (filas '+vistos[prod]+' y '+f+').');
    else vistos[prod]=f;

    // G7 — cantidades numéricas y coherentes
    var cp=num_(v[R.CANT_P-1]), cont=num_(v[R.CONT_P-1]), cr=num_(v[R.CANT_REAL-1]);
    if(cp===null||cp<=0)   bloqueos.push('G7 · '+pre+'— falta la cantidad de presentaciones.');
    if(cont===null||cont<=0) bloqueos.push('G7 · '+pre+'— falta el contenido por presentación en '+(c.base||'unidad base')+'.');
    if(cr===null||cr<0)   bloqueos.push('G7 · '+pre+'— falta la cantidad REAL recibida (hay que pesarla o contarla).');

    // G8 — valor de la factura
    var val=num_(v[R.VALOR-1]), pu2=num_(v[R.PU-1]);
    if(val===null||val<=0){
      bloqueos.push('G8 · '+pre+'— falta el valor de la línea en la factura. Sin eso no hay costo.');
      revision[f]='SIN PRECIO — cargar desde factura';
    }

    // G9 — doble banda de control de precio
    if(pu2 && pu2>0){
      var b=bandaDe_(bn, c.cat, c.base);
      if(b && b.inf && b.sup && (pu2 < b.inf || pu2 > b.sup)){
        bloqueos.push('G9 · '+pre+'— '+prod+' a '+pu2.toFixed(2)+' por '+c.base+
          ' está fuera de la banda razonable de '+c.cat+' ('+b.inf.toFixed(2)+' a '+b.sup.toFixed(2)+
          '). Casi siempre es el precio del paquete puesto como precio por '+c.base+
          ': revise el contenido por presentación.');
        revision[f]='PRECIO FUERA DE BANDA — verificar presentación';
      } else if(hist[prod] && hist[prod].length>=1){
        var med=mediana_(hist[prod]);
        if(med>0){
          var q=pu2/med;
          if(q>cfg.factorProducto || q<1/cfg.factorProducto){
            bloqueos.push('G9 · '+pre+'— '+prod+' a '+pu2.toFixed(2)+' se sale '+
              (q>1?(q.toFixed(1)+' veces por encima'):((1/q).toFixed(1)+' veces por debajo'))+
              ' de su precio habitual ('+med.toFixed(2)+').');
            revision[f]='PRECIO ATÍPICO FRENTE AL HISTÓRICO';
          } else if(Math.abs(q-1) > cfg.umbralBloqueo){
            bloqueos.push('G9 · '+pre+'— '+prod+' varió '+((q-1)*100).toFixed(0)+'% frente a su precio habitual.');
          } else if(Math.abs(q-1) > cfg.umbralAlerta){
            avisos.push('G9 · '+prod+' varió '+((q-1)*100).toFixed(0)+'% frente a su precio habitual.');
          }
        }
      }
    }

    // G11 — cadena de frío y trazabilidad del perecedero
    var cons=txt_(v[R.CONS-1]) || c.cons;
    if(cons && cons!=='Ambiente'){
      var t=num_(v[R.TEMP-1]);
      if(cfg.exigirTemp && t===null){
        bloqueos.push('G11 · '+pre+'— '+prod+' es '+cons.toLowerCase()+' y no tiene temperatura de llegada medida.');
      } else if(t!==null && c.tmax!==null && t>c.tmax){
        bloqueos.push('G11 · '+pre+'— '+prod+' llegó a '+t+' °C y el máximo es '+c.tmax+
          ' °C. Debe rechazarse o retenerse, y notificar a Chef y Administración.');
        revision[f]='TEMPERATURA FUERA DE RANGO';
      }
      if(cfg.exigirVenc && !v[R.VENC-1]){
        bloqueos.push('G11 · '+pre+'— '+prod+' es '+cons.toLowerCase()+' y no tiene fecha de vencimiento.');
      }
    }
    if(esFecha_(v[R.VENC-1])){
      var dv=dias_(hoy_(), new Date(v[R.VENC-1]));
      if(dv<0)      bloqueos.push('G11 · '+pre+'— '+prod+' llegó vencido.');
      else if(dv<=3) avisos.push('G11 · '+prod+' vence en '+dv+' día(s).');
    }
    var dif=num_(v[R.DIF-1]);
    if(dif!==null && dif<0){
      var mer=num_(v[R.MERMA-1]);
      avisos.push('G7 · '+prod+': llegaron '+Math.abs(dif)+' '+c.base+' menos de lo facturado'+
                  (mer?(' ('+mer.toFixed(0)+' pesos)'):'')+'. Descontar en la factura.');
    }
  }
  return {bloqueos:bloqueos, avisos:avisos, fatales:fatales, revision:revision};
}

function recalcularTodo(){
  var cfg=config_(), cat=catalogo_(), bd=leerBD_();
  var act=bd.filter(function(f){ return txt_(f[B.ESTADO-1])==='ACTIVA'; });
  var mae=recalcularMaestro_(cat, act);
  var leg=recalcularLegalizacion_(act, cfg);
  recalcularProveedores_(act, leg);
  recalcularBandas_(cat, act, cfg);
  var pares=[];
  try{ pares=refrescarDuplicados_(cat, act); }catch(e){ bitacora_('DUPLICADOS','ERROR','',0,'',e.message); }
  var diag=recalcularDiagnostico_(cat, bd, act, mae, leg, cfg, pares);
  try{ recalcularTablero_(cat, act, mae, leg, diag); }catch(e){ bitacora_('TABLERO','ERROR','',0,'',e.message); }
  SpreadsheetApp.flush();   // cualquier error de escritura pendiente sale aquí, no en el siguiente bloque
  refrescarAvisos_(cat, act, mae, leg, diag, pares);
  try{ refrescarListas_(cat); }catch(e){ bitacora_('LISTAS','ERROR','',0,'',e.message); }
  SpreadsheetApp.flush();
  return {mae:mae, leg:leg, diag:diag};
}

function recalcularMaestro_(cat, act){
  var porProd={};
  for(var i=0;i<act.length;i++){
    var p=txt_(act[i][B.PRODUCTO-1]), pu=num_(act[i][B.PU-1]);
    if(!p||!pu||pu<=0) continue;
    (porProd[p]=porProd[p]||[]).push({pu:pu, f:act[i][B.FECHA-1], prov:txt_(act[i][B.PROV-1])});
  }
  var nombres=Object.keys(cat).sort(function(a,b){ return a.toLowerCase()<b.toLowerCase()?-1:1; });
  var out=[], res=[];
  for(var k=0;k<nombres.length;k++){
    var nom=nombres[k], c=cat[nom], h=(porProd[nom]||[]).slice();
    h.sort(function(x,y){ return (x.f||0)-(y.f||0); });
    if(!h.length){
      out.push([k+1,nom,c.base,c.cat,0,'','','','','','','','',
                c.vig===INACTIVO?'⏸ Dado de baja · sin precio':'⬜ Sin precio registrado','']);
      res.push({nombre:nom,n:0,med:null,var:null,estado:'⬜',vig:c.vig}); continue;
    }
    var ps=h.map(function(x){return x.pu;});
    var mn=Math.min.apply(null,ps), mx=Math.max.apply(null,ps);
    var prom=ps.reduce(function(a,b){return a+b;},0)/ps.length, med=mediana_(ps);
    var u=h[h.length-1], vr = mn>0 ? (mx-mn)/mn : '';
    var est = ps.length===1 ? '🔵 Un solo dato — sin validar'
            : (vr>0.4 ? '🔴 Variación crítica >40%'
            : (vr>0.15 ? '🟠 Variación alta 15–40%' : '🟢 Precio estable'));
    if(c.vig===INACTIVO) est='⏸ Dado de baja · '+est;
    out.push([k+1,nom,c.base,c.cat,ps.length,u.pu,u.f,u.prov,mn,mx,prom,med,vr,est,
              c.base==='und'?'':u.pu*1000]);
    res.push({nombre:nom,n:ps.length,med:med,var:vr,estado:c.vig===INACTIVO?'⏸':est.substring(0,2),vig:c.vig});
  }
  escribirBloque_(H.MAE, 6, 15, out);
  return res;
}

function recalcularLegalizacion_(act, cfg){
  var ord=[], m={};
  for(var i=0;i<act.length;i++){
    var id=txt_(act[i][B.ID_REC-1]);
    if(!m[id]){ m[id]={id:id,fecha:act[i][B.FECHA-1],prov:txt_(act[i][B.PROV-1]),
      tipo:txt_(act[i][B.TIPO-1]),doc:txt_(act[i][B.DOC-1]),n:0,val:0,sinp:0,ovr:0}; ord.push(m[id]); }
    var r=m[id]; r.n++; r.val += num_(act[i][B.VALOR-1])||0;
    if(!num_(act[i][B.PU-1])) r.sinp++;
    if(txt_(act[i][B.OVERRIDE-1])==='SÍ') r.ovr++;
  }
  ord.sort(function(a,b){ return (a.fecha||0)-(b.fecha||0); });
  var out=[], res=[];
  for(var j=0;j<ord.length;j++){
    var r2=ord[j], dias = r2.fecha ? dias_(new Date(r2.fecha), hoy_()) : '';
    var est;
    if(r2.tipo==='Sin soporte') est = (dias!=='' && dias>cfg.diasEscalamiento) ? '🔴 VENCIDA — escalar a Gerencia' : '🟠 Pendiente de factura';
    else if(!r2.doc)  est='🟠 Falta N° de documento';
    else if(r2.sinp)  est='🟠 Hay líneas sin precio';
    else              est='🟢 Legalizada';
    out.push([j+1,r2.id,r2.fecha,r2.prov,r2.tipo,r2.doc,r2.n,r2.val,r2.sinp,r2.ovr,dias,est]);
    res.push({id:r2.id,prov:r2.prov,tipo:r2.tipo,dias:dias,estado:est,val:r2.val,sinp:r2.sinp});
  }
  escribirBloque_(H.LEG, 6, 12, out);
  return res;
}

function recalcularProveedores_(act, leg){
  var s=hoja_(H.PRV), n=s.getLastRow()-6+1;
  if(n<1) return;
  var nom=s.getRange(6,2,n,1).getValues();
  var out=[];
  for(var i=0;i<n;i++){
    var p=txt_(nom[i][0]);
    if(!p){ out.push(['','','','','','']); continue; }
    var li=0, val=0, mer=0, sins=0;
    for(var j=0;j<act.length;j++){
      if(txt_(act[j][B.PROV-1])!==p) continue;
      li++; val += num_(act[j][B.VALOR-1])||0; mer += num_(act[j][B.MERMA-1])||0;
      if(txt_(act[j][B.TIPO-1])==='Sin soporte') sins++;
    }
    var nrec=0;
    for(var q=0;q<leg.length;q++) if(leg[q].prov===p) nrec++;
    var est = !li ? '⬜ Sin movimientos'
            : (sins/li>0.5 ? '🔴 Más de la mitad sin soporte'
            : (sins>0 ? '🟠 Tiene líneas sin soporte'
            : (mer>0 ? '🟠 Registra merma facturada' : '🟢 Al día')));
    out.push([nrec, li, val, mer, sins, est]);
  }
  s.getRange(6,5,n,6).setValues(out);
}

function recalcularBandas_(cat, act, cfg){
  var F=cfg.factorBanda, g={}, gl={};
  for(var i=0;i<act.length;i++){
    var p=txt_(act[i][B.PRODUCTO-1]), pu=num_(act[i][B.PU-1]), c=cat[p];
    if(!c||!pu||pu<=0) continue;
    (g[c.cat+'|'+c.base]=g[c.cat+'|'+c.base]||[]).push(pu);
    (gl[c.base]=gl[c.base]||[]).push(pu);
  }
  function rob(v){
    if(!v.length) return null;
    var m0=mediana_(v), k=v.filter(function(x){ return x>=m0/F && x<=m0*F; });
    return mediana_(k.length?k:v);
  }
  var out=[], gm={};
  for(var b=0;b<BASES.length;b++){
    var m=rob(gl[BASES[b]]||[]);
    gm[BASES[b]]=m;
    if(m) out.push(['(GLOBAL)',BASES[b],(gl[BASES[b]]||[]).length,m,m/F,m*F,'Global por unidad base']);
  }
  var claves=Object.keys(g).sort();
  for(var k=0;k<claves.length;k++){
    var p2=claves[k].split('|'), v=g[claves[k]], m2=rob(v);
    if(m2 && v.length>=5){
      out.push([p2[0],p2[1],v.length,m2,m2/F,m2*F,'Histórico propio de la categoría']);
    } else if(gm[p2[1]]){
      var FH=F*3;
      out.push([p2[0],p2[1],v.length,gm[p2[1]],gm[p2[1]]/FH,gm[p2[1]]*FH,
        'Heredada y ensanchada ×3 — solo '+v.length+' dato'+(v.length===1?'':'s')+' propios']);
    }
  }
  escribirBloque_(H.BAN, 6, 7, out);
}

function recalcularDiagnostico_(cat, bd, act, mae, leg, cfg, pares){
  var D=[];
  function add(cod,sev,tit,n,det){ D.push([cod,sev,tit,n,det]); }
  var ids={}, dup=0;
  act.forEach(function(f){ var i=txt_(f[B.ID_LINEA-1]); if(ids[i]) dup++; ids[i]=1; });
  add('INT-01','CRÍTICO','Claves primarias duplicadas en el libro mayor',dup,
    dup?'Hay un guardado doble. Use reversarRecepcion sobre la copia.':'Sin duplicados: la idempotencia está funcionando.');
  var hu={}, nhu=0;
  act.forEach(function(f){ var p=txt_(f[B.PRODUCTO-1]); if(p && !cat[p] && !hu[p]){hu[p]=1;nhu++;} });
  add('INT-02','CRÍTICO','Productos del libro mayor que no existen en el catálogo',nhu,
    nhu?Object.keys(hu).join(', ').substring(0,200):'Integridad referencial correcta.');
  var sinPrecio=act.filter(function(f){ return !num_(f[B.PU-1]); });
  var cerradas=sinPrecio.filter(function(f){ return txt_(f[B.REVISION-1])===MARCA_CERRADA; }).length;
  var recuperables=sinPrecio.length-cerradas;
  add('INT-03','ALTO','Líneas sin precio que aún se pueden recuperar',recuperables,
    recuperables?'Pedir la factura al proveedor y cargar el valor.':'Nada pendiente de recuperar.');
  var bn=bandas_(), fuera={}, nf=0;
  act.forEach(function(f){
    var p=txt_(f[B.PRODUCTO-1]), pu=num_(f[B.PU-1]), c=cat[p];
    if(!c||!pu) return;
    var b=bandaDe_(bn,c.cat,c.base);
    if(b && (pu<b.inf||pu>b.sup) && !fuera[p]){ fuera[p]=1; nf++; }
  });
  add('INT-04','ALTO','Productos con precio fuera de la banda de control',nf,
    nf?Object.keys(fuera).join(', ').substring(0,200):'Todos los precios dentro de la banda.');
  var nb=act.filter(function(f){ var b=txt_(f[B.BASE-1]); return b!=='g'&&b!=='ml'&&b!=='und'; }).length;
  add('INT-05','ALTO','Líneas con unidad base no normalizada',nb, nb?'Solo se admite g, ml o und.':'Unidades normalizadas.');
  var sd=leg.filter(function(l){ return l.tipo==='Sin soporte'; });
  add('INT-06','ALTO','Recepciones sin soporte documental',sd.length,
    sd.length?sd.map(function(l){return l.id;}).join(', ').substring(0,200):'Todas con soporte.');
  var vz=leg.filter(function(l){ return l.estado.indexOf('🔴')===0; });
  add('INT-07','CRÍTICO','Recepciones vencidas para escalar a Gerencia',vz.length,
    vz.length?vz.map(function(l){return l.id;}).join(', ').substring(0,200):'Ninguna vencida.');
  var ov=act.filter(function(f){ return txt_(f[B.OVERRIDE-1])==='SÍ'; }).length;
  add('INT-08','MEDIO','Líneas guardadas con override',ov, ov?'Revisar el motivo en BITACORA.':'Ningún guardado forzado.');
  var rv=bd.filter(function(f){ return txt_(f[B.ESTADO-1])==='REVERSADA'; }).length;
  add('INT-09','MEDIO','Líneas reversadas',rv, rv?'Correcciones registradas; el histórico se conserva íntegro.':'Sin reversas.');
  var su=act.filter(function(f){ var u=txt_(f[B.USUARIO-1]); return !u||u==='no-identificado'||u==='migracion@sistema'; }).length;
  add('INT-10','MEDIO','Líneas sin usuario responsable identificado',su,
    su?'Corresponden a la migración del formato v1.0.':'Toda línea tiene responsable.');
  var un=mae.filter(function(m){ return m.n===1; }).length;
  add('INT-11','BAJO','Productos con un solo dato de precio',un,
    un?'El precio no está validado contra una segunda compra.':'Todos validados.');

  add('INT-14','CERRADO','Líneas históricas sin precio, cerradas como no recuperables',cerradas,
    cerradas?('Recepciones de la v1.0 cuya factura no apareció. Reposan junto al resto en el mismo libro mayor; '+
              'no son un pendiente sino un hueco reconocido del histórico.')
            :'Sin huecos históricos.');

  var hp=ss_().getSheetByName(H.PRV), vistos={}, dups=0;
  if(hp && hp.getLastRow()>=6){
    hp.getRange(6,2,hp.getLastRow()-5,1).getValues().forEach(function(f){
      var n2=txt_(f[0]); if(!n2) return;
      if(vistos[n2]) dups++; else vistos[n2]=1;
    });
  }
  var dupc=0, seen={}, crudo=leerCatalogoCrudo_();
  for(var r=0;r<crudo.length;r++){
    var n3=txt_(crudo[r][C.NOMBRE-1]); if(!n3) continue;
    if(seen[n3]) dupc++; else seen[n3]=1;
  }
  add('INT-12','CRÍTICO','Filas duplicadas en PROVEEDORES o CATÁLOGO',dups+dupc,
    (dups+dupc) ? ('PROVEEDORES: '+dups+' · CATÁLOGO: '+dupc+'. Cada duplicado parte el agregado en dos. Usar ⚙ Di Égida → Reparar todo.')
                : 'Sin filas repetidas en las hojas maestras.');

  var orf={}, nOrf=0, lineasOrf=0;
  act.forEach(function(f){
    var p=txt_(f[B.PROV-1]);
    if(p && !vistos[p]){ if(!orf[p]){ orf[p]=0; nOrf++; } orf[p]++; lineasOrf++; }
  });
  add('INT-13','CRÍTICO','Proveedores del libro mayor sin fila en PROVEEDORES',nOrf,
    nOrf ? (lineasOrf+' líneas no suman en ningún agregado: '+Object.keys(orf).join(', ').substring(0,150)+
            '. Pasa al renombrar en la hoja maestra sin renombrar el histórico. Usar ⚙ Di Égida → Reparar todo.')
         : 'Todo el histórico está enlazado a un proveedor vigente.');

  // INT-15 y INT-16 nacen con la v3.4: el catálogo ahora lo alimenta cocina.
  var np=(pares||[]).length;
  add('INT-15','MEDIO','Pares de posible producto duplicado sin decidir',np,
    np ? 'Hoja DUPLICADOS: elegir en la columna DECISIÓN y marcar la casilla «Aplicar».'
       : 'Sin pares pendientes: el catálogo no tiene nombres que se pisen.');
  var inact=0, kc=Object.keys(cat);
  for(var q=0;q<kc.length;q++) if(cat[kc[q]].vig===INACTIVO) inact++;
  add('INT-16','INFO','Productos dados de baja (conservan su historial)',inact,
    inact ? 'No aparecen en RECEPCIÓN. Se reactivan desde la hoja PRODUCTOS.' : 'Ninguno.');

  var out=D.map(function(d){ return d.concat([new Date()]); });
  escribirBloque_(H.DIAG, 6, 6, out);
  return D;
}

function recalcularTablero_(cat, act, mae, leg, diag){
  var s=hoja_(H.TAB);
  var nCat=0, conP=0;
  for(var i=0;i<mae.length;i++){
    if(mae[i].vig===INACTIVO) continue;
    nCat++; if(mae[i].n>0) conP++;
  }
  var sinP=nCat-conP;
  var lineas=act.length;
  var lineasSinP=act.filter(function(f){ return !num_(f[B.PU-1]); }).length;
  var crit=mae.filter(function(m){return m.estado==='🔴';}).length;
  var alta=mae.filter(function(m){return m.estado==='🟠';}).length;
  var unico=mae.filter(function(m){return m.estado==='🔵';}).length;
  var pend=leg.filter(function(l){return l.estado.indexOf('🟠')===0;}).length;
  var venc=leg.filter(function(l){return l.estado.indexOf('🔴')===0;}).length;
  var merma=0; act.forEach(function(f){ merma += num_(f[B.MERMA-1])||0; });
  s.getRange(5,3,6,1).setValues([[nCat],[conP],[sinP],[nCat?conP/nCat:0],[lineas],[lineasSinP]]);
  s.getRange(5,7,6,1).setValues([[crit],[alta],[unico],[pend],[venc],[merma]]);
  // A2 es el ancla de la celda combinada A2:G2. Escribir en C2 (como hacía la v3.3) no se veía.
  s.getRange('A2').setValue('FMT-REC-COC-003 '+VERSION+'  |  Actualizado automáticamente: '+fmt_(new Date()));

  // Bloque de hallazgos. Causa del fallo «índice de nombres» desde el 09/09: la v3.3
  // combinaba B21:F21 DENTRO de la combinación A21:G21 que ya existía. Sheets acumula
  // esa escritura y la hace estallar en la siguiente lectura, en otro bloque.
  var fila=21;
  try{ s.getRange(fila,1,8,7).breakApart(); }catch(e){}
  s.getRange(fila,1,8,7).clearContent();
  s.getRange(fila,2,1,6).merge();
  var t=s.getRange(fila,2);
  var vivos=diag.filter(function(d){ return (d[1]==='CRÍTICO'||d[1]==='ALTO') && d[3]>0; });
  if(!vivos.length){
    t.setValue('✅  Integridad de datos: sin hallazgos críticos ni altos.')
     .setFontColor('#1A6B5C').setBackground('#E2EFDA');
  } else {
    t.setValue('⚠  Integridad de datos — '+vivos.length+' hallazgo(s) que requieren acción:')
     .setFontColor('#CC0000').setBackground('#F4CCCC');
    for(var q=0;q<Math.min(vivos.length,6);q++){
      s.getRange(fila+1+q,2).setValue(vivos[q][0]+' · '+vivos[q][2]).setFontSize(9);
      s.getRange(fila+1+q,3).setValue(vivos[q][3]).setFontWeight('bold')
       .setHorizontalAlignment('center').setFontColor('#CC0000');
    }
  }
  t.setFontWeight('bold').setFontSize(10);
}

/** Escribe una matriz en una hoja limpiando primero el sobrante. */
function escribirBloque_(nombre, fila, nCols, datos){
  var s=hoja_(nombre);
  var libres = s.getMaxRows()-fila+1;
  if(datos.length > libres) s.insertRowsAfter(s.getMaxRows(), datos.length-libres+100);
  var usado = Math.max(0, s.getLastRow()-fila+1);
  if(usado>0) s.getRange(fila,1,usado,nCols).clearContent();
  if(datos.length) s.getRange(fila,1,datos.length,nCols).setValues(datos);
}

function reversarRecepcion(){
  var ui;
  try{ ui = SpreadsheetApp.getUi(); }catch(x){
    throw new Error('reversarRecepcion debe ejecutarse desde un computador, no desde el iPad.');
  }
  var r = ui.prompt('Reversar recepción','Escriba el ID de la recepción a reversar (por ejemplo REC-COC-20260909-01):',ui.ButtonSet.OK_CANCEL);
  if(r.getSelectedButton()!==ui.Button.OK) return;
  var id = txt_(r.getResponse());
  if(!id) return;
  var m = ui.prompt('Motivo de la reversa','Explique por qué se reversa. Queda firmado en la BITÁCORA:',ui.ButtonSet.OK_CANCEL);
  if(m.getSelectedButton()!==ui.Button.OK) return;
  var motivo = txt_(m.getResponse());
  if(motivo.length < config_().minMotivo){ ui.alert('El motivo es obligatorio y debe ser explícito.'); return; }

  var lock=LockService.getDocumentLock(); lock.tryLock(25000);
  try{
    var s=hoja_(H.BD), n=s.getLastRow()-BD_F_INI+1;
    if(n<1){ ui.alert('El libro mayor está vacío.'); return; }
    var rng=s.getRange(BD_F_INI,1,n,B.N), v=rng.getValues(), c=0;
    for(var i=0;i<v.length;i++){
      if(txt_(v[i][B.ID_REC-1])===id && txt_(v[i][B.ESTADO-1])==='ACTIVA'){
        v[i][B.ESTADO-1]='REVERSADA';
        v[i][B.REF_REV-1]='Reversada '+fmt_(new Date())+' por '+usuario_()+' — '+motivo;
        c++;
      }
    }
    if(!c){ ui.alert('No se encontró ninguna línea activa con el ID '+id+'.'); return; }
    rng.setValues(v);
    SpreadsheetApp.flush();
    recalcularTodo();
    bitacora_('REVERSAR','REVERSADA',id,c,motivo,'Se reversaron '+c+' línea(s).');
    ui.alert('Listo. Se reversaron '+c+' línea(s) de '+id+'. El histórico se conserva: las líneas quedan marcadas, no borradas.');
  } finally { lock.releaseLock(); }
}

// ===========================================================================
// 06b · PRODUCTOS — alta, baja y reactivación desde el iPad
// ---------------------------------------------------------------------------
//  Reglas:
//   · El nombre se elige con cuidado porque es la clave: el motor compara con
//     TODO el catálogo (activos e inactivos) antes de crear.
//   · Igual al de uno existente (salvo tildes, mayúsculas o conectores) → no se
//     crea nunca. Si el existente está de baja, se ofrece reactivarlo.
//   · Parecido (es el mismo nombre más una marca, o difiere en una letra) → no
//     se crea hasta que alguien escriba en la NOTA por qué es otro producto.
//   · Una baja con compras deja el producto INACTIVO: sale de la lista de
//     RECEPCIÓN y conserva su historial de precios. Sin compras, se borra.
//   · Todo movimiento queda en la BITÁCORA con el responsable que se eligió.
// ===========================================================================
function msgProductos_(a1, texto, tipo){
  var h=hoja_(H.PRD), c=h.getRange(a1);
  c.setValue(texto);
  var col = tipo==='ok' ? '#1A6B5C' : (tipo==='warn' ? '#9C6500' : '#CC0000');
  var fon = tipo==='ok' ? '#E2EFDA' : (tipo==='warn' ? '#FFF2CC' : '#F4CCCC');
  c.setBackground(fon).setFontColor(col).setFontWeight('bold');
  try{ ss_().toast(String(texto).substring(0,240), tipo==='ok'?'Listo':'Revisar', tipo==='ok'?6:12); }catch(x){}
}

/** Nombre limpio: sin espacios dobles y con la primera letra en mayúscula. */
function nombreLimpio_(s){
  var t=txt_(s).replace(/\s+/g,' ');
  return t ? t.charAt(0).toUpperCase()+t.substring(1) : '';
}

/** Temperatura máxima por defecto cuando cocina no la escribe. */
function tmaxDefecto_(cons, cat){
  if(cons==='Congelado') return -12;
  if(cons!=='Refrigerado') return '';
  if(cat==='PESCA Y MARISCOS') return 2;
  if(cat==='LÁCTEOS') return 6;
  return 4;
}

function detalleCat_(resp, prod, resto){
  return 'Responsable: '+resp+' | Producto: '+prod+(resto?(' | '+resto):'');
}

function crearProducto_(){
  var h=hoja_(H.PRD), cfg=config_();
  var resp=txt_(h.getRange(P.A_RESP).getValue());
  var nom =nombreLimpio_(h.getRange(P.A_NOMBRE).getValue());
  var base=txt_(h.getRange(P.A_BASE).getValue());
  var cat =txt_(h.getRange(P.A_CAT).getValue()).toUpperCase();
  var cons=txt_(h.getRange(P.A_CONS).getValue());
  var tmx =h.getRange(P.A_TMAX).getValue();
  var nota=txt_(h.getRange(P.A_NOTA).getValue());

  // --- datos completos (una validación de celda no detiene un pegado) ---
  var faltan=[];
  if(!enLista_(resp, cfg.responsables)) faltan.push('Responsable: elija quién crea el producto ('+cfg.responsables.join(', ')+').');
  if(nom.length<3) faltan.push('Nombre: escriba el nombre del producto (mínimo 3 letras).');
  if(!enLista_(base, BASES)) faltan.push('Unidad base: elija g (se pesa), ml (líquido) o und (se cuenta).');
  if(!enLista_(cat, CATEGORIAS)) faltan.push('Categoría: elija una de la lista.');
  if(!enLista_(cons, CONSERVACION)) faltan.push('Conservación: elija Ambiente, Refrigerado o Congelado.');
  var tmax='';
  if(cons && cons!=='Ambiente'){
    if(txt_(tmx)===''){ tmax=tmaxDefecto_(cons, cat); }
    else {
      tmax=num_(tmx);
      if(tmax===null || tmax<-30 || tmax>20) faltan.push('Temp. máx.: debe ser un número entre -30 y 20 °C, o dejarse vacía.');
    }
  }
  if(faltan.length){
    msgProductos_(P.A_MSG,'⛔ NO SE CREÓ. Falta:\n• '+faltan.join('\n• '),'err');
    return {ok:false, motivo:'datos'};
  }

  // --- identidad: ¿ya existe o se parece a algo? ---
  var catal=catalogo_(), nombres=Object.keys(catal), parecidos=[];
  for(var i=0;i<nombres.length;i++){
    var ex=catal[nombres[i]], r=esDuplicado_(nom, ex.nombre);
    if(!r) continue;
    if(r.tipo==='IGUAL'){
      var txtI = ex.vig===INACTIVO
        ? ('⛔ NO SE CREÓ: ya existe «'+ex.nombre+'» y está DADO DE BAJA. Si se volvió a comprar, '+
           'reactívelo en el bloque 2 (Acción: Reactivar). Así conserva su historial de precios.')
        : ('⛔ NO SE CREÓ: ya existe «'+ex.nombre+'» ('+ex.cat+' · '+ex.base+'). Úselo en RECEPCIÓN tal como está.');
      msgProductos_(P.A_MSG, txtI, 'err');
      bitacora_('CATÁLOGO','ALTA RECHAZADA: YA EXISTE','',0,'',detalleCat_(resp, nom, 'Existe como «'+ex.nombre+'»'));
      return {ok:false, motivo:'igual', existente:ex.nombre};
    }
    parecidos.push({ex:ex, r:r});
  }
  if(parecidos.length && nota.length<cfg.minMotivo){
    parecidos.sort(function(a,b){ return b.r.pct-a.r.pct; });
    var lis=[];
    for(var k=0;k<Math.min(5,parecidos.length);k++){
      var e2=parecidos[k].ex;
      lis.push('«'+e2.nombre+'» — '+e2.cat+' · '+e2.base+(e2.vig===INACTIVO?' · DADO DE BAJA':'')+
               ' ('+parecidos[k].r.porque+')');
    }
    msgProductos_(P.A_MSG,
      '⚠ NO SE CREÓ TODAVÍA. Ya hay '+parecidos.length+' producto(s) parecido(s):\n• '+lis.join('\n• ')+
      (parecidos.length>5?('\n• …y '+(parecidos.length-5)+' más'):'')+
      '\n\nSi es el mismo producto, NO lo cree: use el existente en RECEPCIÓN.'+
      '\nSi es OTRO producto (otra marca con precio propio, otra presentación), escriba en NOTA por qué es distinto '+
      '(mínimo '+cfg.minMotivo+' caracteres) y vuelva a marcar la casilla.','warn');
    return {ok:false, motivo:'parecido', parecidos:parecidos.length};
  }

  // --- alta ---
  var s=hoja_(H.CAT), f=Math.max(s.getLastRow()+1, CAT_F_INI);
  if(f>s.getMaxRows()) s.insertRowsAfter(s.getMaxRows(), 50);
  var mov='Alta '+fmt_(new Date(),'dd/MM/yyyy')+' · '+resp+(nota?(' · '+nota):'');
  s.getRange(f,1,1,C.N).setValues([[f-CAT_F_INI+1, nom, base, cat, cons, tmax, 0, '', ACTIVO, mov]]);
  ordenarCatalogo_();
  // Si se creó a pesar de los parecidos, esos pares quedan declarados distintos:
  // no tienen por qué volver a aparecer en DUPLICADOS.
  for(var p=0;p<parecidos.length;p++) declararDistintos_(nom, parecidos[p].ex.nombre);
  SpreadsheetApp.flush();
  var cat2=catalogo_();
  refrescarListas_(cat2);
  try{ refrescarDuplicados_(cat2, null); }catch(eD){}
  bitacora_('CATÁLOGO','PRODUCTO CREADO','',1, nota,
    detalleCat_(resp, nom, base+' · '+cat+' · '+cons+(tmax!==''?(' · máx '+tmax+' °C'):'')+
      (parecidos.length?(' · creado pese a '+parecidos.length+' parecido(s)'):'')));
  refrescarHistorialProductos_();

  // limpiar el bloque, conservando el responsable para el siguiente
  var limpiar=[P.A_NOMBRE,P.A_BASE,P.A_CAT,P.A_CONS,P.A_TMAX,P.A_NOTA];
  for(var j=0;j<limpiar.length;j++) h.getRange(limpiar[j]).clearContent();
  msgProductos_(P.A_MSG,'✅ «'+nom+'» creado ('+cat+' · '+base+' · '+cons+
    (tmax!==''?(' · máx '+tmax+' °C'+(txt_(tmx)===''?' por defecto':'')):'')+
    '). Ya aparece en la lista de productos de RECEPCIÓN.','ok');
  return {ok:true, nombre:nom};
}

function bajaProductoHoja(){
  var lock=LockService.getDocumentLock();
  if(!lock.tryLock(20000)){ msgProductos_(P.B_MSG,'⏳ Hay otra operación en curso. Espere unos segundos y vuelva a marcar.','warn');
    hoja_(H.PRD).getRange(P.B_CHECK).setValue(false); return; }
  try{
    var h=hoja_(H.PRD), cfg=config_();
    var resp=txt_(h.getRange(P.B_RESP).getValue());
    var prod=txt_(h.getRange(P.B_PROD).getValue());
    var acc =txt_(h.getRange(P.B_ACCION).getValue());
    var mot =txt_(h.getRange(P.B_MOTIVO).getValue());
    var faltan=[];
    if(!enLista_(resp, cfg.responsables)) faltan.push('Responsable: elija quién hace el cambio.');
    if(!prod) faltan.push('Producto: elíjalo de la lista.');
    if(acc!==ACCION_BAJA && acc!==ACCION_REACT) faltan.push('Acción: elija «'+ACCION_BAJA+'» o «'+ACCION_REACT+'».');
    if(acc===ACCION_BAJA && mot.length<cfg.minMotivo) faltan.push('Motivo: explique por qué se da de baja (mínimo '+cfg.minMotivo+' caracteres).');
    if(faltan.length){ msgProductos_(P.B_MSG,'⛔ NO SE HIZO NINGÚN CAMBIO. Falta:\n• '+faltan.join('\n• '),'err'); return {ok:false}; }

    var r = (acc===ACCION_BAJA) ? darDeBaja_(prod, resp, mot) : reactivar_(prod, resp, mot);
    msgProductos_(P.B_MSG, r.msg, r.ok?'ok':(r.tipo||'err'));
    if(r.ok){
      h.getRange(P.B_PROD).clearContent(); h.getRange(P.B_MOTIVO).clearContent();
      refrescarHistorialProductos_();
    }
    return r;
  } finally {
    try{ hoja_(H.PRD).getRange(P.B_CHECK).setValue(false); }catch(x){}
    lock.releaseLock();
  }
}

/** Cuántas líneas del libro mayor (de cualquier estado) usan ese producto. */
function comprasDe_(prod, bd){
  var n=0; bd=bd||leerBD_();
  for(var i=0;i<bd.length;i++) if(txt_(bd[i][B.PRODUCTO-1])===prod) n++;
  return n;
}

/** Núcleo de la baja. Lo usan la hoja PRODUCTOS y la hoja DUPLICADOS. */
function darDeBaja_(prod, resp, motivo, bd){
  var cat=catalogo_(), c=cat[prod];
  if(!c) return {ok:false, msg:'⛔ «'+prod+'» no está en el CATÁLOGO.'};
  if(c.vig===INACTIVO) return {ok:false, tipo:'warn', msg:'⚠ «'+prod+'» ya estaba dado de baja ('+c.mov+'). No se cambió nada.'};
  var s=hoja_(H.CAT), n=comprasDe_(prod, bd), hoyTxt=fmt_(new Date(),'dd/MM/yyyy');
  var enCurso=filasEnRecepcion_(prod);
  var msg;
  if(n===0){
    s.deleteRow(c.fila);
    ordenarCatalogo_();
    msg='✅ «'+prod+'» eliminado del catálogo: nunca tuvo compras, así que no había historia que conservar.';
    bitacora_('CATÁLOGO','PRODUCTO ELIMINADO','',0,motivo, detalleCat_(resp, prod, 'sin compras registradas'));
  } else {
    s.getRange(c.fila, C.VIG, 1, 2).setValues([[INACTIVO, 'Baja '+hoyTxt+' · '+resp+' · '+motivo]]);
    msg='✅ «'+prod+'» dado de baja. Ya no aparece en RECEPCIÓN y conserva sus '+n+' compra(s) en el historial. '+
        'Se puede reactivar desde este mismo bloque.';
    bitacora_('CATÁLOGO','PRODUCTO DADO DE BAJA','',n,motivo, detalleCat_(resp, prod, n+' compra(s) conservadas'));
  }
  if(enCurso.length) msg+='\n⚠ Está escrito en la recepción en curso (fila '+enCurso.join(', ')+'): cámbielo antes de guardar.';
  SpreadsheetApp.flush();
  var cat2=catalogo_();
  refrescarListas_(cat2);
  try{ refrescarDuplicados_(cat2, null); }catch(e){}
  return {ok:true, msg:msg, eliminado:n===0};
}

function reactivar_(prod, resp, motivo){
  var cat=catalogo_(), c=cat[prod];
  if(!c) return {ok:false, msg:'⛔ «'+prod+'» no está en el CATÁLOGO.'};
  if(c.vig===ACTIVO) return {ok:false, tipo:'warn', msg:'⚠ «'+prod+'» ya está activo. No se cambió nada.'};
  hoja_(H.CAT).getRange(c.fila, C.VIG, 1, 2).setValues([[ACTIVO,
    'Reactivado '+fmt_(new Date(),'dd/MM/yyyy')+' · '+resp+(motivo?(' · '+motivo):'')]]);

// 06c · DETECCIÓN DE DUPLICADOS — que de tres avisos, tres sean reales
// ---------------------------------------------------------------------------
//  El porcentaje solo no decide: «Frambuesa / Frambuesa Berrygood» y
//  «Pimentón / Pimentón rojo» se parecen igual, y solo el primero es el mismo
//  producto con la marca pegada. Lo que los separa es la FORMA del nombre:
//   · Si una palabra exclusiva es una VARIANTE (fresco, seco, rojo, entero…)
//     o lleva números (15x16, 41/50), son productos distintos. Caso cerrado.
//   · Si un nombre es el otro más un detalle → candidato (marca pegada).
//   · Si difieren en una sola palabra mal escrita → candidato (Tomete/Tomate).
//   · Si comparten casi todas las palabras → candidato.
// ===========================================================================
var VARIANTES_ = {};
(function(){
  var l=['fresco','fresca','seco','seca','molido','molida','entero','entera','verde','rojo','roja',
    'blanco','blanca','negro','negra','amarillo','amarilla','morado','morada','cocido','cocida','crudo','cruda',
    'con','sin','relleno','rellena','baby','mix','deshidratado','deshidratada','rallado','rallada',
    'ahumado','ahumada','congelado','congelada','picado','picada','liquido','liquida','polvo','integral',
    'dulce','salado','salada','light','organico','organica','tinto','rosado','rosada','liso','lisa',
    'crespo','crespa','hidroponico','hidroponica','premium','nacional','importado','importada',
    'grande','pequeno','pequena','mini','extra','deslactosado','deslactosada','descremado','descremada',
    'tajado','tajada','rebanado','rebanada','cuna','bloque','pechuga','pierna','muslo','filete','lomo',
    'costilla','hueso','molienda','vino','harina','aceite','vinagre','leche','crema','salsa','pasta','jugo',
    'pulpa','esencia','extracto','semilla','hoja','hojas','rama','flor','flores','enlatado','lata'];
  for(var i=0;i<l.length;i++) VARIANTES_[l[i]]=true;
})();
var STOP_DUP_ = {'de':1,'del':1,'la':1,'el':1,'los':1,'las':1,'y':1,'en':1,'x':1,'a':1,'al':1,
                 'para':1,'por':1,'tipo':1,'brand':1,'sas':1};

/** Palabras comparables: sin tildes, sin conectores y en singular. */
function fichasDup_(s){
  var t=clave_(s).split(' '), out=[], vis={};
  for(var i=0;i<t.length;i++){
    var w=t[i];
    if(!w || STOP_DUP_[w]) continue;
    if(!/[0-9]/.test(w) && w.length>3){
      if(/[lnrdz]es$/.test(w) && w.length>4) w=w.substring(0,w.length-2);
      else if(/s$/.test(w)) w=w.substring(0,w.length-1);
    }
    if(!vis[w]){ vis[w]=1; out.push(w); }
  }
  return out;
}
function trigramas_(s){
  var t='  '+s+' ', m={}, n=0;
  for(var i=0;i<t.length-2;i++){ var g=t.substring(i,i+3); m[g]=(m[g]||0)+1; n++; }
  return {m:m, n:n};
}
/** Similitud de Dice por trigramas: 1 = idénticas. Atrapa errores de escritura. */
function dice_(a,b){
  if(a===b) return 1;
  var A=trigramas_(a), Bt=trigramas_(b), comun=0, k=Object.keys(A.m);
  for(var i=0;i<k.length;i++) if(Bt.m[k[i]]) comun+=Math.min(A.m[k[i]], Bt.m[k[i]]);
  return (2*comun)/(A.n+Bt.n);
}
function esDiscriminante_(w){ return !!VARIANTES_[w] || /[0-9]/.test(w); }

/**
 * ¿Son el mismo producto escrito distinto? Devuelve null si no, o
 * {tipo:'IGUAL'|'CONTENIDO'|'ESCRITURA'|'PARECIDO', pct, porque}.
 */
function esDuplicado_(a, b, fa, fb, ka, kb){
  if((ka||clave_(a))===(kb||clave_(b))) return {tipo:'IGUAL', pct:1, porque:'mismo nombre'};
  fa=fa||fichasDup_(a); fb=fb||fichasDup_(b);
  if(!fa.length || !fb.length) return null;
  var sa={}, sb={}, i;
  for(i=0;i<fa.length;i++) sa[fa[i]]=1;
  for(i=0;i<fb.length;i++) sb[fb[i]]=1;
  var exA=[], exB=[], inter=0;
  for(i=0;i<fa.length;i++){ if(sb[fa[i]]) inter++; else exA.push(fa[i]); }
  for(i=0;i<fb.length;i++){ if(!sa[fb[i]]) exB.push(fb[i]); }
  if(!exA.length && !exB.length) return {tipo:'IGUAL', pct:1, porque:'mismas palabras'};
  var union=inter+exA.length+exB.length, jac=inter/union;

  // una sola palabra distinta a cada lado y casi igual escrita → error de digitación
  if(exA.length===1 && exB.length===1 && !/[0-9]/.test(exA[0]+exB[0]) &&
     !(esDiscriminante_(exA[0]) && esDiscriminante_(exB[0]))){
    var d=dice_(exA[0], exB[0]);
    if(d>=0.6 && Math.min(exA[0].length, exB[0].length)>=4)
      return {tipo:'ESCRITURA', pct:d, porque:'«'+exA[0]+'» y «'+exB[0]+'» parecen la misma palabra mal escrita'};
  }
  if(!inter) return null;
  // una variante o un número exclusivo cierra el caso: son productos distintos
  var ex=exA.concat(exB);
  for(i=0;i<ex.length;i++) if(esDiscriminante_(ex[i])) return null;

  if((!exA.length || !exB.length) && jac>=0.5){
    var extra=(exA.length?exA:exB).join(' ');
    return {tipo:'CONTENIDO', pct:jac, porque:'uno es el otro más «'+extra+'»: ¿marca o presentación del mismo producto?'};
  }
  if(jac>=0.8) return {tipo:'PARECIDO', pct:jac, porque:'comparten '+Math.round(jac*100)+'% de las palabras'};
  return null;
}

/**
 * Pares sospechosos entre productos ACTIVOS, sin los que ya se declararon
 * distintos. Reescribe la hoja DUPLICADOS conservando las decisiones a medio
 * marcar. Devuelve la lista de pares pendientes.
 */
function refrescarDuplicados_(cat, act){
  cat=cat||catalogo_();
  var usos={}, bd=act||leerBD_();
  for(var u=0;u<bd.length;u++){ var p0=txt_(bd[u][B.PRODUCTO-1]); if(p0) usos[p0]=(usos[p0]||0)+1; }
  var activos=[], k=Object.keys(cat);
  for(var i=0;i<k.length;i++) if(cat[k[i]].vig===ACTIVO) activos.push(cat[k[i]]);
  activos.sort(function(a,b){ var x=clave_(a.nombre), y=clave_(b.nombre); return x<y?-1:(x>y?1:0); });
  var fich=[], claves=[];
  for(i=0;i<activos.length;i++){ fich.push(fichasDup_(activos[i].nombre)); claves.push(clave_(activos[i].nombre)); }
  var distintos=paresDistintos_(), pares=[];
  for(i=0;i<activos.length;i++){
    for(var j=i+1;j<activos.length;j++){
      var r=esDuplicado_(activos[i].nombre, activos[j].nombre, fich[i], fich[j], claves[i], claves[j]);
      if(!r) continue;
      if(distintos[clavePar_(activos[i].nombre, activos[j].nombre)]) continue;
      pares.push({a:activos[i], b:activos[j], r:r});
    }
  }
  var h=ss_().getSheetByName(H.DUP);
  if(!h) return pares;

  // decisiones ya escritas y no aplicadas: se conservan por par
  var prev={}, nPrev=h.getLastRow()-DUP_F_INI+1;
  if(nPrev>0){
    var pv=h.getRange(DUP_F_INI,1,nPrev,8).getValues();
    for(var q=0;q<pv.length;q++){
      var dec=txt_(pv[q][7]);
      if(dec && txt_(pv[q][1]) && txt_(pv[q][3])) prev[clavePar_(txt_(pv[q][1]), txt_(pv[q][3]))]=dec;
    }
  }
  var out=[];
  for(var m=0;m<pares.length;m++){
    var pa=pares[m];
    out.push([m+1, pa.a.nombre, usos[pa.a.nombre]||0, pa.b.nombre, usos[pa.b.nombre]||0,
      pa.a.base+' / '+pa.b.base+(pa.a.base!==pa.b.base?'  ⚠ unidad distinta':'')+'  ·  '+pa.a.cat+(pa.a.cat!==pa.b.cat?(' / '+pa.b.cat):''),
      pa.r.porque, prev[clavePar_(pa.a.nombre, pa.b.nombre)]||'']);
  }
  var libres=h.getMaxRows()-DUP_F_INI+1;
  if(out.length>libres) h.insertRowsAfter(h.getMaxRows(), out.length-libres+50);
  if(nPrev>0) h.getRange(DUP_F_INI,1,nPrev,8).clearContent();
  if(out.length) h.getRange(DUP_F_INI,1,out.length,8).setValues(out);
  try{ validacionDecisiones_(h, Math.max(out.length, 1)); }catch(e){}
  avisoDuplicados_(h, pares.length, '');
  return pares;
}

function avisoDuplicados_(h, n, prefijo){
  var t = (prefijo?prefijo+'  ':'') + (n
    ? ('⚠ '+n+' par(es) por decidir. Lea la columna «Por qué se marcó», elija la DECISIÓN, el responsable arriba, y marque la casilla «Aplicar».')
    : '✅ Sin pares pendientes: ningún nombre activo del catálogo pisa a otro.');
  var c=h.getRange(DUPC.AVISO);
  c.setValue(t).setFontWeight('bold').setWrap(true)
   .setBackground(n?'#FFF2CC':'#E2EFDA').setFontColor(n?'#9C6500':'#1A6B5C');
}

function validacionDecisiones_(h, n){
  h.getRange(DUP_F_INI,8,Math.max(n,1),1).setDataValidation(
    SpreadsheetApp.newDataValidation().requireValueInList([DEC_DISTINTOS, DEC_QUEDA_A, DEC_QUEDA_B], true)
      .setAllowInvalid(false).build());
}

/** Casilla «Aplicar» de la hoja DUPLICADOS. */
function aplicarDecisionesDuplicados(){
  var lock=LockService.getDocumentLock();
  var h=hoja_(H.DUP);
  if(!lock.tryLock(20000)){ h.getRange(DUPC.CHECK).setValue(false); return; }
  try{
    var cfg=config_(), resp=txt_(h.getRange(DUPC.RESP).getValue());
    if(!enLista_(resp, cfg.responsables)){
      avisoDuplicados_(h, Math.max(0,h.getLastRow()-DUP_F_INI+1), '⛔ No se aplicó nada: elija el responsable en '+DUPC.RESP+'.');
      return {ok:false};
    }
    var n=h.getLastRow()-DUP_F_INI+1;
    if(n<1){ avisoDuplicados_(h,0,''); return {ok:true, hechas:0}; }
    var v=h.getRange(DUP_F_INI,1,n,8).getValues(), bd=leerBD_();
    var hechas=0, bajas=0, distintos=0, saltadas=[], yaDeBaja={};
    for(var i=0;i<v.length;i++){
      var a=txt_(v[i][1]), b=txt_(v[i][3]), dec=txt_(v[i][7]);
      if(!dec || !a || !b) continue;
      if(dec===DEC_DISTINTOS){
        declararDistintos_(a,b);
        bitacora_('CATÁLOGO','PAR DECLARADO DISTINTO','',0,'', detalleCat_(resp, a+' ≠ '+b, 'hoja DUPLICADOS'));
        distintos++; hechas++; continue;
      }
      var queda = dec===DEC_QUEDA_A ? a : b, sale = dec===DEC_QUEDA_A ? b : a;
      if(yaDeBaja[queda]){ saltadas.push('«'+queda+'» ya se dio de baja en otra fila'); continue; }
      if(yaDeBaja[sale]){ hechas++; continue; }
      var r=darDeBaja_(sale, resp, 'Duplicado de «'+queda+'» (hoja DUPLICADOS)', bd);
      if(r.ok){ yaDeBaja[sale]=true; bajas++; hechas++; }
      else saltadas.push(r.msg);
    }
    SpreadsheetApp.flush();
    var cat=catalogo_();
    var pares=refrescarDuplicados_(cat, bd);
    refrescarListas_(cat);
    refrescarHistorialProductos_();
    avisoDuplicados_(h, pares.length, '✅ Aplicadas '+hechas+' decisión(es): '+bajas+' baja(s), '+distintos+' par(es) declarados distintos.'+
      (saltadas.length?(' ⚠ Sin aplicar: '+saltadas.join(' · ')):''));
    return {ok:true, hechas:hechas, bajas:bajas, distintos:distintos, pendientes:pares.length};
  } finally {
    try{ h.getRange(DUPC.CHECK).setValue(false); }catch(x){}
    lock.releaseLock();
  }
}

    b.push('<table style="width:100%;border-collapse:collapse;font-size:12px">');
    criticos.concat(altos).forEach(function(d){
      var col = d[1]==='CRÍTICO' ? '#CC0000' : '#9C6500';
      b.push('<tr><td style="padding:5px;border-bottom:1px solid #ddd;color:'+col+';font-weight:bold;width:70px">'+d[0]+
             '</td><td style="padding:5px;border-bottom:1px solid #ddd">'+d[2]+'</td>'+
             '<td style="padding:5px;border-bottom:1px solid #ddd;text-align:center;font-weight:bold">'+d[3]+'</td></tr>');
    });
    b.push('</table>');
  }
  b.push('<p style="font-size:11px;color:#888;margin-top:18px;border-top:1px solid #ddd;padding-top:10px">'+
    'Generado automáticamente por el motor de recepción. El detalle completo está en las hojas LEGALIZACIÓN, DIAGNÓSTICO y BITÁCORA del archivo: '+
    '<a href="'+ss_().getUrl()+'">abrir el archivo</a>.</p></div>');

  var destinos=[cfg.correoAdmin, cfg.correoChef].filter(function(x){ return x && x.indexOf('@')>0; });
  if(!destinos.length){
    bitacora_('RUTINA DIARIA','SIN DESTINATARIOS','',0,'','Llenar CORREO_ADMIN y CORREO_CHEF en la hoja CONFIG.');
    return;
  }
  var partes=[];
  if(vencidas.length) partes.push(vencidas.length+' vencida(s)');
  if(pendientes.length) partes.push(pendientes.length+' por legalizar');
  if(criticos.length) partes.push(criticos.length+' alerta(s) de integridad');
  if(movs.length) partes.push(movs.length+' movimiento(s) de catálogo');
  var asunto='Di Égida · Recepción cocina — '+partes.join(', ');
  MailApp.sendEmail({to:destinos.join(','), subject:asunto, htmlBody:b.join('')});
  bitacora_('RUTINA DIARIA','CORREO ENVIADO','',0,'', asunto+' → '+destinos.join(', '));
}

/**
 * Las fórmulas de RECEPCIÓN buscaban el producto solo hasta la fila 355 del
 * CATÁLOGO: con 350 productos la unidad y la conservación dejaban de llegar.
 */
function formulasRecepcion_(){
  var r=hoja_(H.REC), out=[];
  for(var f=REC.F_INI; f<=REC.F_FIN; f++){
    var m='MATCH($B'+f+',CATALOGO!$B$6:$B$2000,0)';
    out.push({f:f,
      c:'=IFERROR(INDEX(CATALOGO!$C$6:$C$2000,'+m+'),"")',
      q:'=IFERROR(INDEX(CATALOGO!$E$6:$E$2000,'+m+'),"")',
      s:'=IF($B'+f+'="","",IF($Q'+f+'="Ambiente","N/A",IF($R'+f+'="","⚠ FALTA",IF(NOT(ISNUMBER($R'+f+')),"⚠ REVISAR",IF($R'+f+'<=IFERROR(INDEX(CATALOGO!$F$6:$F$2000,'+m+'),99),"✓ OK","✕ RECHAZAR")))))'});
  }
  var cC=[], cQ=[], cS=[];
  for(var i=0;i<out.length;i++){ cC.push([out[i].c]); cQ.push([out[i].q]); cS.push([out[i].s]); }
  var n=REC.F_FIN-REC.F_INI+1;
  r.getRange(REC.F_INI,R.BASE,n,1).setFormulas(cC);
  r.getRange(REC.F_INI,R.CONS,n,1).setFormulas(cQ);
  r.getRange(REC.F_INI,R.TEMP_OK,n,1).setFormulas(cS);
}

/** CATÁLOGO: columnas VIGENCIA y ÚLTIMO MOVIMIENTO. No toca ningún dato existente. */
function asegurarColumnasCatalogo_(){
  var s=hoja_(H.CAT);
  if(s.getMaxColumns()<C.N) s.insertColumnsAfter(s.getMaxColumns(), C.N-s.getMaxColumns());
  s.getRange(5,C.VIG,1,2).setValues([['Vigencia','Último movimiento\n(alta · baja · reactivación)']])
   .setFontWeight('bold').setFontColor('#FFFFFF').setBackground('#1A6B5C').setWrap(true)
   .setHorizontalAlignment('center').setVerticalAlignment('middle');
  var n=s.getLastRow()-CAT_F_INI+1, puestos=0;
  if(n>0){
    var v=s.getRange(CAT_F_INI,1,n,C.N).getValues(), vig=[];
    for(var i=0;i<n;i++){
      var actual=txt_(v[i][C.VIG-1]).toUpperCase();
      if(txt_(v[i][C.NOMBRE-1]) && actual!==ACTIVO && actual!==INACTIVO){ vig.push([ACTIVO]); puestos++; }
      else vig.push([txt_(v[i][C.NOMBRE-1]) ? (actual===INACTIVO?INACTIVO:ACTIVO) : '']);
    }
    s.getRange(CAT_F_INI,C.VIG,n,1).setValues(vig);
  }
  var nMax=s.getMaxRows()-CAT_F_INI+1;
  s.getRange(CAT_F_INI,C.VIG,nMax,1).setDataValidation(
    SpreadsheetApp.newDataValidation().requireValueInList([ACTIVO,INACTIVO],true).setAllowInvalid(false).build());
  try{ s.setColumnWidth(C.VIG, 90); s.setColumnWidth(C.MOV, 320); }catch(e){}
  // Sin combinar celdas: la columna A del CATÁLOGO está inmovilizada y Sheets no deja
  // combinar a través de ese borde. En la v3.4.0 este intento hacía fallar el paso de CONFIG.
  SpreadsheetApp.flush();
  return puestos;
}

/** Corre un paso con nombre, sin dejar que tumbe a los demás. */
function pasoSeguro_(lista, nombre, fn){
  // flush dentro del paso: Sheets acumula escrituras y un error de un paso salía
  // en el siguiente, con el nombre equivocado.
  try{ fn(); SpreadsheetApp.flush(); lista.ok.push(nombre); }
  catch(e){ lista.mal.push(nombre+' → '+e.message); }
}

/** Protege una hoja o un rango. canDomainEdit lanza excepción en cuentas Gmail personales. */
function protegerSeguro_(objetoProtegible, descripcion){
  var p=objetoProtegible.protect().setDescription(descripcion);
  try{ p.removeEditors(p.getEditors()); }catch(e){}
  try{ p.addEditor(Session.getEffectiveUser()); }catch(e){}
  try{ if(p.canDomainEdit()) p.setDomainEdit(false); }catch(e){}
  return p;
}

function refrescarAvisos_(cat, act, mae, leg, diag, pares){
  var fallos=[];
  function paso(nombre, fn){
    try{ fn(); SpreadsheetApp.flush(); }catch(e){ fallos.push(nombre+': '+e.message); }
  }
  var idx=null;
  paso('índice de nombres', function(){ idx=indiceNombres_(cat, act, pares); });
  if(idx){
    paso('avisos de catálogo',    function(){ avisosCatalogo_(cat, act, idx); });
    paso('avisos de proveedores', function(){ avisosProveedores_(act, leg); });
    paso('bandas superiores',     function(){ avisosBandas_(act, leg, mae, diag, idx, cat); });

function indiceNombres_(cat, act, pares){
  var porClave={}, usos={}, sinPrecio={}, fueraBanda={}, basesEnBD={}, parDe={};
  var bn=bandas_();
  act.forEach(function(f){
    var p=txt_(f[B.PRODUCTO-1]); if(!p) return;
    usos[p]=(usos[p]||0)+1;
    var pu=num_(f[B.PU-1]);
    if(!pu) sinPrecio[p]=(sinPrecio[p]||0)+1;
    var b=txt_(f[B.BASE-1]); if(b){ (basesEnBD[p]=basesEnBD[p]||{})[b]=1; }
    var c=cat[p];
    if(c && pu){
      var bb=bandaDe_(bn, c.cat, c.base);
      if(bb && (pu<bb.inf||pu>bb.sup)) fueraBanda[p]=(fueraBanda[p]||0)+1;
    }
  });
  Object.keys(cat).forEach(function(n){
    var k=clave_(n);
    (porClave[k]=porClave[k]||[]).push(n);
  });
  (pares||[]).forEach(function(p){
    (parDe[p.a.nombre]=parDe[p.a.nombre]||[]).push(p.b.nombre);
    (parDe[p.b.nombre]=parDe[p.b.nombre]||[]).push(p.a.nombre);
  });
  return {porClave:porClave, usos:usos, sinPrecio:sinPrecio,
          fueraBanda:fueraBanda, basesEnBD:basesEnBD, parDe:parDe};
}

/** Sin String.normalize: no existe en el motor antiguo de Apps Script. */
var ACENTOS_ = {'á':'a','à':'a','ä':'a','â':'a','ã':'a','å':'a','Á':'a','À':'a','Ä':'a','Â':'a','Ã':'a',
 'é':'e','è':'e','ë':'e','ê':'e','É':'e','È':'e','Ë':'e','Ê':'e',
 'í':'i','ì':'i','ï':'i','î':'i','Í':'i','Ì':'i','Ï':'i','Î':'i',
 'ó':'o','ò':'o','ö':'o','ô':'o','õ':'o','Ó':'o','Ò':'o','Ö':'o','Ô':'o','Õ':'o',
 'ú':'u','ù':'u','ü':'u','û':'u','Ú':'u','Ù':'u','Ü':'u','Û':'u',
 'ñ':'n','Ñ':'n','ç':'c','Ç':'c','º':' ','ª':' ','’':' ',"'":' '};
function sinTildes_(s){
  var t=String(s==null?'':s), r='';
  for(var i=0;i<t.length;i++){
    var c=t.charAt(i);
    r += (ACENTOS_[c]!==undefined) ? ACENTOS_[c] : c;
  }
  return r;
}
function clave_(s){
  return sinTildes_(s).toLowerCase().replace(/[^a-z0-9 ]+/g,' ').replace(/\s+/g,' ').replace(/^ +| +$/g,'');
}
function fichas_(s){
  var stop={'de':1,'del':1,'la':1,'el':1,'los':1,'las':1,'y':1,'en':1,'con':1,'sas':1,'brand':1};
  return clave_(s).split(' ').filter(function(t){
    if(stop[t]) return false;
    return t.length>2 || /^[0-9]+$/.test(t);
  });
}
function parecido_(a,b){
  var ta=fichas_(a), tb=fichas_(b);
  if(!ta.length||!tb.length) return 0;
  var set={}, union={};
  ta.forEach(function(t){ set[t]=1; union[t]=1; });
  var comunes=0;
  tb.forEach(function(t){ if(set[t]) comunes++; union[t]=1; });
  return comunes/Object.keys(union).length;
}

function normalizarNombres_(){
  var act=leerBD_(), cambios={prod:0, prov:0, lineas:0};
  function canonizar(col){
    var frec={};
    act.forEach(function(f){
      var v=txt_(f[col-1]); if(!v) return;
      var k=clave_(v);
      (frec[k]=frec[k]||{})[v]=(frec[k][v]||0)+1;
    });
    var mapa={};
    Object.keys(frec).forEach(function(k){
      var opciones=Object.keys(frec[k]);
      if(opciones.length<2) return;
      opciones.sort(function(a,b){
        if(frec[k][b]!==frec[k][a]) return frec[k][b]-frec[k][a];
        return a.length-b.length;
      });
      var bueno=opciones[0];
      opciones.slice(1).forEach(function(o){ mapa[o]=bueno; });
    });
    return mapa;
  }
  var mp=canonizar(B.PRODUCTO), mv=canonizar(B.PROV);
  var hb=hoja_(H.BD), n=hb.getLastRow()-BD_F_INI+1;
  if(n>0 && (Object.keys(mp).length || Object.keys(mv).length)){
    desproteger_(hb);
    var rg=hb.getRange(BD_F_INI,1,n,B.N), v=rg.getValues();
    for(var i=0;i<v.length;i++){
      if(!txt_(v[i][B.ID_LINEA-1])) continue;
      var t=false;
      var p=txt_(v[i][B.PRODUCTO-1]); if(mp[p]){ v[i][B.PRODUCTO-1]=mp[p]; cambios.prod++; t=true; }
      var q=txt_(v[i][B.PROV-1]);     if(mv[q]){ v[i][B.PROV-1]=mv[q];   cambios.prov++; t=true; }
      if(t) cambios.lineas++;
    }
    rg.setValues(v);
  }
  return cambios;
}

/** Reconstruye CATÁLOGO y PROVEEDORES desde el libro mayor, sin duplicados.

function reconstruirMaestras_(){

    meta[nom0]={base:txt_(crudo[r][C.BASE-1]), cat:txt_(crudo[r][C.CAT-1]), cons:txt_(crudo[r][C.CONS-1]),
                tmax:crudo[r][C.TMAX-1], vig:txt_(crudo[r][C.VIG-1]).toUpperCase()===INACTIVO?INACTIVO:ACTIVO,
                mov:txt_(crudo[r][C.MOV-1])};
  }
  var usos={}, baseBD={};
  act.forEach(function(f){
    var p=txt_(f[B.PRODUCTO-1]); if(!p) return;
    usos[p]=(usos[p]||0)+1;
    var b=txt_(f[B.BASE-1]); if(b) (baseBD[p]=baseBD[p]||{})[b]=(baseBD[p][b]||0)+1;

  var nombres=Object.keys(todos).sort(function(a,b){ return clave_(a)<clave_(b)?-1:(clave_(a)>clave_(b)?1:0); });

    var base=m.base;
    if(!base && baseBD[nom]){
      base=Object.keys(baseBD[nom]).sort(function(a,b){ return baseBD[nom][b]-baseBD[nom][a]; })[0];
    }
    return [i+1, nom, base||'g', m.cat||'POR CLASIFICAR', m.cons||'Ambiente', m.tmax===undefined?'':m.tmax,
            usos[nom]||0, '', m.vig, m.mov];
  });
  if(hc.getMaxColumns()<C.N) hc.insertColumnsAfter(hc.getMaxColumns(), C.N-hc.getMaxColumns());
  var libres=hc.getMaxRows()-CAT_F_INI+1;

  var hp=hoja_(H.PRV); desproteger_(hp);
  var nP=Math.max(0, hp.getLastRow()-6+1), cont={};

      var nom=txt_(f[0]); if(!nom) return;
      if(!cont[nom]) cont[nom]=[txt_(f[1]),txt_(f[2])];
      else { if(!cont[nom][0]) cont[nom][0]=txt_(f[1]); if(!cont[nom][1]) cont[nom][1]=txt_(f[2]); }
    });
  }
  var provs={}; Object.keys(cont).forEach(function(k){ provs[k]=1; });
  act.forEach(function(f){ var p=txt_(f[B.PROV-1]); if(p) provs[p]=1; });
  var lp=Object.keys(provs).sort(function(a,b){ return clave_(a)<clave_(b)?-1:(clave_(a)>clave_(b)?1:0); });
  var fp=lp.map(function(nom,i){
    var c=cont[nom]||['',''];
    return [i+1, nom, c[0], c[1], '', '', '', '', '', '', ''];
  });
  var libresP=hp.getMaxRows()-6+1;
  if(fp.length+2>libresP) hp.insertRowsAfter(hp.getMaxRows(), fp.length-libresP+50);
  if(nP>0){ try{ hp.getRange(6,1,nP+2,11).breakApart(); }catch(e){} hp.getRange(6,1,nP+2,11).clearContent(); }
  if(fp.length) hp.getRange(6,1,fp.length,11).setValues(fp);
  return {productos:filas.length, proveedores:fp.length};
}

function reporteParaAsesor(){
  var act=leerBD_().filter(function(f){ return txt_(f[B.ESTADO-1])==='ACTIVA'; });
  var res=recalcularTodo();
  var cfg=config_(), cat=catalogo_();
  var L=[];
  L.push('ESTADO DEL SISTEMA DE RECEPCIÓN '+VERSION+' — Di Égida');
  L.push('Generado: '+fmt_(new Date()));
  L.push('');
  var nRec={}; act.forEach(function(f){ nRec[txt_(f[B.ID_REC-1])]=1; });
  var conP=res.mae.filter(function(m){ return m.n>0; }).length;
  var valor=0; act.forEach(function(f){ valor += num_(f[B.VALOR-1])||0; });
  L.push('CIFRAS');
  L.push('  Líneas activas: '+act.length+' en '+Object.keys(nRec).length+' recepciones');
  L.push('  Productos: '+res.mae.length+' ('+Object.keys(activos_(cat)).length+' activos) · con precio: '+conP+' ('+Math.round(conP/Math.max(1,res.mae.length)*100)+'%)');
  L.push('  Valor total del libro mayor: $'+Math.round(valor).toLocaleString('es-CO'));
  L.push('');
  L.push('DIAGNÓSTICO (solo lo que está en rojo)');
  var vivos=res.diag.filter(function(d){ return d[3]>0 && d[1]!=='CERRADO' && d[1]!=='INFO'; });
  if(!vivos.length) L.push('  Sin hallazgos.');
  vivos.forEach(function(d){ L.push('  '+d[0]+' ['+d[1]+'] '+d[2]+': '+d[3]+' — '+String(d[4]).substring(0,140)); });
  L.push('');
  L.push('MOVIMIENTOS DE CATÁLOGO (7 días)');
  var mv=movimientosCatalogo_(24*7);
  L.push(mv.length ? mv.map(function(x){ return '  '+fmt_(x.t)+' · '+x.res+' · '+x.det; }).join('\n') : '  Ninguno.');
  L.push('');
  L.push('MARCAS EN CATÁLOGO');
  var hc=ss_().getSheetByName(H.CAT), marcas=[];
  if(hc && hc.getLastRow()>=CAT_F_INI){
    var vv=hc.getRange(CAT_F_INI,2,hc.getLastRow()-CAT_F_INI+1,7).getValues();
    vv.forEach(function(f){
      var e=txt_(f[6]);
      if(e.indexOf('⛔')>=0 || e.indexOf('⚠')>=0) marcas.push('  '+txt_(f[0])+' ['+txt_(f[1])+'] → '+e);
    });
  }
  L.push(marcas.length ? marcas.slice(0,40).join('\n') : '  Ninguna.');
  if(marcas.length>40) L.push('  … y '+(marcas.length-40)+' más.');
  L.push('');
  L.push('CONFIGURACIÓN');
  L.push('  Correo admin: '+(cfg.correoAdmin||'SIN DEFINIR')+' · Correo chef: '+(cfg.correoChef||'SIN DEFINIR'));
  L.push('  Responsables de catálogo: '+cfg.responsables.join(', '));
  L.push('  Umbrales: aviso '+(cfg.umbralAlerta*100)+'% · bloqueo '+(cfg.umbralBloqueo*100)+
         '% · banda categoría ×'+cfg.factorBanda+' · banda producto ×'+cfg.factorProducto);
  var txt=L.join('\n');
  try{
    var ui=SpreadsheetApp.getUi();
    var html=HtmlService.createHtmlOutput(
      '<p style="font:13px Arial">Copie este texto y péguelo en el chat. Reemplaza mandar el archivo completo.</p>'+
      '<textarea style="width:100%;height:340px;font:11px monospace" onclick="this.select()">'+
      txt.replace(/&/g,'&amp;').replace(/</g,'&lt;')+'</textarea>')
      .setWidth(700).setHeight(430);
    ui.showModalDialog(html,'Reporte para asesoría');
  }catch(x){ Logger.log(txt); }
  return txt;
}
