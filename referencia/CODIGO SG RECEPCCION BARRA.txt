function txt_(v){
  if (v === null || v === undefined) return '';
  var s = String(v);
  return s.replace(/^[\s ]+/, '').replace(/[\s ]+$/, '');
}

function num_(v){
  if (v === null || v === undefined || v === '') return '';
  if (typeof v === 'number') return isNaN(v) ? '' : v;
  var s = String(v).replace(/[$\s ]/g, '').replace(/\./g, '').replace(/,/g, '.');
  var n = parseFloat(s);
  return isNaN(n) ? '' : n;
}

function esNumero_(v){ return typeof v === 'number' && !isNaN(v); }

function sinTildes_(s){
  var t = String(s === null || s === undefined ? '' : s);
  var out = '';
  for (var i = 0; i < t.length; i++){
    var c = t.charAt(i);
    out += (MAPA_TILDES[c] !== undefined) ? MAPA_TILDES[c] : c;
  }
  return out;
}

/* Clave de comparación: sin tildes, minúsculas, sin signos, espacios colapsados */
function clave_(s){
  var t = sinTildes_(txt_(s)).toLowerCase();
  t = t.replace(/[^a-z0-9 ]/g, ' ');
  t = t.replace(/\s+/g, ' ');
  return txt_(t);
}

function empiezaCon_(cadena, prefijo){

/* ---------------------------------------------------------------------------------------
   MOTOR DE SIMILITUD  (v1.1)

   Tokens únicos de un nombre, ya sin tildes ni signos.
--------------------------------------------------------------------------------------- */
function tokens_(s){
  var partes = clave_(s).split(' '), out = [], visto = {}, i;
  for (i = 0; i < partes.length; i++){
    var p = partes[i];
    if (p === '' || visto[p]) continue;
    visto[p] = true;
    out.push(p);
  }
  return out;
}

function tieneDigito_(t){
  var i;
  for (i = 0; i < t.length; i++){
    var c = t.charAt(i);
    if (c >= '0' && c <= '9') return true;
  }
  return false;
}

function hayDiscriminante_(lista){
  var i;
  for (i = 0; i < lista.length; i++){
    if (TOKENS_DISCRIMINANTES[lista[i]]) return true;
    if (tieneDigito_(lista[i])) return true;   /* x300, x250, 750, «12 años» */
  }
  return false;
}

/* Jaccard = intersección / UNIÓN. Nunca sobre el mínimo:
   con el mínimo «Camarón 51/60» y «Camarón 36/40» darían 100%. */
function jaccard_(a, b){
  var A = tokens_(a), Bx = tokens_(b), i;
  if (A.length === 0 || Bx.length === 0) return 0;
  var setA = {}, setB = {};
  for (i = 0; i < A.length; i++) setA[A[i]] = true;
  for (i = 0; i < Bx.length; i++) setB[Bx[i]] = true;
  var inter = 0, exclB = 0;
  for (i = 0; i < A.length; i++){ if (setB[A[i]]) inter++; }
  for (i = 0; i < Bx.length; i++){ if (!setA[Bx[i]]) exclB++; }
  var union = A.length + exclB;
  return union === 0 ? 0 : (inter / union);
}

/*  Decide si dos nombres de PRODUCTO son el mismo insumo escrito de dos maneras.
    Devuelve { dup, pct, razon }.

    Por qué no basta el Jaccard: «Hielo iglú» / «Hielo iglú bolsa» y
    «Mítica Easy Lager x300» / «Mítica Pale Lager x300» dan ambos 67%, y solo el
    primero es duplicado. Lo que los separa no es el porcentaje sino la FORMA:
      · contención  → un nombre es el otro más un detalle  → mismo insumo
      · divergencia → cada nombre tiene un token propio     → productos hermanos
    Y un token propio que sea variante o lleve dígitos cierra el caso: no son el mismo. */
function esDuplicadoProbable_(a, b){
  var A = tokens_(a), Bx = tokens_(b), i;
  var nulo = { dup:false, pct:0, razon:'' };
  if (A.length === 0 || Bx.length === 0) return nulo;

  var setA = {}, setB = {};
  for (i = 0; i < A.length; i++) setA[A[i]] = true;
  for (i = 0; i < Bx.length; i++) setB[Bx[i]] = true;

  var inter = 0, exclA = [], exclB = [];
  for (i = 0; i < A.length; i++){ if (setB[A[i]]) inter++; else exclA.push(A[i]); }
  for (i = 0; i < Bx.length; i++){ if (!setA[Bx[i]]) exclB.push(Bx[i]); }
  if (inter === 0) return nulo;

  var union = A.length + exclB.length;
  var pct = Math.round(inter / union * 100);

  /* Un token propio discriminante = son productos distintos, no un duplicado. */
  if (hayDiscriminante_(exclA) || hayDiscriminante_(exclB)) return { dup:false, pct:pct, razon:'' };

  /* Contención: uno es el otro más un detalle. */
  if ((exclA.length === 0 || exclB.length === 0) && pct >= 50){
    return { dup:true, pct:pct, razon:'uno contiene al otro' };
  }
  if (pct >= 80) return { dup:true, pct:pct, razon:'nombres casi iguales' };
  return { dup:false, pct:pct, razon:'' };
}

/*  Para PROVEEDORES la comparación por palabras no sirve: los nombres se abrevian y
    se parten («Surtitienda Cañaveral» vs «Súper tiendas cañaveral» comparten una sola
    palabra y dan 25%). Ahí lo que se parece es la cadena de letras, no el juego de
    palabras, así que se usa similitud por trigramas (Dice), que los sube a 65%.       */
function trigramas_(s){
  var t = clave_(s).replace(/ /g, '');
  var out = [], visto = {}, i;
  if (t.length < 3) { if (t !== '') out.push(t); return out; }
  for (i = 0; i <= t.length - 3; i++){
    var g = t.substring(i, i + 3);
    if (!visto[g]){ visto[g] = true; out.push(g); }
  }
  return out;
}
function simTrigramas_(a, b){
  var A = trigramas_(a), Bx = trigramas_(b), i;
  if (A.length === 0 || Bx.length === 0) return 0;
  var setB = {};
  for (i = 0; i < Bx.length; i++) setB[Bx[i]] = true;
  var inter = 0;
  for (i = 0; i < A.length; i++){ if (setB[A[i]]) inter++; }
  return (2 * inter) / (A.length + Bx.length);   /* Dice */
}

function mediana_(arr){
  var v = [], i;
  for (i = 0; i < arr.length; i++){ if (esNumero_(arr[i]) && arr[i] > 0) v.push(arr[i]); }
  if (v.length === 0) return '';
  v.sort(function(x, y){ return x - y; });
  var n = v.length, mit = Math.floor(n / 2);
  return (n % 2) ? v[mit] : (v[mit - 1] + v[mit]) / 2;
}

function redondear_(n, d){
  if (!esNumero_(n)) return n;
  var f = Math.pow(10, d || 0);
  return Math.round(n * f) / f;
}

function soloFecha_(d){
  if (!(d instanceof Date) || isNaN(d.getTime())) return '';
  return new Date(d.getFullYear(), d.getMonth(), d.getDate());
}
function aFecha_(v){
  /* Se normaliza a medianoche: una celda con hora hacía que hoy pareciera futuro. */
  if (v instanceof Date && !isNaN(v.getTime())) return soloFecha_(v);
  var s = txt_(v);
  if (s === '') return '';
  var m = s.match(/^(\d{4})-(\d{1,2})-(\d{1,2})$/);
  if (m) { var d1 = new Date(Number(m[1]), Number(m[2]) - 1, Number(m[3])); return isNaN(d1.getTime()) ? '' : soloFecha_(d1); }
  m = s.match(/^(\d{1,2})[\/\-](\d{1,2})[\/\-](\d{4})$/);
  if (m){
    var dd = Number(m[1]), mm = Number(m[2]), aa = Number(m[3]);
    if (dd < 1 || dd > 31 || mm < 1 || mm > 12) return '';
    var d2 = new Date(aa, mm - 1, dd);
    if (d2.getDate() !== dd || d2.getMonth() !== mm - 1) return '';
    return soloFecha_(d2);
  }
  return '';
}

/* ---- CONFIG por nombre de parámetro ---- */
function cfg_(nombre, porDefecto){
  var h = hojaOpcional_(H.CFG);
  if (!h) return porDefecto;
  var n = h.getLastRow();
  if (n < 2) return porDefecto;
  var d = h.getRange(2, 1, n - 1, 2).getValues();
  for (var i = 0; i < d.length; i++){
    if (clave_(d[i][0]) === clave_(nombre)){
      var v = d[i][1];
      if (v === '' || v === null || v === undefined) return porDefecto;
      return v;
    }
  }
  return porDefecto;
}
function cfgNum_(nombre, porDefecto){
  var v = num_(cfg_(nombre, porDefecto));
  return esNumero_(v) ? v : porDefecto;
}
function cfgSi_(nombre, porDefecto){
  var v = clave_(cfg_(nombre, porDefecto));
  return v === 'si' || v === 'true' || v === '1';
}

/* canDomainEdit() LANZA EXCEPCIÓN en cuentas Gmail personales.
   Un guardián del tipo if(p.canDomainEdit && p.canDomainEdit()) la LLAMA y revienta. */
function protegerSeguro_(hojaObj, descripcion, rangosLibres){
  try {
    var protecciones = hojaObj.getProtections(SpreadsheetApp.ProtectionType.SHEET);
    var i;
    for (i = 0; i < protecciones.length; i++){ try { protecciones[i].remove(); } catch(eR){} }
    var p = hojaObj.protect().setDescription(descripcion || ('Protegida por SGI ' + CODIGO));
    try { p.setWarningOnly(true); } catch(eW){}
    try { p.removeEditors(p.getEditors()); } catch(eE){}
    try { p.setDomainEdit(false); } catch(eD){ /* cuenta personal: se ignora */ }
    if (rangosLibres && rangosLibres.length){
      try { p.setUnprotectedRanges(rangosLibres); } catch(eU){}
    }
    return true;
  } catch(e){ return false; }

/*  Detecta la unidad incoherente con la cantidad, en los dos sentidos:
      · 800 BOT de whisky   → no es conteo, es peso o volumen mal rotulado

function cantidadIncoherente_(unidad, cantidad){
  if (!esNumero_(cantidad) || cantidad <= 0) return false;

}

/*  Hallazgos que invalidan el PRECIO de una línea: esa línea no puede alimentar
    bandas ni maestro. Los demás hallazgos son documentales (fecha, documento,
    sello, observación) y no ensucian la estadística de costo, así que una línea
    en REVISAR por esos motivos SÍ computa: si no, el 82% del histórico migrado
    quedaría fuera y las bandas nacerían vacías. */
var HALL_INVALIDAN_PRECIO = ['H02','H03','H04','H09','H12'];
function computable_(fila){
  if (clave_(fila[B.ESTADO - 1]) === 'reversada') return false;
  var hall = txt_(fila[B.HALLAZGOS - 1]);
  var i;
  for (i = 0; i < HALL_INVALIDAN_PRECIO.length; i++){
    if (tiene_(hall, HALL_INVALIDAN_PRECIO[i])) return false;
  }
  return true;
}

/* ============================== 5 · BANDAS DE PRECIO (doble capa) ============================== */

/*  Capa 1 — rango razonable por (CATEGORÍA, UNIDAD): mediana robusta × FACTOR.
    Funciona en la PRIMERA compra de un producto, donde la capa 2 todavía no puede.
    Filtro crítico: la mediana de referencia EXCLUYE los precios ya marcados fuera de
    banda; si no, un dato corrupto se vuelve la referencia y el sistema rechaza lo bueno. */
function calcularBandas_(){
  var filas = leerBD_(), i;
  var grupos = {};
  for (i = 0; i < filas.length; i++){
    var f = filas[i];
    if (!computable_(f)) continue;
    var p = num_(f[B.PRECIO_UNIT - 1]);
    if (!esNumero_(p) || p <= 0) continue;
    var cat = txt_(f[B.CATEGORIA - 1]) || 'OTROS';
    var uni = txt_(f[B.UNIDAD - 1]) || '(sin unidad)';
    var hall = txt_(f[B.HALLAZGOS - 1]);
    if (tiene_(hall, 'H12')) continue;   /* no se autoalimenta con lo que ya está fuera de banda */
    var k = cat + '|' + uni;
    if (!grupos[k]) grupos[k] = { cat: cat, uni: uni, v: [] };
    grupos[k].v.push(p);
  }
  var factor = cfgNum_('FACTOR_BANDA_PRECIO', 20);
  var heredado = cfgNum_('FACTOR_BANDA_HEREDADA', 3);
  var salida = [];
  var porCategoria = {};
  var k2;
  for (k2 in grupos){
    if (!grupos.hasOwnProperty(k2)) continue;
    var g = grupos[k2];
    var med = mediana_(g.v);
    if (!esNumero_(med) || med <= 0) continue;
    if (!porCategoria[g.cat]) porCategoria[g.cat] = [];
    porCategoria[g.cat].push(med);
    /* Menos de 3 datos limpios = poca confianza: se ensancha igual que una banda heredada. */
    var f = (g.v.length < 3) ? (factor * heredado) : factor;
    salida.push({
      cat: g.cat, uni: g.uni, n: g.v.length, mediana: med,
      min: med / f, max: med * f, heredada: (g.v.length < 3) ? 'POCOS DATOS' : 'NO'
    });
  }
  return { bandas: salida, porCategoria: porCategoria, factor: factor, heredado: heredado };
}

function buscarBanda_(bandas, cat, uni){
  var i;
  for (i = 0; i < bandas.length; i++){
    if (bandas[i].cat === cat && bandas[i].uni === uni) return bandas[i];
  }
  return null;
}

/*  Mediana del propio producto EN LA MISMA UNIDAD, excluyendo lo ya marcado fuera
    de banda. Sin el filtro de unidad se compara $/gramo contra $/kilo del mismo
    insumo: el maracuyá a $8.000/kg salía «100% desviado» frente a su propio
    histórico en gramos. La unidad es parte de la identidad del precio. */
function medianaProducto_(filas, prodNorm, unidad){
  var v = [], i;
  for (i = 0; i < filas.length; i++){
    var f = filas[i];
    if (!computable_(f)) continue;
    if (clave_(f[B.PROD_NORM - 1]) !== clave_(prodNorm)) continue;
    if (txt_(unidad) !== '' && txt_(f[B.UNIDAD - 1]) !== txt_(unidad)) continue;
    if (tiene_(txt_(f[B.HALLAZGOS - 1]), 'H12')) continue;
    var p = num_(f[B.PRECIO_UNIT - 1]);
    if (esNumero_(p) && p > 0) v.push(p);
  }
  return { mediana: mediana_(v), n: v.length };
}

/*  Evalúa un precio. Devuelve {nivel:'ok'|'aviso'|'bloquea', texto:''}
    Capa 1 primero (sirve desde la primera observación), luego capa 2. */
function evaluarPrecio_(filas, bandasCalc, prodNorm, cat, uni, precio){
  if (!esNumero_(precio) || precio <= 0) return { nivel:'ok', texto:'' };
  var res = { nivel:'ok', texto:'' };
  var b = buscarBanda_(bandasCalc.bandas, cat, uni);
  if (!b){
    /* banda heredada de la categoría, ensanchada ×FACTOR_BANDA_HEREDADA */
    var meds = bandasCalc.porCategoria[cat];
    if (meds && meds.length){
      var m = mediana_(meds);
      if (esNumero_(m) && m > 0){
        b = { cat:cat, uni:uni, n:0, mediana:m,
              min: m / (bandasCalc.factor * bandasCalc.heredado),
              max: m * bandasCalc.factor * bandasCalc.heredado, heredada:'SI' };
      }
    }
  }
  if (b && (precio < b.min || precio > b.max)){
    var veces = redondear_(precio / b.mediana, 1);
    return { nivel:'bloquea',
             texto:'Precio ' + redondear_(precio,2) + ' fuera de la banda de ' + cat + '/' + (uni||'sin unidad') +
                   ' (' + redondear_(b.min,2) + ' a ' + redondear_(b.max,2) + '; ' + veces + '× la mediana' +
                   (b.heredada !== 'NO' ? ', banda ' + (b.heredada === 'SI' ? 'heredada' : 'ensanchada por pocos datos') : '') + ')' };
  }
  var mp = medianaProducto_(filas, prodNorm, uni);
  if (esNumero_(mp.mediana) && mp.mediana > 0){
    var x = precio / mp.mediana;
    var pct = Math.abs(x - 1) * 100;
    var atip = cfgNum_('DESVIO_ATIPICO_X', 3);
    var blo  = cfgNum_('DESVIO_BLOQUEO_PCT', 60);
    var avi  = cfgNum_('DESVIO_AVISO_PCT', 15);
    if (x >= atip || x <= (1 / atip)){
      return { nivel:'bloquea', texto:'Precio ' + redondear_(x,1) + '× la mediana histórica del producto en ' + (uni || 'esa unidad') + ' (' + redondear_(mp.mediana,2) + ')' };
    }
    if (pct > blo){
      return { nivel:'bloquea', texto:'Precio se desvía ' + redondear_(pct,0) + '% de la mediana del producto en ' + (uni || 'esa unidad') + ' (' + redondear_(mp.mediana,2) + ')' };
    }
    if (pct > avi){
      return { nivel:'aviso', texto:'Precio se desvía ' + redondear_(pct,0) + '% de la mediana del producto en ' + (uni || 'esa unidad') + ' (' + redondear_(mp.mediana,2) + ')' };
    }
  }
  return res;
}

/* ============================== 7 · COMPUERTAS DE VALIDACIÓN ============================== */

/*  Dos niveles:
      FATAL      — no guarda nunca.
      BLOQUEANTE — no guarda, pero admite forzado con motivo (≥ MIN_MOTIVO_OVERRIDE),
                   firmado en BITÁCORA. Nunca bloqueo duro sin salida: eso empuja al
                   equipo a llevar registros paralelos en papel.
    Los AVISOS no impiden nada. */
function validarRecepcion_(cap){
  var fatales = [], bloqueantes = [], avisos = [];
  var enc = cap.enc, L = cap.lineas, i, j;

  /* ---- F1 encabezado ---- */
  var faltan = [];
  if (!(enc.fecha instanceof Date)) faltan.push('fecha');
  if (enc.proveedor === '') faltan.push('proveedor');
  if (enc.recibe === '')    faltan.push('responsable que recibe');
  if (faltan.length) fatales.push('F1 · Encabezado incompleto: falta ' + faltan.join(', ') + '.');

  /* ---- F2 fecha razonable ---- */
  if (enc.fecha instanceof Date){
    var hoy = hoy_();
    var dif = diasEntre_(enc.fecha, hoy);
    if (dif < 0) fatales.push('F2 · La fecha de recepción es futura (' + fmtFecha_(enc.fecha) + ').');
    else if (dif > cfgNum_('DIAS_ATRAS_MAXIMO', 60)) avisos.push('Fecha de hace ' + dif + ' días.');
  } else if (txt_(enc.fechaCruda) !== ''){
    fatales.push('F2 · La fecha "' + txt_(enc.fechaCruda) + '" no es una fecha válida.');
  }

  /* ---- F3 al menos una línea ---- */
  if (L.length === 0) fatales.push('F3 · No hay ninguna línea de producto.');

  /*  ---- B11 proveedor dentro del catálogo (v1.1) ----
      Es BLOQUEANTE CON OVERRIDE, no fatal, a propósito: la validación de D6 ya impide
      escribirlo, pero un PEGADO se salta cualquier validación de datos. B11 es la red
      que atrapa ese caso, y admite forzado firmado para no dejar al bar sin poder
      registrar una compra de urgencia un domingo a las once de la noche.              */
  if (enc.proveedor !== '' && !existeProveedor_(enc.proveedor)){
    var listaP = proveedoresCatalogo_(), parecidos = [];
    var umbralP = cfgNum_('SIMILITUD_DUPLICADO_PCT', 55) / 100;
    for (i = 0; i < listaP.length; i++){
      if (simTrigramas_(enc.proveedor, listaP[i].nombre) >= umbralP) parecidos.push(listaP[i].nombre);
    }
    bloqueantes.push('B11 · "' + enc.proveedor + '" no está en el catálogo de proveedores.' +
      (parecidos.length ? ' ¿Querías decir ' + parecidos.join(' / ') + '?'
                        : ' Si es realmente nuevo, dalo de alta en ⚙ Di Égida Bar → Dar de alta un proveedor.'));
  }

  /* ---- F4 documento duplicado ---- */
  if (enc.documento === ''){
    bloqueantes.push('B1 · Recepción sin N° de remisión o factura.');
  } else {
    var filas = leerBD_();
    var avisoOtroProv = '';
    for (i = 0; i < filas.length; i++){
      if (!esActiva_(filas[i])) continue;
      if (clave_(filas[i][B.DOCUMENTO - 1]) !== clave_(enc.documento)) continue;
      var provBD = txt_(filas[i][B.PROVEEDOR - 1]);
      if (clave_(provBD) === clave_(enc.proveedor)){
        fatales.push('F4 · El documento "' + enc.documento + '" ya está registrado para ' + provBD +
                     ' (' + fmtFecha_(filas[i][B.FECHA - 1]) + '). Si es una corrección, usa Reversar línea.');
        avisoOtroProv = '';
        break;
      }
      avisoOtroProv = 'El documento "' + enc.documento + '" ya existe con otro proveedor (' + provBD +
                      '). Verifica que no sea un número copiado por error.';
    }
    if (avisoOtroProv !== '') avisos.push(avisoOtroProv);
  }

  /* ---- Por línea ---- */
  var filasBD = leerBD_();
  var bandas = calcularBandas_();
  var vistos = {};
  for (i = 0; i < L.length; i++){
    var l = L[i], ref = 'Fila ' + l.fila + ' (' + (l.producto || 'sin nombre') + ')';

    if (l.producto === ''){
      fatales.push('F5 · ' + ref + ': hay datos pero el nombre del producto está vacío.');
      continue;
    }
    if (vistos[l.prodNorm]) avisos.push(ref + ': producto repetido en esta misma recepción.');
    vistos[l.prodNorm] = true;

    /* unidad */
    if (l.unidad === ''){
      bloqueantes.push('B2 · ' + ref + ': unidad vacía o no reconocida ("' + l.unidadCruda + '"). ' +
                       'Sin unidad no hay control de precio posible.');
    }

    /* cantidades */
    if (!esNumero_(l.recibida)){
      bloqueantes.push('B3 · ' + ref + ': cantidad recibida vacía o no numérica.');
    } else if (l.recibida < 0){
      bloqueantes.push('B3 · ' + ref + ': cantidad recibida negativa.');
    }

    /* coherencia cantidad / condición */
    if (cfgSi_('BLOQUEA_COHERENCIA', 'SI')){
      if (esNumero_(l.recibida) && l.recibida === 0 && tiene_(clave_(l.condicion), 'ok')){
        bloqueantes.push('B4 · ' + ref + ': recibido 0 pero marcado como "✓ OK". Corrige la cantidad o la condición.');
      }
      if (tiene_(clave_(l.condicion), 'no llego') && esNumero_(l.recibida) && l.recibida > 0){
        bloqueantes.push('B4 · ' + ref + ': marcado "✕ No llegó" pero con cantidad recibida ' + l.recibida + '.');
      }
      if (esNumero_(l.recibida) && l.recibida === 0 && esNumero_(l.precio) && l.precio > 0 && !tiene_(clave_(l.condicion), 'no llego')){
        bloqueantes.push('B4 · ' + ref + ': cantidad 0 con precio ' + l.precio + '. El valor del ítem quedaría en cero.');
      }
    }

    /* diferencia pedida vs recibida */
    if (esNumero_(l.pedida) && esNumero_(l.recibida) && l.pedida !== l.recibida){
      var d = l.recibida - l.pedida;
      var texto = ref + ': diferencia de ' + (d > 0 ? '+' : '') + d + ' frente a lo pedido sin observación.';
      if (l.obs === ''){
        if (cfgSi_('BLOQUEA_DIFERENCIA', 'NO')) bloqueantes.push('B5 · ' + texto);
        else avisos.push(texto);
      }
    }

    /* unidad incoherente con la cantidad */
    if (cantidadIncoherente_(l.unidad, l.recibida)){
      bloqueantes.push('B10 · ' + ref + ': ' + l.recibida + ' ' + l.unidad +
                       ' no es una cantidad plausible. ¿Estás digitando gramos o mililitros bajo otra unidad?');
    }

    /* precio */
    if (!esNumero_(l.precio) || l.precio <= 0){
      bloqueantes.push('B6 · ' + ref + ': sin precio unitario. Una línea sin precio no cuesta nada y ensucia el costeo.');
    } else if (cfgSi_('BLOQUEA_BANDA', 'SI')){
      var ev = evaluarPrecio_(filasBD, bandas, l.prodNorm, l.categoria, l.unidad, l.precio);
      if (ev.nivel === 'bloquea') bloqueantes.push('B7 · ' + ref + ': ' + ev.texto + '.');
      else if (ev.nivel === 'aviso') avisos.push(ref + ': ' + ev.texto + '.');
    }

    /* sello en licores y vinos */
    if (esLicorOVino_(l.categoria) && cfgSi_('BLOQUEA_SELLO', 'SI')){
      var s = clave_(l.sello);
      if (s === '') bloqueantes.push('B8 · ' + ref + ': licor o vino sin marcar el sello de seguridad.');
      else if (s === 'no') bloqueantes.push('B8 · ' + ref + ': sello ROTO. Deja constancia en observaciones antes de forzar.');
    }

    /* vencimiento */
    if (l.vencimiento instanceof Date){
      var dv = diasEntre_(hoy_(), l.vencimiento);
      if (dv < 0) bloqueantes.push('B9 · ' + ref + ': producto VENCIDO desde hace ' + Math.abs(dv) + ' días.');
      else if (dv <= cfgNum_('DIAS_VENCIMIENTO_AVISO', 15)) avisos.push(ref + ': vence en ' + dv + ' días.');
    } else if (esPerecedero_(l.categoria) && l.vencCrudo !== '' && clave_(l.vencCrudo) !== 'no ap' && clave_(l.vencCrudo) !== 'n a'){
      avisos.push(ref + ': el vencimiento "' + l.vencCrudo + '" no es una fecha válida.');
    }

    /* catálogo de productos: producto nuevo o gemelo de uno existente (v1.1) */
    if (!existeEnCatalogo_(l.prodNorm)){
      var gemelo = productoParecido_(l.producto, l.categoria);
      if (gemelo !== '') avisos.push(ref + ': producto nuevo, pero se parece a "' + gemelo +
                                     '" que ya está en el catálogo. Si es el mismo, úsalo desde la lista.');
      else avisos.push(ref + ': producto nuevo, se agregará al CATÁLOGO.');
    }
  }

  return { fatales: fatales, bloqueantes: bloqueantes, avisos: avisos };
}

function existeEnCatalogo_(prodNorm){
  var h = hojaOpcional_(H.CAT);
  if (!h) return false;
  var n = h.getLastRow();
  if (n < 2) return false;
  var d = h.getRange(2, 1, n - 1, 1).getValues(), i;
  for (i = 0; i < d.length; i++){ if (clave_(d[i][0]) === clave_(prodNorm)) return true; }
  return false;
}

/* v1.1 · gemelo probable dentro de la misma categoría, para avisar al capturar */
function productoParecido_(nombre, categoria){
  var h = hojaOpcional_(H.CAT);
  if (!h) return '';
  var n = h.getLastRow();
  if (n < FILA_DATOS) return '';
  var d = h.getRange(FILA_DATOS, 1, n - FILA_DATOS + 1, 3).getValues(), i;
  for (i = 0; i < d.length; i++){
    var otro = txt_(d[i][1]);
    if (otro === '') continue;
    if (txt_(categoria) !== '' && txt_(d[i][2]) !== '' && txt_(d[i][2]) !== txt_(categoria)) continue;
    var r = esDuplicadoProbable_(nombre, otro);
    if (r.dup) return otro;
  }
  return '';
}

function guardarRecepcion(){
  var lock = LockService.getScriptLock();
  try { lock.waitLock(30000); } catch(e){ escribirMensaje_('⏳ Otra operación está en curso. Espera unos segundos y vuelve a marcar la casilla.', 'aviso'); return; }
  try {
    var cap = leerCaptura_();
    var v = validarRecepcion_(cap);

    if (v.fatales.length){
      escribirMensaje_('✕ NO SE GUARDÓ — ' + v.fatales.length + ' problema(s) que no admiten forzado:\n· ' +
                       v.fatales.join('\n· ') + (v.bloqueantes.length ? '\nAdemás: ' + v.bloqueantes.length + ' bloqueo(s).' : ''), 'mal');
      bitacora_('GUARDADO RECHAZADO', v.fatales.join(' | '), '');
      return;
    }

    var minMotivo = cfgNum_('MIN_MOTIVO_OVERRIDE', 15);
    var motivo = cap.enc.motivo;
    var forzado = 'NO';

    if (v.bloqueantes.length){
      if (motivo.length < minMotivo){
        escribirMensaje_('⛔ NO SE GUARDÓ — ' + v.bloqueantes.length + ' bloqueo(s):\n· ' + v.bloqueantes.join('\n· ') +
                         '\n\nCorrige, o escribe el motivo del forzado en ' + C.MOTIVO + ' (mínimo ' + minMotivo +
                         ' caracteres) y vuelve a marcar la casilla.', 'mal');
        bitacora_('GUARDADO BLOQUEADO', v.bloqueantes.join(' | '), '');
        return;
      }
      forzado = 'SI';
    }

    var idDoc = siguienteIdDocumento_(cap.enc.fecha);
    var hBD = hoja_(H.BD);
    var ahora = new Date();
    var user = usuario_();
    var provNorm = clave_(cap.enc.proveedor);
    var filasNuevas = [], i;

    for (i = 0; i < cap.lineas.length; i++){
      var l = cap.lineas[i];
      var hallazgos = hallazgosDeLinea_(l, v);
      var f = [];
      f[B.ID_LINEA - 1]     = idDoc + '-' + pad3_(i + 1);
      f[B.ID_DOC - 1]       = idDoc;
      f[B.FECHA - 1]        = cap.enc.fecha;
      f[B.PROVEEDOR - 1]    = cap.enc.proveedor;
      f[B.PROV_NORM - 1]    = provNorm;
      f[B.DOCUMENTO - 1]    = cap.enc.documento;
      f[B.PRODUCTO - 1]     = l.producto;
      f[B.PROD_NORM - 1]    = l.prodNorm;
      f[B.CATEGORIA - 1]    = l.categoria;
      f[B.PRESENTACION - 1] = l.presentacion;
      f[B.UNIDAD - 1]       = l.unidad;
      f[B.CANT_PED - 1]     = esNumero_(l.pedida) ? l.pedida : '';
      f[B.CANT_REC - 1]     = esNumero_(l.recibida) ? l.recibida : '';
      f[B.DIFERENCIA - 1]   = (esNumero_(l.pedida) && esNumero_(l.recibida)) ? (l.recibida - l.pedida) : '';
      f[B.PRECIO_UNIT - 1]  = esNumero_(l.precio) ? l.precio : '';
      f[B.VALOR_TOTAL - 1]  = (esNumero_(l.precio) && esNumero_(l.recibida)) ? redondear_(l.precio * l.recibida, 2) : '';
      f[B.CONDICION - 1]    = l.condicion;
      f[B.SELLO - 1]        = l.sello;
      f[B.VENCIMIENTO - 1]  = (l.vencimiento instanceof Date) ? l.vencimiento : l.vencCrudo;
      f[B.LOTE - 1]         = l.lote;
      f[B.OBSERVACIONES - 1]= l.obs;
      f[B.RECIBE - 1]       = cap.enc.recibe;
      f[B.USUARIO - 1]      = user;
      f[B.TIMESTAMP - 1]    = ahora;
      f[B.OVERRIDE - 1]     = forzado;
      f[B.MOTIVO - 1]       = forzado === 'SI' ? motivo : '';
      f[B.ESTADO - 1]       = 'ACTIVA';
      f[B.REF_REVERSA - 1]  = '';
      f[B.ORIGEN - 1]       = 'CAPTURA';
      f[B.HALLAZGOS - 1]    = hallazgos;
      var j;
      for (j = 0; j < BD_ULT_COL; j++){ if (f[j] === undefined) f[j] = ''; }
      filasNuevas.push(f);
    }

    hBD.getRange(hBD.getLastRow() + 1, 1, filasNuevas.length, BD_ULT_COL).setValues(filasNuevas);

    bitacora_('RECEPCIÓN GUARDADA',
              idDoc + ' · ' + cap.enc.proveedor + ' · doc ' + cap.enc.documento + ' · ' + filasNuevas.length + ' línea(s)' +
              (forzado === 'SI' ? ' · FORZADA' : ''), forzado === 'SI' ? motivo : '');

    /* El refresco de derivados es NO FATAL: jamás puede revertir un guardado. */
    var problemasRecalc = '';
    try { recalcularTodo_(); } catch(eR){ problemasRecalc = ' (los tableros se actualizarán en el próximo recálculo)'; }

    var resumen = '✔ GUARDADO — ' + idDoc + ' · ' + filasNuevas.length + ' línea(s) en el libro mayor' +
                  (forzado === 'SI' ? ' · FORZADA con motivo registrado' : '') + problemasRecalc;
    if (v.avisos.length) resumen += '\nAvisos: · ' + v.avisos.join('\n· ');
    resumen += '\n\nMarca "Nueva recepción" (' + C.CHK_NUEVA + ') para limpiar el formulario.';
    escribirMensaje_(resumen, forzado === 'SI' ? 'aviso' : 'ok');

  } catch(e){
    escribirMensaje_('✕ Error al guardar: ' + (e && e.message ? e.message : e), 'mal');
    bitacora_('ERROR AL GUARDAR', (e && e.message ? e.message : String(e)), '');
  } finally {
    try { hoja_(H.CAP).getRange(C.CHK_GUARDAR).setValue(false); } catch(e2){}
    try { lock.releaseLock(); } catch(e3){}
  }
}

function hallazgosDeLinea_(l, v){
  var cods = [];
  if (!esNumero_(l.precio) || l.precio <= 0) cods.push('H03');
  if (l.unidad === '') cods.push('H02');
  if (esNumero_(l.recibida) && l.recibida === 0 && tiene_(clave_(l.condicion), 'ok')) cods.push('H06');
  if (tiene_(clave_(l.condicion), 'no llego') && esNumero_(l.recibida) && l.recibida > 0) cods.push('H07');
  if (esNumero_(l.pedida) && esNumero_(l.recibida) && l.pedida !== l.recibida && l.obs === '') cods.push('H08');
  if (cantidadIncoherente_(l.unidad, l.recibida)) cods.push('H09');
  if (esLicorOVino_(l.categoria) && clave_(l.sello) === '') cods.push('H10');
  var i;
  for (i = 0; i < v.bloqueantes.length; i++){
    if (tiene_(v.bloqueantes[i], 'Fila ' + l.fila + ' ') && tiene_(v.bloqueantes[i], 'B7 ·')) { cods.push('H12'); break; }
  }
  return cods.join(';');
}

function pad3_(n){
  var s = String(n);
  while (s.length < 3) s = '0' + s;
  return s;
}

function siguienteIdDocumento_(fecha){
  var f = (fecha instanceof Date) ? fecha : hoy_();
  var base = 'BAR-' + Utilities.formatDate(f, Session.getScriptTimeZone(), 'yyyyMMdd');
  var filas = leerBD_(), max = 0, i;
  for (i = 0; i < filas.length; i++){
    var id = txt_(filas[i][B.ID_DOC - 1]);
    if (empiezaCon_(id, base + '-')){
      var n = parseInt(id.substring(base.length + 1), 10);
      if (!isNaN(n) && n > max) max = n;
    }
  }
  return base + '-' + pad3_(max + 1);
}

/*  v1.1 · la marca de duplicado usa esDuplicadoProbable_ y solo compara productos de la
    MISMA categoría. Con el umbral viejo (Jaccard ≥ 0.6 a secas) el catálogo real
    marcaba 3 pares y solo 1 era duplicado; dos avisos falsos de tres enseñan a ignorar
    la columna, que es la peor manera de perder un control.                              */
function reconstruirCatalogo_(){
  var filas = leerBD_(), i;
  var mapa = {};
  for (i = 0; i < filas.length; i++){
    var f = filas[i];
    if (!esActiva_(f)) continue;
    var k = clave_(f[B.PROD_NORM - 1] || f[B.PRODUCTO - 1]);
    if (k === '') continue;
    if (!mapa[k]){
      mapa[k] = { clave:k, nombre: txt_(f[B.PRODUCTO - 1]), cat: txt_(f[B.CATEGORIA - 1]),
                  unidades:{}, n:0, ultima:'' };
    }
    var u = txt_(f[B.UNIDAD - 1]);
    if (u !== '') mapa[k].unidades[u] = (mapa[k].unidades[u] || 0) + 1;
    mapa[k].n++;
    var fe = f[B.FECHA - 1];
    if (fe instanceof Date && (!(mapa[k].ultima instanceof Date) || fe > mapa[k].ultima)) mapa[k].ultima = fe;
  }
  var claves = [], k2;
  for (k2 in mapa){ if (mapa.hasOwnProperty(k2)) claves.push(k2); }
  claves.sort();
  var out = [], j;
  for (j = 0; j < claves.length; j++){
    var m = mapa[claves[j]];
    var us = [], uk, mejor = '', mejorN = -1;
    for (uk in m.unidades){
      if (!m.unidades.hasOwnProperty(uk)) continue;
      us.push(uk);
      if (m.unidades[uk] > mejorN){ mejorN = m.unidades[uk]; mejor = uk; }
    }
    var estado = 'OK';
    if (us.length === 0) estado = 'SIN UNIDAD';
    else if (us.length > 1) estado = 'UNIDAD AMBIGUA (' + us.join('/') + ')';
    out.push([m.clave, m.nombre, m.cat, mejor, us.join(' / '), m.n, m.ultima, estado, '']);
  }

  /* Posibles duplicados: misma categoría + forma del nombre, no porcentaje a secas. */
  var a, b2;
  for (a = 0; a < out.length; a++){
    for (b2 = a + 1; b2 < out.length; b2++){
      if (txt_(out[a][2]) !== '' && txt_(out[b2][2]) !== '' && txt_(out[a][2]) !== txt_(out[b2][2])) continue;
      var r = esDuplicadoProbable_(out[a][1], out[b2][1]);
      if (r.dup){
        out[a][8]  = txt_(out[a][8]  + ' ¿= ' + out[b2][1] + '? (' + r.pct + '%, ' + r.razon + ')');
        out[b2][8] = txt_(out[b2][8] + ' ¿= ' + out[a][1]  + '? (' + r.pct + '%, ' + r.razon + ')');
      }
    }
  }
  var h = hoja_(H.CAT);
  limpiarCuerpo_(h, FILA_DATOS, 9);
  if (out.length) h.getRange(FILA_DATOS, 1, out.length, 9).setValues(out);
  h.getRange(FILA_DATOS, 7, Math.max(out.length, 1), 1).setNumberFormat('yyyy-mm-dd');
  return out.length;
}

var _cacheProv = null;
function invalidarCacheProv_(){ _cacheProv = null; }

function proveedoresCatalogo_(){
  if (_cacheProv) return _cacheProv;
  var out = [];
  try {
    var h = hojaOpcional_(H.PRV);
    if (h){
      var n = h.getLastRow();
      if (n >= FILA_DATOS){
        var d = h.getRange(FILA_DATOS, 1, n - FILA_DATOS + 1, 2).getValues(), i;
        for (i = 0; i < d.length; i++){
          var nombre = txt_(d[i][1]);
          if (nombre === '') continue;
          out.push({ clave: clave_(txt_(d[i][0]) !== '' ? d[i][0] : nombre), nombre: nombre });
        }
      }
    }
  } catch(e){ out = []; }
  _cacheProv = out;
  return out;
}

function existeProveedor_(nombre){
  var lista = proveedoresCatalogo_(), k = clave_(nombre), i;
  if (k === '') return false;
  for (i = 0; i < lista.length; i++){ if (lista[i].clave === k) return true; }
  return false;
}

/*  Deriva la lista del libro mayor y CONSERVA las altas manuales que todavía no
    tienen compras. Sin esa conservación un proveedor nuevo nunca podría entrar:
    la lista cerrada lo rechaza y el libro mayor no lo conoce todavía.                  */
function reconstruirProveedores_(){
  var h = hoja_(H.PRV);
  invalidarCacheProv_();

  /* 1 · altas manuales previas (0 recepciones) que hay que preservar */
  var manuales = {}, n = h.getLastRow(), i;
  if (n >= FILA_DATOS){
    var prev = h.getRange(FILA_DATOS, 1, n - FILA_DATOS + 1, 3).getValues();
    for (i = 0; i < prev.length; i++){
      var nom = txt_(prev[i][1]);
      if (nom === '') continue;
      var rec = num_(prev[i][2]);
      if (!esNumero_(rec) || rec === 0) manuales[clave_(nom)] = nom;
    }
  }

  /* 2 · lo que dice el libro mayor */
  var filas = leerBD_();
  var mapa = {};
  for (i = 0; i < filas.length; i++){
    var f = filas[i];
    if (!esActiva_(f)) continue;
    var nombre = txt_(f[B.PROVEEDOR - 1]);
    if (nombre === '') continue;
    var k = clave_(f[B.PROV_NORM - 1] !== '' ? f[B.PROV_NORM - 1] : nombre);
    if (!mapa[k]) mapa[k] = { clave:k, nombre:nombre, docs:{}, lineas:0, ultima:'', valor:0 };
    mapa[k].docs[txt_(f[B.ID_DOC - 1])] = true;
    mapa[k].lineas++;
    var v = num_(f[B.VALOR_TOTAL - 1]); if (esNumero_(v)) mapa[k].valor += v;
    var fe = f[B.FECHA - 1];
    if (fe instanceof Date && (!(mapa[k].ultima instanceof Date) || fe > mapa[k].ultima)) mapa[k].ultima = fe;
  }
  var mk;
  for (mk in manuales){
    if (!manuales.hasOwnProperty(mk)) continue;
    if (!mapa[mk]) mapa[mk] = { clave:mk, nombre:manuales[mk], docs:{}, lineas:0, ultima:'', valor:0 };
  }

  /* 3 · a filas ordenadas */
  var claves = [], k2;
  for (k2 in mapa){ if (mapa.hasOwnProperty(k2)) claves.push(k2); }
  claves.sort();
  var out = [], j;
  for (j = 0; j < claves.length; j++){
    var m = mapa[claves[j]];
    var nd = 0, dk;
    for (dk in m.docs){ if (m.docs.hasOwnProperty(dk) && dk !== '') nd++; }
    var estado = 'OK';
    if (m.lineas === 0) estado = 'ALTA MANUAL (sin compras)';
    else if (nd <= 1) estado = 'UNA SOLA RECEPCIÓN';
    out.push([m.clave, m.nombre, nd, m.lineas, m.ultima, redondear_(m.valor, 0), estado, '']);
  }

  /* 4 · posibles duplicados por trigramas */
  var umbral = cfgNum_('SIMILITUD_DUPLICADO_PCT', 55) / 100;
  var a, b2;
  for (a = 0; a < out.length; a++){
    for (b2 = a + 1; b2 < out.length; b2++){
      var s = simTrigramas_(out[a][1], out[b2][1]);
      if (s >= umbral){
        var pc = Math.round(s * 100);
        out[a][7]  = txt_(out[a][7]  + ' ¿= ' + out[b2][1] + '? (' + pc + '%)');
        out[b2][7] = txt_(out[b2][7] + ' ¿= ' + out[a][1]  + '? (' + pc + '%)');
      }
    }
  }

  limpiarCuerpo_(h, FILA_DATOS, 8);
  if (out.length) h.getRange(FILA_DATOS, 1, out.length, 8).setValues(out);
  h.getRange(FILA_DATOS, 5, Math.max(out.length, 1), 1).setNumberFormat('yyyy-mm-dd');
  h.getRange(FILA_DATOS, 6, Math.max(out.length, 1), 1).setNumberFormat('$#,##0');
  invalidarCacheProv_();
  return out.length;
}

/*  Apunta la validación de D6 al catálogo. Si el catálogo está vacío NO se aplica
    lista cerrada: dejaría el formato inutilizable en una instalación nueva.            */
function aplicarValidacionProveedor_(){
  var h = hoja_(H.CAP);
  var hp = hojaOpcional_(H.PRV);
  if (!hp || hp.getLastRow() < FILA_DATOS){
    try { h.getRange(C.PROV).clearDataValidations(); } catch(e){}
    return false;
  }
  var cerrada = cfgSi_('PROVEEDOR_LISTA_CERRADA', 'SI');
  var dv = SpreadsheetApp.newDataValidation()
    .requireValueInRange(hp.getRange('B' + FILA_DATOS + ':B1000'), true)
    .setAllowInvalid(!cerrada)
    .setHelpText('Elige un proveedor de la lista. Para uno nuevo: menú ⚙ Di Égida Bar → Dar de alta un proveedor.')
    .build();
  h.getRange(C.PROV).setDataValidation(dv);
  return true;
}

function altaProveedor(){
  var ui;
  try { ui = SpreadsheetApp.getUi(); }
  catch(e){ toast_('El alta de proveedor solo funciona desde computador.', CODIGO); return; }

  var r1 = ui.prompt('Dar de alta un proveedor',
                     'Nombre del proveedor, tal como debe quedar en todos los registros:',
                     ui.ButtonSet.OK_CANCEL);
  if (r1.getSelectedButton() !== ui.Button.OK) return;
  var nombre = txt_(r1.getResponseText());
  if (nombre === ''){ ui.alert('Nombre vacío. No se dio de alta nada.'); return; }

  if (existeProveedor_(nombre)){ ui.alert('«' + nombre + '» ya está en el catálogo.'); return; }

  /* Compuerta antiduplicado: se avisa ANTES de crear, no después. */
  var lista = proveedoresCatalogo_(), i, parecidos = [];
  var umbral = cfgNum_('SIMILITUD_DUPLICADO_PCT', 55) / 100;
  for (i = 0; i < lista.length; i++){
    var s = simTrigramas_(nombre, lista[i].nombre);
    if (s >= umbral) parecidos.push(lista[i].nombre + '  (' + Math.round(s * 100) + '% de parecido)');
  }
  if (parecidos.length){
    var r2 = ui.alert('Posible duplicado',
      '«' + nombre + '» se parece a:\n\n· ' + parecidos.join('\n· ') +
      '\n\n¿Aun así es un proveedor DISTINTO y quieres crearlo?',
      ui.ButtonSet.YES_NO);
    if (r2 !== ui.Button.YES){
      ui.alert('No se creó nada. Si era el mismo, úsalo desde la lista; ' +
               'si hay dos versiones ya escritas, usa Fusionar proveedores.');
      return;
    }
  }

  var lock = LockService.getScriptLock();
  try { lock.waitLock(30000); } catch(e){ ui.alert('Otra operación en curso.'); return; }
  try {
    var h = hoja_(H.PRV);
    h.appendRow([clave_(nombre), nombre, 0, 0, '', 0, 'ALTA MANUAL (sin compras)', '']);
    invalidarCacheProv_();
    pasoSeguro_([], 'Reconstruir proveedores', function(){ reconstruirProveedores_(); });
    pasoSeguro_([], 'Revalidar D6',            function(){ aplicarValidacionProveedor_(); });
    bitacora_('ALTA PROVEEDOR', nombre, parecidos.length ? 'confirmado pese a ' + parecidos.length + ' parecido(s)' : '');
    ui.alert('«' + nombre + '» quedó en el catálogo y ya aparece en la lista de D6.');
  } catch(e){ ui.alert('Error: ' + (e && e.message ? e.message : e)); }
  finally { try { lock.releaseLock(); } catch(e3){} }
}

/*  Unifica dos nombres del MISMO proveedor ya escritos en el libro mayor.
    El nombre del proveedor es un atributo de identidad, no un dato de la transacción:
    corregirlo no altera cantidades, precios ni fechas, y queda firmado en BITÁCORA.     */
function fusionarProveedor(){
  var ui;
  try { ui = SpreadsheetApp.getUi(); }
  catch(e){ toast_('La fusión solo funciona desde computador.', CODIGO); return; }

  var lista = proveedoresCatalogo_(), i, nombres = [];
  for (i = 0; i < lista.length; i++) nombres.push('· ' + lista[i].nombre);

  var r1 = ui.prompt('Fusionar proveedores (1 de 2)',
    'Escribe el nombre EQUIVOCADO, el que hay que hacer desaparecer:\n\n' + nombres.join('\n'),
    ui.ButtonSet.OK_CANCEL);
  if (r1.getSelectedButton() !== ui.Button.OK) return;
  var viejo = txt_(r1.getResponseText());

  var r2 = ui.prompt('Fusionar proveedores (2 de 2)',
    'Escribe el nombre CORRECTO, el que debe quedar:', ui.ButtonSet.OK_CANCEL);
  if (r2.getSelectedButton() !== ui.Button.OK) return;
  var nuevo = txt_(r2.getResponseText());

  if (viejo === '' || nuevo === '' || clave_(viejo) === clave_(nuevo)){
    ui.alert('Nombres vacíos o iguales. No se fusionó nada.');
    return;
  }

  var lock = LockService.getScriptLock();
  try { lock.waitLock(30000); } catch(e){ ui.alert('Otra operación en curso.'); return; }
  try {
    var h = hoja_(H.BD);
    var n = h.getLastRow();
    if (n < 2){ ui.alert('El libro mayor está vacío.'); return; }
    var rng = h.getRange(2, B.PROVEEDOR, n - 1, 2);
    var d = rng.getValues(), cambios = 0, j;
    for (j = 0; j < d.length; j++){
      if (clave_(d[j][0]) === clave_(viejo)){ d[j][0] = nuevo; d[j][1] = clave_(nuevo); cambios++; }
    }
    if (cambios === 0){ ui.alert('No hay ninguna línea con «' + viejo + '».'); return; }
    rng.setValues(d);

    /* quitar la fila del nombre viejo del catálogo */
    var hp = hoja_(H.PRV);
    var np = hp.getLastRow();
    if (np >= FILA_DATOS){
      var dp = hp.getRange(FILA_DATOS, 1, np - FILA_DATOS + 1, 2).getValues();
      for (j = dp.length - 1; j >= 0; j--){
        if (clave_(dp[j][1]) === clave_(viejo)){ hp.deleteRow(FILA_DATOS + j); }
      }
    }
    invalidarCacheProv_();
    bitacora_('FUSIÓN PROVEEDOR', viejo + ' → ' + nuevo + ' · ' + cambios + ' línea(s)', 'unificación de nombre');
    try { recalcularTodo_(); } catch(eR){}
    ui.alert(cambios + ' línea(s) pasaron de «' + viejo + '» a «' + nuevo + '».');
    return cambios;
  } catch(e){ ui.alert('Error: ' + (e && e.message ? e.message : e)); }
  finally { try { lock.releaseLock(); } catch(e3){} }
}

function reconstruirBandas_(){
  var calc = calcularBandas_();
  var out = [], i;
  for (i = 0; i < calc.bandas.length; i++){
    var b = calc.bandas[i];
    out.push([b.cat, b.uni, b.n, redondear_(b.mediana, 2), redondear_(b.min, 2), redondear_(b.max, 2),
              b.heredada !== 'NO' ? 'ENSANCHADA (pocos datos)' : 'OK']);
  }
  out.sort(function(x, y){ return (x[0] + x[1] < y[0] + y[1]) ? -1 : 1; });
  var h = hoja_(H.BAN);
  limpiarCuerpo_(h, FILA_DATOS, 7);
  if (out.length) h.getRange(FILA_DATOS, 1, out.length, 7).setValues(out);
  return out.length;
}

/*  Marca H12 en el libro mayor. Se corre DESPUÉS de reconstruir bandas y
    ANTES del maestro, para que la mediana del producto excluya lo corrupto. */
function marcarFueraDeBanda_(){
  var h = hojaOpcional_(H.BD);
  if (!h) return 0;
  var n = h.getLastRow();
  if (n < 2) return 0;
  var filas = leerBD_();
  var calc = calcularBandas_();
  var col = h.getRange(2, B.HALLAZGOS, n - 1, 1);
  var vals = col.getValues();
  var cambios = 0, i;
  for (i = 0; i < filas.length; i++){
    var f = filas[i];
    var p = num_(f[B.PRECIO_UNIT - 1]);
    var actual = txt_(vals[i] ? vals[i][0] : '');
    var sinH12 = actual.replace(/H12;?/g, '').replace(/;;/g, ';').replace(/;$/, '');
    var nuevo = sinH12;
    if (esNumero_(p) && p > 0 && computable_(f)){
      var b = buscarBanda_(calc.bandas, txt_(f[B.CATEGORIA - 1]), txt_(f[B.UNIDAD - 1]));
      if (b && (p < b.min || p > b.max)) nuevo = sinH12 === '' ? 'H12' : (sinH12 + ';H12');
    }
    if (nuevo !== actual){ vals[i][0] = nuevo; cambios++; }
  }
  if (cambios) col.setValues(vals);
  return cambios;
}

function reconstruirMaestro_(){
  var filas = leerBD_(), i;
  var mapa = {};
  for (i = 0; i < filas.length; i++){
    var f = filas[i];
    if (!computable_(f)) continue;
    var k = clave_(f[B.PROD_NORM - 1] || f[B.PRODUCTO - 1]);
    if (k === '') continue;
    if (!mapa[k]) mapa[k] = { nombre: txt_(f[B.PRODUCTO - 1]), cat: txt_(f[B.CATEGORIA - 1]),
                              uni: txt_(f[B.UNIDAD - 1]), precios: [], limpios: [], ult: '', ultP: '',
                              prov: {}, cant: 0, valor: 0, n: 0 };
    var m = mapa[k];
    var p = num_(f[B.PRECIO_UNIT - 1]);
    var fe = f[B.FECHA - 1];
    m.n++;
    if (esNumero_(p) && p > 0){
      m.precios.push(p);
      if (!tiene_(txt_(f[B.HALLAZGOS - 1]), 'H12')) m.limpios.push(p);
      if (!(m.ult instanceof Date) || (fe instanceof Date && fe >= m.ult)){ m.ult = fe; m.ultP = p; }
    }
    var c = num_(f[B.CANT_REC - 1]); if (esNumero_(c)) m.cant += c;
    var v = num_(f[B.VALOR_TOTAL - 1]); if (esNumero_(v)) m.valor += v;
    var pr = txt_(f[B.PROVEEDOR - 1]); if (pr !== '') m.prov[pr] = true;
  }
  var claves = [], k2;
  for (k2 in mapa){ if (mapa.hasOwnProperty(k2)) claves.push(k2); }
  claves.sort();
  var out = [], j;
  for (j = 0; j < claves.length; j++){
    var d = mapa[claves[j]];
    var base = d.limpios.length ? d.limpios : d.precios;
    var med = mediana_(base);
    var min = '', max = '', desv = '';
    if (base.length){
      min = base[0]; max = base[0];
      var z;
      for (z = 1; z < base.length; z++){ if (base[z] < min) min = base[z]; if (base[z] > max) max = base[z]; }
      if (esNumero_(med) && med > 0 && esNumero_(d.ultP)) desv = redondear_((d.ultP / med - 1) * 100, 1);
    }
    var nProv = 0, pk;
    for (pk in d.prov){ if (d.prov.hasOwnProperty(pk)) nProv++; }
    var estado = 'OK';
    if (base.length === 0) estado = 'SIN PRECIO';
    else if (base.length === 1) estado = 'UN SOLO DATO';
    else if (esNumero_(desv) && Math.abs(desv) > cfgNum_('DESVIO_AVISO_PCT', 15)) estado = 'ÚLTIMA COMPRA DESVIADA';
    out.push([claves[j], d.nombre, d.cat, d.uni, d.n, d.cant ? redondear_(d.cant, 3) : '',
              redondear_(d.valor, 0), d.ult, esNumero_(d.ultP) ? redondear_(d.ultP, 2) : '',
              esNumero_(med) ? redondear_(med, 2) : '', esNumero_(min) ? redondear_(min, 2) : '',
              esNumero_(max) ? redondear_(max, 2) : '', desv, nProv, estado]);
  }
  var h = hoja_(H.MAE);
  limpiarCuerpo_(h, FILA_DATOS, 15);
  if (out.length) h.getRange(FILA_DATOS, 1, out.length, 15).setValues(out);
  h.getRange(FILA_DATOS, 8, Math.max(out.length, 1), 1).setNumberFormat('yyyy-mm-dd');
  return out.length;
}

function reconstruirDiagnostico_(){
  var filas = leerBD_(), i;
  var c = { dupClave:0, refRota:0, sinPrecio:0, fueraBanda:0, sinUnidad:0, sinDoc:0,
            vencidas:0, overrides:0, reversas:0, sinUsuario:0, unDato:0, revisar:0,
            selloFalta:0, difSinObs:0, incoherente:0 };
  var vistos = {}, prodPrecios = {};
  var hoy = hoy_();
  for (i = 0; i < filas.length; i++){
    var f = filas[i];
    var id = txt_(f[B.ID_LINEA - 1]);
    if (vistos[id]) c.dupClave++; else vistos[id] = true;
    if (txt_(f[B.PRODUCTO - 1]) === '' || txt_(f[B.ID_DOC - 1]) === '') c.refRota++;
    var h12 = txt_(f[B.HALLAZGOS - 1]);
    var p = num_(f[B.PRECIO_UNIT - 1]);
    if (!esNumero_(p) || p <= 0) c.sinPrecio++;
    if (tiene_(h12, 'H12')) c.fueraBanda++;
    if (txt_(f[B.UNIDAD - 1]) === '') c.sinUnidad++;
    if (txt_(f[B.DOCUMENTO - 1]) === '') c.sinDoc++;
    var ve = f[B.VENCIMIENTO - 1];
    var fr = f[B.FECHA - 1];
    /* Vencido AL RECIBIRLO. Comparar contra hoy marcaría como hallazgo toda
       remisión antigua y un aviso falso hace que dejen de creerse todos. */
    if (ve instanceof Date && fr instanceof Date && ve < fr && esActiva_(f)) c.vencidas++;
    if (clave_(f[B.OVERRIDE - 1]) === 'si') c.overrides++;
    if (clave_(f[B.ESTADO - 1]) === 'reversada') c.reversas++;
    if (clave_(f[B.ESTADO - 1]) === 'revisar') c.revisar++;
    if (txt_(f[B.RECIBE - 1]) === '') c.sinUsuario++;
    var deCaptura = clave_(f[B.ORIGEN - 1]) === 'captura';
    if (deCaptura && tiene_(h12, 'H10')) c.selloFalta++;
    if (deCaptura && tiene_(h12, 'H08')) c.difSinObs++;
    if (deCaptura && (tiene_(h12, 'H06') || tiene_(h12, 'H07'))) c.incoherente++;
    if (computable_(f) && esNumero_(p) && p > 0){
      var k = clave_(f[B.PROD_NORM - 1]);
      prodPrecios[k] = (prodPrecios[k] || 0) + 1;
    }
  }
  var kk;
  for (kk in prodPrecios){ if (prodPrecios.hasOwnProperty(kk) && prodPrecios[kk] === 1) c.unDato++; }

  /* v1.1 · INT-16 e INT-17: duplicados abiertos de nombre */
  var dupProd = contarMarcados_(H.CAT, 9);
  var dupProv = contarMarcados_(H.PRV, 8);

  var F = [
    ['INT-01','Claves ID_LINEA duplicadas', c.dupClave, 0, 'Cada línea debe tener clave única. >0 = el motor escribió dos veces.'],
    ['INT-02','Integridad referencial rota', c.refRota, 0, 'Líneas sin producto o sin documento padre.'],
    ['INT-03','Líneas sin precio unitario', c.sinPrecio, 0, 'Sin precio no hay costeo: recupéralo de la factura original.'],
    ['INT-04','Precios fuera de banda (H12)', c.fueraBanda, 0, 'Casi siempre precio de paquete metido como unitario.'],
    ['INT-05','Líneas sin unidad normalizada', c.sinUnidad, 0, 'Sin unidad la banda de precio no puede compararse.'],
    ['INT-06','Líneas sin documento soporte', c.sinDoc, 0, 'Recepción sin remisión ni factura: no es defendible ante Gerencia.'],
    ['INT-07','Productos recibidos ya vencidos', c.vencidas, 0, 'Llegaron con fecha de vencimiento anterior a la recepción: escalar al proveedor.'],
    ['INT-08','Guardados forzados (override)', c.overrides, '—', 'Estado reconocido: cada uno tiene motivo firmado en BITÁCORA.'],
    ['INT-09','Líneas reversadas', c.reversas, '—', 'Estado reconocido: la original queda intacta con su reversa.'],
    ['INT-10','Líneas sin responsable que recibe', c.sinUsuario, 0, 'Sin responsable no hay a quién preguntar.'],
    ['INT-11','Productos con un solo dato de precio', c.unDato, '—', 'Estado reconocido: la capa 2 de banda necesita dos.'],
    ['INT-12','Líneas migradas en estado REVISAR', c.revisar, 0, 'Histórico marcado: ciérralas contra la factura original.'],
    ['INT-13','Licores/vinos sin sello marcado (H10)', c.selloFalta, 0, 'Solo recepciones capturadas con el motor. El control antimerma más caro del bar.'],
    ['INT-14','Diferencias sin observación (H08)', c.difSinObs, 0, 'Solo recepciones capturadas con el motor: faltante o excedente sin explicación.'],
    ['INT-15','Cantidad o condición incoherente (H06/H07)', c.incoherente, 0, 'Solo recepciones capturadas con el motor: recibido 0 marcado OK, o "no llegó" con cantidad.'],
    ['INT-16','Productos con posible duplicado abierto', dupProd, 0, 'Dos nombres para el mismo insumo parten su histórico de precio en dos.'],
    ['INT-17','Proveedores con posible duplicado abierto', dupProv, 0, 'Unifícalos con ⚙ Di Égida Bar → Fusionar proveedores.']
  ];
  var out = [], j, activos = 0;
  for (j = 0; j < F.length; j++){
    var r = F[j];
    var meta = r[3];
    var estado;
    if (meta === '—') estado = 'RECONOCIDO';
    else if (r[2] <= meta) estado = 'OK';
    else { estado = 'ABIERTO'; activos++; }
    out.push([r[0], r[1], r[2], meta, estado, r[4]]);
  }
  var h = hoja_(H.DIA);
  limpiarCuerpo_(h, FILA_DATOS, 6);
  if (out.length) h.getRange(FILA_DATOS, 1, out.length, 6).setValues(out);
  /* Un control que nunca puede llegar a cero se cierra como estado reconocido
     y sale del contador activo: un aviso falso hace que dejen de creerse todos. */
  return { filas: out, activos: activos, c: c };
}

/* ============================== 12 · TABLERO ============================== */

function reconstruirTablero_(){
  var filas = leerBD_(), i;
  var nDocs = {}, valor = 0, lineas = 0, sinPrecio = 0, fueraBanda = 0, revisar = 0, overrides = 0;
  var ultima = '', prov = {};
  var hace30 = new Date(hoy_().getTime() - 30 * 86400000);
  var valor30 = 0;
  for (i = 0; i < filas.length; i++){
    var f = filas[i];
    if (!esActiva_(f)) continue;
    lineas++;
    nDocs[txt_(f[B.ID_DOC - 1])] = true;
    var v = num_(f[B.VALOR_TOTAL - 1]); if (esNumero_(v)) valor += v;
    var fe = f[B.FECHA - 1];
    if (fe instanceof Date){
      if (!(ultima instanceof Date) || fe > ultima) ultima = fe;
      if (fe >= hace30 && esNumero_(v)) valor30 += v;
    }
    var p = num_(f[B.PRECIO_UNIT - 1]); if (!esNumero_(p) || p <= 0) sinPrecio++;
    if (tiene_(txt_(f[B.HALLAZGOS - 1]), 'H12')) fueraBanda++;
    if (clave_(f[B.ESTADO - 1]) === 'revisar') revisar++;
    if (clave_(f[B.OVERRIDE - 1]) === 'si') overrides++;
    var pr = txt_(f[B.PROVEEDOR - 1]); if (pr !== '') prov[pr] = true;
  }
  var nD = 0, kd; for (kd in nDocs){ if (nDocs.hasOwnProperty(kd) && kd !== '') nD++; }
  var nP = 0, kp; for (kp in prov){ if (prov.hasOwnProperty(kp)) nP++; }
  var cob = lineas ? redondear_((lineas - sinPrecio) / lineas * 100, 1) : 0;

  var kpis = [
    ['Recepciones registradas', nD],
    ['Líneas activas en el libro mayor', lineas],
    ['Proveedores distintos', nP],
    ['Última recepción', ultima],
    ['Valor total registrado', redondear_(valor, 0)],
    ['Valor recibido últimos 30 días', redondear_(valor30, 0)],
    ['Cobertura de costeo (líneas con precio)', cob / 100],
    ['Líneas sin precio', sinPrecio],
    ['Líneas con precio fuera de banda', fueraBanda],
    ['Líneas en estado REVISAR', revisar],
    ['Guardados forzados', overrides]
  ];
  var h = hoja_(H.TAB);
  limpiarCuerpo_(h, 5, 2);
  h.getRange(5, 1, kpis.length, 2).setValues(kpis);
  h.getRange(8, 2).setNumberFormat('yyyy-mm-dd');
  h.getRange(9, 2, 2, 1).setNumberFormat('$#,##0');
  h.getRange(11, 2).setNumberFormat('0.0%');

  /* Hallazgos críticos, derivados — nunca texto escrito a mano */
  var diag = reconstruirDiagnostico_();
  var crit = [], j;
  for (j = 0; j < diag.filas.length; j++){
    if (diag.filas[j][4] === 'ABIERTO') crit.push([diag.filas[j][0], diag.filas[j][1], diag.filas[j][2], diag.filas[j][5]]);
  }
  var fila0 = 5 + kpis.length + 2;
  h.getRange(fila0, 1).setValue('HALLAZGOS ABIERTOS').setFontWeight('bold').setFontColor(COL.VINO);
  limpiarCuerpo_(h, fila0 + 1, 4);
  if (crit.length){
    h.getRange(fila0 + 1, 1, crit.length, 4).setValues(crit);
  } else {
    h.getRange(fila0 + 1, 1).setValue('✔ Sin hallazgos abiertos.').setFontColor(COL.OK).setFontWeight('bold');
  }
  return kpis;
}

function limpiarCuerpo_(h, filaInicio, nCols){
  var maxF = h.getMaxRows();
  if (maxF < filaInicio) return;
  var nC = Math.min(nCols, h.getMaxColumns());
  h.getRange(filaInicio, 1, maxF - filaInicio + 1, nC).clearContent();
}

/*  Todo aviso se deriva de una comprobación que corre en cada recálculo:
    aparece cuando el problema existe y desaparece al corregirlo.
    Nunca texto de diagnóstico escrito a mano. */
function refrescarAvisos_(){
  var diag = reconstruirDiagnosticoLigero_();
  var pend = [];
  if (diag.c.sinPrecio > 0)   pend.push(diag.c.sinPrecio + ' línea(s) sin precio');
  if (diag.c.fueraBanda > 0)  pend.push(diag.c.fueraBanda + ' precio(s) fuera de banda');
  if (diag.c.revisar > 0)     pend.push(diag.c.revisar + ' línea(s) del histórico por revisar');
  if (diag.c.sinUnidad > 0)   pend.push(diag.c.sinUnidad + ' sin unidad');
  if (diag.c.selloFalta > 0)  pend.push(diag.c.selloFalta + ' licor/vino sin sello marcado');
  if (diag.c.vencidas > 0)    pend.push(diag.c.vencidas + ' recibido(s) ya vencido(s)');

  var dupProd = contarMarcados_(H.CAT, 9);
  var dupProv = contarMarcados_(H.PRV, 8);
  if (dupProd > 0) pend.push(dupProd + ' producto(s) con posible duplicado');
  if (dupProv > 0) pend.push(dupProv + ' proveedor(es) con posible duplicado');

  var correos = correosAlerta_();
  if (correos.length === 0) pend.push('CONFIG sin correos de alerta');

  var texto = pend.length ? ('⚠ PENDIENTES: ' + pend.join('  ·  ')) : '✔ Sin pendientes abiertos.';
  var nivelBg = pend.length ? COL.AVISOBG : COL.OKBG;
  var nivelFg = pend.length ? COL.AVISO : COL.OK;

  var hojas = [H.TAB, H.MAE, H.CAT, H.PRV, H.BAN, H.DIA];
  var i;
  for (i = 0; i < hojas.length; i++){
    try {
      var h = hojaOpcional_(hojas[i]);

function reconstruirDiagnosticoLigero_(){
  var filas = leerBD_(), i;
  var c = { sinPrecio:0, fueraBanda:0, revisar:0, sinUnidad:0, selloFalta:0, vencidas:0 };
  var hoy = hoy_();
  for (i = 0; i < filas.length; i++){
    var f = filas[i];
    if (!esActiva_(f)) continue;
    var p = num_(f[B.PRECIO_UNIT - 1]);
    if (!esNumero_(p) || p <= 0) c.sinPrecio++;
    var hall = txt_(f[B.HALLAZGOS - 1]);
    if (tiene_(hall, 'H12')) c.fueraBanda++;
    if (tiene_(hall, 'H10')) c.selloFalta++;
    if (txt_(f[B.UNIDAD - 1]) === '') c.sinUnidad++;
    if (clave_(f[B.ESTADO - 1]) === 'revisar') c.revisar++;

    var fr = f[B.FECHA - 1];
    if (ve instanceof Date && fr instanceof Date && ve < fr) c.vencidas++;
  }
  return { c: c };
}

/*  El libro mayor es append-only: una corrección NUNCA edita ni borra la línea original.
    Marca ESTADO_LINEA = REVERSADA con fecha, responsable y motivo, y la original queda
    intacta. Si un precio de hace dos meses se puede cambiar sin rastro, ningún margen
    calculado sobre ese histórico es defendible ante Gerencia. */
function reversarLinea(){
  var ui;
  try { ui = SpreadsheetApp.getUi(); }
  catch(e){ toast_('Reversar solo funciona desde computador.', CODIGO); return; }

  var r1 = ui.prompt('Reversar línea', 'ID_LINEA a reversar (ej. BAR-20260909-001-003):', ui.ButtonSet.OK_CANCEL);
  if (r1.getSelectedButton() !== ui.Button.OK) return;
  var id = txt_(r1.getResponseText());
  if (id === '') return;

  var r2 = ui.prompt('Motivo de la reversa', 'Motivo (mínimo ' + cfgNum_('MIN_MOTIVO_OVERRIDE', 15) + ' caracteres):', ui.ButtonSet.OK_CANCEL);
  if (r2.getSelectedButton() !== ui.Button.OK) return;
  var motivo = txt_(r2.getResponseText());
  if (motivo.length < cfgNum_('MIN_MOTIVO_OVERRIDE', 15)){
    ui.alert('Motivo demasiado corto. No se reversó nada.');
    return;
  }

  var lock = LockService.getScriptLock();
  try { lock.waitLock(30000); } catch(e){ ui.alert('Otra operación en curso.'); return; }
  try {
    var h = hoja_(H.BD);
    var n = h.getLastRow();
    if (n < 2){ ui.alert('El libro mayor está vacío.'); return; }
    var d = h.getRange(2, 1, n - 1, BD_ULT_COL).getValues();
    var i, encontrada = -1;
    for (i = 0; i < d.length; i++){ if (clave_(d[i][B.ID_LINEA - 1]) === clave_(id)) { encontrada = i; break; } }
    if (encontrada < 0){ ui.alert('No existe la línea ' + id + '.'); return; }
    if (clave_(d[encontrada][B.ESTADO - 1]) === 'reversada'){ ui.alert('Esa línea ya está reversada.'); return; }

    var refRev = 'REV-' + Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyyMMdd-HHmmss');
    h.getRange(encontrada + 2, B.ESTADO).setValue('REVERSADA');
    h.getRange(encontrada + 2, B.REF_REVERSA).setValue(refRev);
    h.getRange(encontrada + 2, B.MOTIVO).setValue(motivo);
    bitacora_('REVERSA', id + ' → ' + refRev, motivo);
    try { recalcularTodo_(); } catch(eR){}
    ui.alert('Línea ' + id + ' marcada como REVERSADA. La original queda intacta.');
  } catch(e){
    try { ui.alert('Error: ' + (e && e.message ? e.message : e)); } catch(e2){}
  } finally { try { lock.releaseLock(); } catch(e3){} }
}

/* Fecha de respaldo tomada del nombre de la hoja cuando la celda tenía basura
   (caso real: la hoja 31072026 traía «32-08/2026»). */
function fechaDeNombreHoja_(nombre){
  var k = clave_(nombre).replace(/ /g, '');
  var m = k.match(/^(\d{2})(\d{2})(\d{4})$/);
  if (m) return new Date(Number(m[3]), Number(m[2]) - 1, Number(m[1]));
  m = k.match(/^(\d{1,2})[^0-9]?(\d{2})[^0-9]?(\d{4})$/);
  if (m) return new Date(Number(m[3]), Number(m[2]) - 1, Number(m[1]));
  return '';
}

        contadorDoc[clv] = { base: base, id: base + '-' + pad3_(usados + 1), n: 0 };
      }
      contadorDoc[clv].n++;
      var idDoc = contadorDoc[clv].id;

      var f = [];
      f[B.ID_LINEA - 1]     = idDoc + '-' + pad3_(contadorDoc[clv].n);
      f[B.ID_DOC - 1]       = idDoc;
      f[B.FECHA - 1]        = (fecha instanceof Date) ? fecha : '';
      f[B.PROVEEDOR - 1]    = proveedor;
      f[B.PROV_NORM - 1]    = clave_(proveedor);
      f[B.DOCUMENTO - 1]    = documento;
      f[B.PRODUCTO - 1]     = producto;
      f[B.PROD_NORM - 1]    = clave_(producto);
      f[B.CATEGORIA - 1]    = categoria;
      f[B.PRESENTACION - 1] = presentacion;
      f[B.UNIDAD - 1]       = unidad;
      f[B.CANT_PED - 1]     = esNumero_(pedida) ? pedida : '';
      f[B.CANT_REC - 1]     = esNumero_(recibida) ? recibida : '';
      f[B.DIFERENCIA - 1]   = (esNumero_(pedida) && esNumero_(recibida)) ? (recibida - pedida) : '';
      f[B.PRECIO_UNIT - 1]  = esNumero_(precio) ? precio : '';
      f[B.VALOR_TOTAL - 1]  = esNumero_(totalCalc) ? totalCalc : '';
      f[B.CONDICION - 1]    = condicion;
      f[B.SELLO - 1]        = sello;
      f[B.VENCIMIENTO - 1]  = (venc instanceof Date) ? venc : txt_(r[15]);
      f[B.LOTE - 1]         = lote;
      f[B.OBSERVACIONES - 1]= obs;
      f[B.RECIBE - 1]       = txt_(r[5]);
      f[B.USUARIO - 1]      = '(migración)';
      f[B.TIMESTAMP - 1]    = ahora;
      f[B.OVERRIDE - 1]     = 'NO';
      f[B.MOTIVO - 1]       = '';
      f[B.ESTADO - 1]       = hall.length ? 'REVISAR' : 'ACTIVA';
      f[B.REF_REVERSA - 1]  = '';
      f[B.ORIGEN - 1]       = 'MIGRACION (' + hojaOrig + ')';
      f[B.HALLAZGOS - 1]    = hall.join(';');
      var z;
      for (z = 0; z < BD_ULT_COL; z++){ if (f[z] === undefined) f[z] = ''; }
      if (hall.length) nRevisar++;
      salida.push(f);
    }

    if (salida.length) hBD.getRange(hBD.getLastRow() + 1, 1, salida.length, BD_ULT_COL).setValues(salida);
    bitacora_('MIGRACIÓN', salida.length + ' línea(s) del histórico · ' + nRevisar + ' en REVISAR', '');
    try { recalcularTodo_(); } catch(eR){}
    avisarUI_(CODIGO, 'Histórico migrado: ' + salida.length + ' línea(s), ' + nRevisar +
                      ' marcada(s) como REVISAR con su código de hallazgo.\n\n' +
                      'Revísalas en DIAGNÓSTICO (INT-12) y ciérralas contra la factura original.');
    return { total: salida.length, revisar: nRevisar };
  } catch(e){
    avisarUI_(CODIGO, 'Error migrando: ' + (e && e.message ? e.message : e));
    bitacora_('ERROR MIGRACIÓN', (e && e.message ? e.message : String(e)), '');
  } finally { try { lock.releaseLock(); } catch(e3){} }
}

/* Unifica variantes de nombre de proveedor ya escritas en el libro mayor. */
function depurarNombres(){
  var lock = LockService.getScriptLock();
  try { lock.waitLock(30000); } catch(e){ return; }
  try {
    var h = hoja_(H.BD);
    var n = h.getLastRow();
    if (n < 2){ avisarUI_(CODIGO, 'El libro mayor está vacío.'); return; }
    var rng = h.getRange(2, B.PROVEEDOR, n - 1, 2);
    var d = rng.getValues();
    var cambios = 0, i;
    for (i = 0; i < d.length; i++){
      var canon = proveedorCanonico_(d[i][0]);
      if (canon !== txt_(d[i][0])){ d[i][0] = canon; cambios++; }
      d[i][1] = clave_(canon);
    }
    rng.setValues(d);
    bitacora_('DEPURAR NOMBRES', cambios + ' nombre(s) de proveedor unificado(s)', '');
    try { recalcularTodo_(); } catch(eR){}
    avisarUI_(CODIGO, cambios + ' nombre(s) de proveedor unificado(s).');
    return cambios;
  } catch(e){ avisarUI_(CODIGO, 'Error: ' + (e && e.message ? e.message : e)); }
  finally { try { lock.releaseLock(); } catch(e3){} }
}

function aplicarValidacionesCaptura_(h){

    return SpreadsheetApp.newDataValidation().requireValueInList(valores, true)
      .setAllowInvalid(!estricto).build();
  };
  h.getRange(i0, R.UNIDAD, nF, 1).setDataValidation(lista(UNIDADES, false));

    var hc = hojaOpcional_(H.CAT);
    if (hc){
      var rango = hc.getRange('B' + FILA_DATOS + ':B1000');
      var dv = SpreadsheetApp.newDataValidation().requireValueInRange(rango, true).setAllowInvalid(true)
        .setHelpText('Elige el producto de la lista. Escribir un nombre nuevo crea un producto nuevo en el catálogo.')

      lo que calcula el motor, y a 21 px de alto no se acierta la celda en un iPad.

    LA REGLA NUEVA, aprendible en diez segundos
      Fondo BLANCO con borde suave  →  aquí se escribe.
      Fondo GRIS sin borde propio   →  lo calcula el motor, no se toca (A, G, I).

    Y la rejilla se reduce a lo que sí significa algo:
      · separador horizontal entre filas, más marcado cada 5 para seguir el renglón
      · vertical solo en los tres cortes lógicos: identificación | cantidades | dinero
      · alto de fila a 30 px, que es el mínimo táctil real
      · cuadrícula nativa de Sheets apagada, para que deje de competir con la del formato */
function aplicarFormatoCaptura_(h){
  var i0 = C.FILA_INI, i1 = C.FILA_FIN, nF = i1 - i0 + 1;
  var SOL = SpreadsheetApp.BorderStyle.SOLID;
  var GRIS_CALC = '#F1EFEA';   /* columnas que calcula el motor */
  var LINEA_SUAVE = COL.BORDE; /* separador entre filas */
  var LINEA_CORTE = '#B9B1A2'; /* cortes verticales y cada 5 filas */

  /* ---------- Encabezado del documento ---------- */
  h.getRange('A1:N1').setBackground(COL.VINO).setFontColor('#FFFFFF').setFontSize(13)
   .setFontWeight('bold').setHorizontalAlignment('center').setVerticalAlignment('middle');
  h.setRowHeight(1, 34);
  h.getRange('A2:N2').setBackground(COL.HUESO).setFontColor(COL.GRIS).setFontSize(9).setFontStyle('italic');
  h.getRange('H2:N2').setHorizontalAlignment('right');
  h.getRange('A4:N4').setBackground(COL.ORO).setFontColor('#FFFFFF').setFontWeight('bold').setFontSize(10);
  h.setRowHeight(4, 26);

  h.getRange('A5:A7').setFontWeight('bold'); h.getRange('G5:G7').setFontWeight('bold');
  h.getRange('D5:F7').setBackground('#FFFFFF')
   .setBorder(true, true, true, true, false, true, LINEA_SUAVE, SOL);
  h.getRange('J5:N7').setBackground('#FFFFFF')
   .setBorder(true, true, true, true, false, true, LINEA_SUAVE, SOL);
  h.setRowHeights(5, 3, 26);

  h.getRange('A8:N8').setFontSize(10).setFontWeight('bold').setHorizontalAlignment('left')
   .setVerticalAlignment('middle').setWrap(true).setBackground(COL.AVISOBG).setFontColor(COL.AVISO);
  h.setRowHeight(8, 30);

  /* ---------- Encabezado de la tabla ---------- */
  h.getRange(9, 1, 1, C.ULT_COL).setBackground(COL.TINTA).setFontColor('#FFFFFF').setFontWeight('bold')
   .setFontSize(9).setWrap(true).setHorizontalAlignment('center').setVerticalAlignment('middle');
  h.setRowHeight(9, 38);

  /* ---------- Cuerpo de la tabla ---------- */
  var cuerpo = h.getRange(i0, 1, nF, C.ULT_COL);
  cuerpo.setFontSize(10).setVerticalAlignment('middle');
  try { cuerpo.setWrapStrategy(SpreadsheetApp.WrapStrategy.CLIP); } catch(e){}
  h.setRowHeights(i0, nF, 30);

  /* fondos: blanco donde se digita, gris donde calcula el motor */
  cuerpo.setBackground('#FFFFFF');
  h.getRange(i0, R.NUM,        nF, 1).setBackground(GRIS_CALC).setFontColor(COL.GRIS);
  h.getRange(i0, R.DIFERENCIA, nF, 1).setBackground(GRIS_CALC);
  h.getRange(i0, R.TOTAL,      nF, 1).setBackground(GRIS_CALC);

  /* bordes: primero se borra la rejilla entera, luego se dibuja solo lo que informa */
  cuerpo.setBorder(false, false, false, false, false, false, null, null);
  cuerpo.setBorder(true, true, true, true, false, true, LINEA_SUAVE, SOL);

  /* tres cortes verticales: fin de identificación (D) · fin de cantidades (G) · fin de dinero (I) */
  h.getRange(i0, R.UNIDAD,     nF, 1).setBorder(null, null, null, true, null, null, LINEA_CORTE, SOL);
  h.getRange(i0, R.DIFERENCIA, nF, 1).setBorder(null, null, null, true, null, null, LINEA_CORTE, SOL);
  h.getRange(i0, R.TOTAL,      nF, 1).setBorder(null, null, null, true, null, null, LINEA_CORTE, SOL);

  /* refuerzo cada 5 filas para no perder el renglón hasta la columna N */
  var r;
  for (r = i0 + 4; r < i1; r += 5){
    h.getRange(r, 1, 1, C.ULT_COL).setBorder(null, null, true, null, null, null, LINEA_CORTE, SOL);
  }

  h.getRange(i0, R.PRECIO,   nF, 2).setNumberFormat('$#,##0.00');
  h.getRange(i0, R.CANT_PED, nF, 3).setNumberFormat('#,##0.###');
  h.getRange(i0, R.NUM,      nF, 1).setHorizontalAlignment('center');
  h.getRange(i0, R.SELLO,    nF, 1).setHorizontalAlignment('center');

  /* ---------- Totales ---------- */
  h.getRange(C.TOTALES, 1, 1, C.ULT_COL).setBackground(COL.HUESO).setFontWeight('bold')
   .setBorder(true, true, true, true, false, false, COL.TINTA, SOL);
  h.setRowHeight(C.TOTALES, 30);
  h.getRange(C.TOTALES, R.TOTAL).setNumberFormat('$#,##0');
  h.getRange(C.TOTALES, R.CONDICION).setFontWeight('normal').setFontStyle('italic')
   .setFontSize(9).setFontColor(COL.GRIS);

  /* ---------- Cierre y firmas ---------- */
  h.getRange('A47').setFontWeight('bold');
  h.getRange('H47').setFontWeight('bold');
  h.getRange('D47:F47').setBackground('#FFFFFF')
   .setBorder(true, true, true, true, false, false, LINEA_SUAVE, SOL);
  h.getRange('H48:N49').setBackground('#FFFFFF')
   .setBorder(true, true, true, true, false, false, LINEA_SUAVE, SOL);

  h.getRange('A50').setFontWeight('bold'); h.getRange('E50').setFontWeight('bold'); h.getRange('J50').setFontWeight('bold');
  h.setRowHeights(51, 2, 24);
  /* la línea sobre la que se firma: antes no existía ninguna */
  h.getRange(53, 1, 1, 4).setBorder(null, null, true, null, null, null, COL.TINTA, SOL);
  h.getRange(53, 5, 1, 5).setBorder(null, null, true, null, null, null, COL.TINTA, SOL);
  h.getRange(53, 10, 1, 5).setBorder(null, null, true, null, null, null, COL.TINTA, SOL);
  h.getRange(54, 1, 1, C.ULT_COL).setFontSize(8).setFontColor(COL.GRIS);

  /* ---------- Controles ---------- */
  h.getRange('A56:B56').setFontWeight('bold').setFontColor(COL.VINO);
  h.getRange('A58:B58').setFontWeight('bold').setFontColor(COL.AZUL);
  h.getRange(C.CHK_GUARDAR).setHorizontalAlignment('left');
  h.getRange(C.CHK_NUEVA).setHorizontalAlignment('left');
  h.setRowHeights(56, 1, 30);
  h.setRowHeights(58, 1, 30);
  h.getRange('D56:G56').setFontSize(9).setFontColor(COL.GRIS).setHorizontalAlignment('right');
  h.getRange('H56:N56').setBackground('#FFFFFF')
   .setBorder(true, true, true, true, false, false, LINEA_SUAVE, SOL);
  h.getRange('A57:N57').setWrap(true).setVerticalAlignment('middle')
   .setBackground(COL.OKBG).setFontColor(COL.OK).setFontWeight('bold');
  h.setRowHeight(57, 54);

  /* ---------- Anchos ---------- */
  var anchos = [40, 235, 112, 84, 78, 84, 86, 100, 104, 108, 80, 104, 98, 250];
  var i;
  for (i = 0; i < anchos.length; i++){ h.setColumnWidth(i + 1, anchos[i]); }

  /* ---------- Semáforos ---------- */
  var rangoG = h.getRange(i0, R.DIFERENCIA, nF, 1);
  var reglas = [];
  reglas.push(SpreadsheetApp.newConditionalFormatRule().whenNumberLessThan(0)
    .setBackground(COL.MALBG).setFontColor(COL.MAL).setBold(true).setRanges([rangoG]).build());
  reglas.push(SpreadsheetApp.newConditionalFormatRule().whenNumberGreaterThan(0)
    .setBackground(COL.AVISOBG).setFontColor(COL.AVISO).setBold(true).setRanges([rangoG]).build());
  reglas.push(SpreadsheetApp.newConditionalFormatRule().whenNumberEqualTo(0)
    .setBackground(COL.OKBG).setFontColor(COL.OK).setRanges([rangoG]).build());
  var rangoJ = h.getRange(i0, R.CONDICION, nF, 1);
  reglas.push(SpreadsheetApp.newConditionalFormatRule().whenTextContains('No llegó')
    .setBackground(COL.MALBG).setFontColor(COL.MAL).setRanges([rangoJ]).build());
  reglas.push(SpreadsheetApp.newConditionalFormatRule().whenTextContains('Dañado')
    .setBackground(COL.AVISOBG).setFontColor(COL.AVISO).setRanges([rangoJ]).build());
  var rangoK = h.getRange(i0, R.SELLO, nF, 1);
  reglas.push(SpreadsheetApp.newConditionalFormatRule().whenTextEqualTo('NO')
    .setBackground(COL.MALBG).setFontColor(COL.MAL).setBold(true).setRanges([rangoK]).build());
  /* el renglón con producto pero sin precio se resalta: es el error más caro del bar */
  reglas.push(SpreadsheetApp.newConditionalFormatRule()
    .whenFormulaSatisfied('=AND($B' + i0 + '<>"",$H' + i0 + '="")')
    .setBackground('#FFFBF0')
    .setRanges([h.getRange(i0, R.PRODUCTO, nF, 1)]).build());
  h.setConditionalFormatRules(reglas);

  try { h.setFrozenRows(9); } catch(e){}
  try { h.setFrozenColumns(2); } catch(e){}
  /* la cuadrícula nativa de Sheets deja de competir con la del formato */
  try { h.setHiddenGridlines(true); } catch(e){}
}

/* Las 12 hojas por fecha quedan intactas, solo ocultas. No se borra nada. */
function archivarHojasViejas_(){
  var libro = ss_();
  var hojas = libro.getSheets(), i, n = 0;
  var conocidas = {};
  for (i = 0; i < DEF_HOJAS.length; i++){ conocidas[DEF_HOJAS[i].n] = true; }
  activarHojaSegura_();
  for (i = 0; i < hojas.length; i++){
    var nom = hojas[i].getName();
    if (conocidas[nom]) continue;
    try { hojas[i].setTabColor('#9E9E9E'); } catch(e){}
    if (ocultarSeguro_(nom)) n++;
  }
  return n;
}

function crearDisparadores_(){
  var libro = ss_();
  var existentes = ScriptApp.getProjectTriggers(), i;
  for (i = 0; i < existentes.length; i++){
    var f = existentes[i].getHandlerFunction();
    if (f === 'alEditar_' || f === 'enviarAlertaDiaria'){
      try { ScriptApp.deleteTrigger(existentes[i]); } catch(e){}
    }
  }
  ScriptApp.newTrigger('alEditar_').forSpreadsheet(libro).onEdit().create();
  ScriptApp.newTrigger('enviarAlertaDiaria').timeBased().atHour(8).everyDays(1).create();
  return 2;
}

function protegerHojas_(){
  var i, n = 0;
  for (i = 0; i < HOJAS_TECNICAS.length; i++){
    var h = hojaOpcional_(HOJAS_TECNICAS[i]);
    if (!h) continue;
    var libres = null;
    if (HOJAS_TECNICAS[i] === 'CONFIG') libres = [h.getRange('B2:B100')];
    if (protegerSeguro_(h, 'SGI ' + CODIGO + ' — no editar a mano', libres)) n++;
  }
  return n;
}

  try {
    var t = ScriptApp.getProjectTriggers();
    if (!t.length) L.push('  ✕ Ninguno. Ejecuta instalar.');
    for (i = 0; i < t.length; i++){ L.push('  · ' + t[i].getHandlerFunction() + '  (' + t[i].getEventType() + ')'); }
  } catch(e3){ L.push('  no verificable: ' + (e3 && e3.message ? e3.message : e3)); }
  L.push('');

  L.push('CASILLAS DE CONTROL:');
  try {
    var hc = hoja_(H.CAP);
    L.push('  · Guardar ' + C.CHK_GUARDAR + ' = ' + hc.getRange(C.CHK_GUARDAR).getValue());
    L.push('  · Nueva ' + C.CHK_NUEVA + ' = ' + hc.getRange(C.CHK_NUEVA).getValue());
    L.push('  · Banda de avisos = ' + String(hc.getRange(C.AVISOS).getDisplayValue()).substring(0, 120));
  } catch(e4){ L.push('  ✕ ' + (e4 && e4.message ? e4.message : e4)); }
  L.push('');

  L.push('DATOS:');
  var filas = leerBD_();
  var act = 0, rev = 0, revi = 0;
  for (i = 0; i < filas.length; i++){
    var e = clave_(filas[i][B.ESTADO - 1]);
    if (e === 'activa') act++; else if (e === 'reversada') rev++; else if (e === 'revisar') revi++;
  }
  L.push('  · Líneas en el libro mayor: ' + filas.length + '  (ACTIVA ' + act + ' · REVISAR ' + revi + ' · REVERSADA ' + rev + ')');
  var hcat = hojaOpcional_(H.CAT);
  L.push('  · Productos en catálogo: ' + (hcat ? Math.max(hcat.getLastRow() - FILA_DATOS + 1, 0) : 0));
  var hprv = hojaOpcional_(H.PRV);
  L.push('  · Proveedores en catálogo: ' + (hprv ? Math.max(hprv.getLastRow() - FILA_DATOS + 1, 0) : 0) +
         (hprv ? '' : '  ✕ falta CAT_PROVEEDORES: ejecuta aplicarParche11'));
  L.push('  · Lista de proveedores cerrada: ' + (cfgSi_('PROVEEDOR_LISTA_CERRADA', 'SI') ? 'SÍ' : 'NO'));
  var hban = hojaOpcional_(H.BAN);
  L.push('  · Bandas calculadas: ' + (hban ? Math.max(hban.getLastRow() - FILA_DATOS + 1, 0) : 0));
  L.push('');

  L.push('CORREOS DE ALERTA: ' + (correosAlerta_().length ? correosAlerta_().join(', ') : '✕ ninguno configurado en CONFIG'));
  L.push('AVISOS VIVOS: ' + probarAvisos(true));

  var texto = L.join('\n');
  avisarUI_('Diagnóstico técnico', texto);
  bitacora_('DIAGNÓSTICO TÉCNICO', 'Motor ' + motor + ' · ' + filas.length + ' línea(s)', '');
  return texto;
}

/* Corre los bloques por separado y dice cuál falla. */
function probarAvisos(silencioso){
  var pasos = [];
  pasoSeguro_(pasos, 'Leer libro mayor',        function(){ leerBD_(); });
  pasoSeguro_(pasos, 'Calcular bandas',         function(){ calcularBandas_(); });
  pasoSeguro_(pasos, 'Diagnóstico ligero',      function(){ reconstruirDiagnosticoLigero_(); });
  pasoSeguro_(pasos, 'Leer catálogo de proveedores', function(){ proveedoresCatalogo_(); });
  pasoSeguro_(pasos, 'Leer correos de CONFIG',  function(){ correosAlerta_(); });
  pasoSeguro_(pasos, 'Escribir bandas de aviso',function(){ refrescarAvisos_(); });
  var r = pasos.join('  |  ');
  if (!silencioso) avisarUI_('Prueba de avisos', pasos.join('\n'));
  return r;
}

/* Reporte corto en texto, para diagnosticar sin mover el archivo. */
function reporteEstado(){
  var filas = leerBD_(), i;
  var lig = reconstruirDiagnosticoLigero_();
  var docs = {}, valor = 0;
  for (i = 0; i < filas.length; i++){
    if (!esActiva_(filas[i])) continue;
    docs[txt_(filas[i][B.ID_DOC - 1])] = true;
    var v = num_(filas[i][B.VALOR_TOTAL - 1]);
    if (esNumero_(v)) valor += v;
  }
  var nd = 0, k; for (k in docs){ if (docs.hasOwnProperty(k) && k !== '') nd++; }
  var t = CODIGO + ' v' + VERSION + '  ·  ' + fmtFecha_(hoy_()) + '\n' +
          nd + ' recepciones · ' + filas.length + ' líneas · $' + Math.round(valor) + ' registrados\n' +
          'Proveedores en catálogo: ' + proveedoresCatalogo_().length + '\n' +
          'Pendientes: ' + lig.c.sinPrecio + ' sin precio · ' + lig.c.fueraBanda + ' fuera de banda · ' +
          lig.c.revisar + ' por revisar · ' + lig.c.sinUnidad + ' sin unidad · ' +
          lig.c.selloFalta + ' sin sello · ' + lig.c.vencidas + ' vencidos';
  avisarUI_('Estado del formato', t);
  return t;
}
